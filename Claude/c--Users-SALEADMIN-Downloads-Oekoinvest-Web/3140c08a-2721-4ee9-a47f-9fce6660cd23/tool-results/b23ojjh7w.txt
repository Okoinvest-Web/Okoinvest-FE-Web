const mio = (v) => `${fmt(v / 1e6, 1)} Mio. €`;

/** Spanne min–max auf einer Skala 0..skala, Marke bei „mittel“ */
function Spanne({ titel, min, mittel, max, skala, fuss, ton = "gruen" }) {
  const pct = (v) => `${((v / skala) * 100).toFixed(1)}%`;
  const farbe = ton === "gruen" ? "from-ov-500 to-ov-300" : "from-navy-400 to-navy-200";
  return (
    <div className="min-w-0">
      <div className="flex flex-wrap items-baseline justify-between gap-x-4 gap-y-1">
        <p className="text-[14px] font-semibold text-white/85">{titel}</p>
        <p className="ov-num font-display text-[22px] font-extrabold tracking-tight text-white md:text-[26px]">
          {mio(min)} – {mio(max)}
        </p>
      </div>
      <div className="relative mt-4 h-3 rounded-full bg-white/[0.08]" aria-hidden="true">
        <div className={`absolute inset-y-0 rounded-full bg-gradient-to-r ${farbe}`} style={{ left: pct(min), width: `calc(${pct(max)} - ${pct(min)})` }} />
        <div className="absolute -top-1.5 h-6 w-1 -translate-x-1/2 rounded-full bg-white shadow-[0_0_0_3px_rgba(3,18,43,0.9)]" style={{ left: pct(mittel) }} />
      </div>
      <div className="relative mt-2 h-5 text-[12px] text-white/55" aria-hidden="true">
        <span className="absolute left-0">0</span>
        <span className="ov-num absolute -translate-x-1/2 whitespace-nowrap font-semibold text-white/80" style={{ left: pct(mittel) }}>
          mittel {mio(mittel)}
        </span>
      </div>
      <p className="sr-only">Mittlere Annahme {mio(mittel)}.</p>
      <p className="mt-2 text-[13px] text-white/60">{fuss}</p>
    </div>
  );
}

function Container({ x, y, i }) {
  const w = 3.2;
  const d = 1.15;
  const h = 1.15;
  const f = quader(pts, x, y, 0, w, d, h);
  const rippen = Array.from({ length: 9 }, (_, k) => x + 0.25 + k * 0.34);
  return (
    <g>
      <polygon points={f.links} fill="#c4cfdc" />
      <polygon points={f.rechts} fill="#9fb0c4" />
      <polygon points={f.oben} fill="#eef2f7" />
      {rippen.map((rx) => (
        <line key={rx} x1={p(rx, y + d, 0.08)[0]} y1={p(rx, y + d, 0.08)[1]} x2={p(rx, y + d, h - 0.08)[0]} y2={p(rx, y + d, h - 0.08)[1]} stroke="#aebbcb" strokeWidth="1" />
      ))}
      {/* Statusleiste */}
      <polygon points={pts([x + 0.2, y + d, h - 0.22], [x + 1.4, y + d, h - 0.22], [x + 1.4, y + d, h - 0.3], [x + 0.2, y + d, h - 0.3])} fill={i === 1 ? "#ffc53d" : "#8cba58"} />
      {/* Lüfter oben */}
      <polygon points={pts([x + 2.3, y + 0.25, h], [x + 2.9, y + 0.25, h], [x + 2.9, y + 0.85, h], [x + 2.3, y + 0.85, h])} fill="#d3dbe6" />
    </g>
  );
}

function Kasten({ x, y, w, d, h, farben = ["#5b6b82", "#43526a", "#7b8aa0"] }) {
  const f = quader(pts, x, y, 0, w, d, h);
  return (
    <g>
      <polygon points={f.links} fill={farben[0]} />
      <polygon points={f.rechts} fill={farben[1]} />
      <polygon points={f.oben} fill={farben[2]} />
    </g>
  );
}

function Modulreihe({ x0, x1, y }) {
  const felder = [];
  for (let x = x0; x < x1 - 0.1; x += 0.9) felder.push(x);
  return (
    <g>
      {felder.map((x) => (
        <polygon key={x} points={pts([x, y + 0.7, 0.25], [x + 0.82, y + 0.7, 0.25], [x + 0.82, y, 0.75], [x, y, 0.75])} fill="#1f5aa1" stroke="#4a7cbd" strokeWidth="0.8" />
      ))}
    </g>
  );
}

function Szene() {
