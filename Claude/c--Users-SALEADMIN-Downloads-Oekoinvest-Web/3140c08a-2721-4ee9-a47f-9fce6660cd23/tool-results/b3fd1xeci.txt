// Browserfenster mit App-Navigation, Daten-Visualisierungen als präzises Inline-SVG (Tageskurve, Abweichungsbalken,
// Frist-Segmente). Die Hero-Komposition ist so gebaut, dass keine Ebene wichtige Inhalte einer anderen verdeckt.

/** Hülle: dekorativ (aria-hidden) oder – mit Beschreibung – ein Bild mit Alternativtext */
function BeispielBild({ beschreibung, className, children }) {
  if (beschreibung) {
    return (
      <div role="img" aria-label={beschreibung} className={className}>
        <div aria-hidden="true">{children}</div>
      </div>
    );
  }
  return <div aria-hidden="true" className={className}>{children}</div>;
}

/** Sichtbare Kennzeichnung fiktiver Werte (sun-700 auf sun-50 ≈ 6,4:1) */
function BeispielPlakette({ hell = false, className }) {
  return (
    <span
      className={cn(
        "inline-flex shrink-0 items-center rounded-full font-semibold uppercase leading-none",
        hell
          ? "bg-white/10 px-1.5 py-[3px] text-[8.5px] tracking-[0.14em] text-white/80 ring-1 ring-white/15"
          : "bg-sun-50 px-2 py-[3px] text-[9.5px] tracking-[0.12em] text-sun-700 ring-1 ring-sun-200",
        className
      )}
    >
      Beispiel
    </span>
  );
}

const GRUEN = [
  { von: 10, bis: 15, stark: true },
  { von: 8.5, bis: 10, stark: false },
];
const JETZT = 12.5; // Uhrzeit der Jetzt-Marke (Beispiel), passt zur Statusleiste 12:30
const prozent = (h) => `${(h / 24) * 100}%`;

// Tageskurve „Gemeinschaftsstrom laut Prognose“ (rein illustrativ, ohne Achsenwerte): Glockenform um die Mittagszeit
const KURVE_B = 240;
const KURVE_H = 44;
const kx = (h) => (h / 24) * KURVE_B;
const ky = (h) => {
  const a = Math.exp(-(((h - 12.8) / 3.1) ** 2));
  return +(KURVE_H - 3 - a * (KURVE_H - 9)).toFixed(2);
};
const KURVE_PUNKTE = Array.from({ length: 97 }, (_, i) => i / 4).map((h) => [+kx(h).toFixed(2), ky(h)]);
const KURVE_LINIE = "M" + KURVE_PUNKTE.map(([x, y]) => `${x} ${y}`).join(" L");
const KURVE_FLAECHE = `${KURVE_LINIE} L${KURVE_B} ${KURVE_H} L0 ${KURVE_H} Z`;

/** Tageskurve mit grünen Fenstern und Jetzt-Marke – Inline-SVG, skaliert scharf */
function GrueneStundenKurve() {
  return (
    <div>
      <div className="relative">
        <svg viewBox={`0 0 ${KURVE_B} ${KURVE_H}`} preserveAspectRatio="none" className="block h-[46px] w-full overflow-visible" aria-hidden="true">
          {GRUEN.map((g) => (
            <rect key={g.von} x={kx(g.von)} y="0" width={kx(g.bis) - kx(g.von)} height={KURVE_H} fill="#aed083" fillOpacity={g.stark ? 0.16 : 0.07} />
          ))}
          {[6, 12, 18].map((h) => (
            <line key={h} x1={kx(h)} x2={kx(h)} y1="0" y2={KURVE_H} stroke="#fff" strokeOpacity=".08" vectorEffect="non-scaling-stroke" />
          ))}
          <path d={KURVE_FLAECHE} fill="#aed083" fillOpacity=".2" />
          <path d={KURVE_LINIE} fill="none" stroke="#aed083" strokeWidth="1.6" strokeLinejoin="round" vectorEffect="non-scaling-stroke" />
          <line x1="0" x2={KURVE_B} y1={KURVE_H - 0.5} y2={KURVE_H - 0.5} stroke="#fff" strokeOpacity=".18" vectorEffect="non-scaling-stroke" />
          <line x1={kx(JETZT)} x2={kx(JETZT)} y1="0" y2={KURVE_H} stroke="#fff" strokeOpacity=".75" strokeDasharray="2 2" vectorEffect="non-scaling-stroke" />
        </svg>
        {/* Punkt auf der Kurve als HTML, damit er trotz preserveAspectRatio="none" rund bleibt */}
        <span
          className="absolute h-[7px] w-[7px] -translate-x-1/2 -translate-y-1/2 rounded-full bg-white shadow-[0_0_0_3px_rgba(174,208,131,0.35),0_0_10px_2px_rgba(255,255,255,0.45)]"
          style={{ left: prozent(JETZT), top: `${(ky(JETZT) / KURVE_H) * 100}%` }}
        />
        <span className="absolute -top-[15px] -translate-x-1/2 rounded bg-white px-1 py-px text-[8.5px] font-bold leading-none text-navy-900" style={{ left: prozent(JETZT) }}>
          jetzt
        </span>
      </div>
      <div className="ov-num relative mt-1 h-3 text-[9px] font-medium text-white/65">
        <span className="absolute left-0">0</span>
        {[6, 12, 18].map((h) => <span key={h} className="absolute -translate-x-1/2" style={{ left: prozent(h) }}>{h}</span>)}
        <span className="absolute right-0">24 Uhr</span>
      </div>
    </div>
  );
}

/** Zehn Werktage als Segmente: verbrauchte Tage gedeckt, verbleibende in Sonne (dringend ab sieben) */
function FristSegmente({ rest = 6, gesamt = 10 }) {
  const verbraucht = gesamt - rest;
  return (
    <div>
      <div className="flex gap-[3px]">
        {Array.from({ length: gesamt }, (_, i) => (
          <span key={i} className={cn("h-1.5 flex-1 rounded-full", i < verbraucht ? "bg-ink-200" : "bg-sun-500")} />
        ))}
      </div>
      <p className="ov-num mt-1.5 flex justify-between text-[10.5px] text-ink-500">
        <span>{verbraucht} Werktage vergangen</span>
        <span className="font-semibold text-sun-700">noch {rest} von {gesamt}</span>
      </p>
    </div>
  );
}

        <span className="absolute left-1/2 top-1/2 h-1.5 w-1.5 -translate-x-1/2 -translate-y-1/2 rounded-full bg-navy-400" />
      </span>
    );
  }
  const breite = `${(Math.min(Math.abs(delta), 20) / 20) * 50}%`;
  return (
    <span className="relative block h-2 w-full rounded-full bg-ink-100">
      <span className="absolute inset-y-0 left-1/2 w-px bg-ink-300" />
      <span className={cn("absolute inset-y-0 rounded-full", delta > 0 ? "left-1/2 bg-sun-500" : "right-1/2 bg-sun-400")} style={{ width: breite }} />
    </span>
  );
}
