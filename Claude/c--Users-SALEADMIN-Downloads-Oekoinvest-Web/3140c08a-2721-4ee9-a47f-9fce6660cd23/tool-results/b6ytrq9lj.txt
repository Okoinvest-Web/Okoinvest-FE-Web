"use client";

// Land im Browser: aus der Route (/de, /at, /ch) oder – auf Stammseiten – aus der aktiven
// Länderwahl (Cookie „land“, erst nach Klick gesetzt). Ohne beides: Deutschland.
// Das Cookie wird nur nach dem ersten Rendern gelesen (keine Hydration-Abweichung).

import { useCallback, useEffect, useState } from "react";
import { usePathname } from "next/navigation";
import { landAusCookie, landAusPfad, landCookieZeile } from "@/lib/land";

/** Aktive Länderwahl speichern – nur aus einem Klick heraus aufrufen. */
export function landWaehlen(code) {
  const zeile = landCookieZeile(code, { https: typeof location !== "undefined" && location.protocol === "https:" });
  if (zeile) document.cookie = zeile;
}

export default function useLand() {
  const pfad = usePathname() || "/";
  const routenLand = landAusPfad(pfad).land;
  const [gewaehlt, setGewaehlt] = useState(null);
  useEffect(() => {
    setGewaehlt(landAusCookie(document.cookie));
  }, [pfad]);
  const waehlen = useCallback((code) => {
    landWaehlen(code);
    setGewaehlt(code);
  }, []);
  return { pfad, routenLand, gewaehlt, land: routenLand || gewaehlt || "DE", waehlen };
}

=== LaenderHinweis
"use client";

// Länderhinweis (Welle 2, Masterplan Abschnitt 4): schließbares Banner mit einem Vorschlag aus
// /api/land. Keine Weiterleitung, keine Pflichtentscheidung, kein Dark Pattern.
//  - erscheint nur auf der Startseite und in den Länderzonen, und nur wenn der Vorschlag vom
//    Land der Seite abweicht und noch keine Länderwahl getroffen ist
//  - „Zur Fassung für …“ und „Bei … bleiben“ sind aktive Wahlen → Cookie „land“
//  - Schließen (X oder Esc) ist keine Wahl → nur sessionStorage, kein Cookie
//  - erst nach der Entscheidung im Cookie-Banner (keine zwei Abfragen übereinander)
//  - Bots sehen ihn praktisch nie (eigene Anfrage nach dem Laden, kein Inhalt im HTML)

import { useEffect, useRef, useState } from "react";
import Link from "next/link";
import { MapPin, X } from "lucide-react";
import { HINWEIS_ZU, LAENDER, entsprechung, landAusCookie } from "@/lib/land";
import { EINWILLIGUNG_EREIGNIS, einwilligungLesen } from "@/lib/ereignis/einwilligung";
import useLand from "./useLand";

function sitzungLesen() {
  try {
    return window.sessionStorage.getItem(HINWEIS_ZU) === "1";
  } catch {
    return false;
  }
}

function sitzungMerken() {
  try {
    window.sessionStorage.setItem(HINWEIS_ZU, "1");
  } catch {
    /* Speicher gesperrt – Hinweis erscheint dann beim nächsten Laden erneut */
  }
}

export default function LaenderHinweis() {
  const { pfad, routenLand, waehlen } = useLand();
  const [vorschlag, setVorschlag] = useState(null);
  const [offen, setOffen] = useState(false);
  const regionRef = useRef(null);
  const passendeSeite = pfad === "/" || !!routenLand;

  function schliessen() {
    sitzungMerken();
    setOffen(false);
  }

  function bleiben() {
    if (routenLand) waehlen(routenLand);
    setOffen(false);
  }

  useEffect(() => {
    if (!passendeSeite || landAusCookie(document.cookie) || sitzungLesen()) return undefined;
    const abbruch = new AbortController();
    const laden = () =>
      fetch("/api/land", { cache: "no-store", signal: abbruch.signal })
        .then((r) => (r.ok ? r.json() : null))
        .then((d) => {
          const v = d?.vorschlag;
          if (v && LAENDER[v] && v !== routenLand) {
            setVorschlag(v);
            setOffen(true);
          }
        })
        .catch(() => {});
    // Nie zwei Abfragen gleichzeitig: erst nach der Entscheidung im Cookie-Banner
    if (einwilligungLesen()) {
      laden();
      return () => abbruch.abort();
    }
    const nachEinwilligung = () => laden();
    window.addEventListener(EINWILLIGUNG_EREIGNIS, nachEinwilligung, { once: true });
    return () => {
      window.removeEventListener(EINWILLIGUNG_EREIGNIS, nachEinwilligung);
      abbruch.abort();
    };
  }, [passendeSeite, routenLand]);

  useEffect(() => {
    if (!offen) return undefined;
    const taste = (e) => {
      if (e.key === "Escape" && regionRef.current?.contains(document.activeElement)) schliessen();
    };
    window.addEventListener("keydown", taste);
    return () => window.removeEventListener("keydown", taste);
  });

  if (!offen || !vorschlag) return null;

  const ziel = LAENDER[vorschlag];
  const hier = routenLand ? LAENDER[routenLand] : null;
  const z = entsprechung(pfad, vorschlag);

  return (
    <div
      ref={regionRef}
      role="region"
      aria-label="Ländervorschlag"
      className="ov-hero-in fixed inset-x-3 top-[92px] z-[9550] md:inset-x-auto md:right-6 md:top-[140px] md:max-w-[400px]"
    >
      <div className="relative rounded-3xl bg-white p-5 pr-12 shadow-e4 ring-1 ring-ink-200/80">
        <button
          type="button"
          onClick={schliessen}
          aria-label="Hinweis schließen"
          className="absolute right-3 top-3 flex h-9 w-9 items-center justify-center rounded-full text-ink-500 transition-colors hover:bg-ink-50 hover:text-ink-900 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ov-500"
        >
          <X aria-hidden="true" className="h-4 w-4" />
        </button>
        <p className="flex items-center gap-2 text-[12px] font-semibold uppercase tracking-[0.14em] text-ov-700">
          <MapPin aria-hidden="true" className="h-3.5 w-3.5" />
          Vorschlag
        </p>
        <p className="mt-2 text-[15px] leading-relaxed text-ink-800">
          {hier ? (
            <>
              Sie sehen die Seiten <strong className="font-semibold text-ink-900">{hier.fuerLand}</strong>. Möchten Sie zu den Seiten {ziel.fuerLand} wechseln?
            </>
          ) : (
            <>
              Es gibt eigene Seiten <strong className="font-semibold text-ink-900">{ziel.fuerLand}</strong> mit dem dort geltenden Recht.
            </>
          )}
          {z.art === "einstieg" && hier && <span className="mt-1 block text-[13.5px] text-ink-600">Diese Seite gibt es dort noch nicht – Sie kommen zur Übersicht.</span>}
        </p>
        <div className="mt-4 flex flex-wrap gap-2">
          <Link
            href={z.href}
            hrefLang={ziel.lang}
            onClick={() => waehlen(vorschlag)}
            className="ov-press inline-flex min-h-[40px] items-center justify-center rounded-full bg-ov-600 px-4 text-[14px] font-semibold text-white transition-colors hover:bg-ov-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ov-500 focus-visible:ring-offset-2"
          >
            Zu den Seiten {ziel.fuerLand}
          </Link>
          {hier && (
            <button
              type="button"
              onClick={bleiben}
              className="inline-flex min-h-[40px] items-center justify-center rounded-full px-4 text-[14px] font-semibold text-ink-800 ring-1 ring-inset ring-ink-200 transition-colors hover:bg-ink-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ov-500"
            >
              Bei {hier.name} bleiben
            </button>
          )}
        </div>
      </div>
    </div>
  );
}

 * @param {{ land: string, titel: string, worum: string, weiter?: Array<{ href: string, titel: string, text: string }> }} p

export default function InVorbereitung({ land, titel, worum, weiter = [] }) {
  const info = LAENDER[land];
  return (
    <div>
      <HeroBuehne breadcrumbs={[{ name: info.name, href: einstiegPfad(land) }, { name: titel }]} unten="pb-16 md:pb-20">
        <div className="max-w-[44rem] pb-6">
          <p className="ov-hero-in inline-flex items-center gap-2 rounded-full bg-sun-400/15 px-3.5 py-1.5 text-[13px] font-semibold text-sun-200 ring-1 ring-inset ring-sun-300/30">
            <Clock aria-hidden="true" className="h-4 w-4" />
            In Vorbereitung
          </p>
          <h1 id="hero-titel" className="ov-h1 ov-hero-in mt-6 text-balance" style={{ "--ov-delay": "80ms" }}>
            {titel} <span className="ov-text-gradient-light">{info.fuerLand}</span>
          </h1>
          <p className="ov-lead ov-hero-in mt-5 text-pretty text-white/75" style={{ "--ov-delay": "160ms" }}>
            {worum} Die Seite erscheint, sobald Inhalt, Zahlen und Begriffe {info.fuerLand} fachlich geprüft sind.
          </p>
        </div>
      </HeroBuehne>

      <Section tone="white" space="lg">
        <div className="grid gap-10 lg:grid-cols-[0.8fr_1.2fr] lg:gap-16">
          <SectionHeading eyebrow="Bis dahin" title="Diese Wege gibt es schon" lead="Schreiben Sie uns gern direkt – wir melden uns persönlich." />
          <ul className="grid gap-3 sm:grid-cols-2">
            {[...weiter, { href: einstiegPfad(land), titel: `Übersicht ${info.name}`, text: `Alle Seiten ${info.fuerLand} und was in Vorbereitung ist.` }, { href: "/kontakt", titel: "Kontakt", text: "Anfrage an das Projektteam." }].map((w, i) => (
              <Reveal as="li" key={w.href} delay={i * 60} className="flex">
                <Link href={w.href} className="group flex w-full flex-col rounded-3xl bg-sand-50 p-5 ring-1 ring-ink-200/70 transition-colors hover:bg-ov-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ov-500">
                  <span className="flex items-center justify-between gap-3 font-display text-[17px] font-bold text-ink-900">
                    {w.titel}
                    <ArrowRight aria-hidden="true" className="h-4 w-4 shrink-0 text-ov-600 transition-transform group-hover:translate-x-0.5" />
                  </span>
                  <span className="mt-1.5 text-[14.5px] leading-relaxed text-ink-600">{w.text}</span>
                </Link>
              </Reveal>
            ))}
          </ul>
        </div>
      </Section>
    </div>
  );
}
"use client";
import Navbar from "@/components/Navbar/navbar";
import Footer from "@/components/Reusable/footer";
import dynamic from "next/dynamic";
import { usePathname } from "next/navigation";
import { useEffect, useRef } from "react";
import { herkunftErfassen } from "@/lib/herkunft";
import RevealObserver from "@/components/ui/RevealObserver";
import MobileCta from "@/components/Reusable/MobileCta";
import MotionInteraktion from "@/components/motion/MotionInteraktion";
import ScrollFortschritt from "@/components/motion/ScrollFortschritt";
import Offenlegung from "@/components/Gruppe/Offenlegung";
import useLand from "@/components/Land/useLand";

// Lange Leseseiten: dünner Lesefortschritt oben (ScrollFortschritt)
const LESESEITEN = [/^\/wissen\//, /^\/(de|at|ch)\/faq$/, /^\/datenschutz$/, /^\/impressum$/, /^\/nutzungsbedingungen$/, /^\/bedingungen\//];

// const GoogleTagManager = dynamic(
//   (entfernt)
//   { ssr: false }
// );

const CookieComponentLazy = dynamic(() => import("@/components/Cookies/cookiecomponent"), { ssr: false });
const ToTopButtonLazy = dynamic(() => import("@/components/Home/BackToTop"), { ssr: false });
// Länderhinweis (Welle 2): Vorschlag aus /api/land, schließbar, Cookie erst nach aktiver Wahl
const LaenderHinweisLazy = dynamic(() => import("@/components/Land/LaenderHinweis"), { ssr: false });

export default function LayoutWrapper({ children }) {
  // Kampagnen-Zuordnung beim ersten Laden im Arbeitsspeicher festhalten (keine Cookies)
  useEffect(() => herkunftErfassen(), []);

  // Info-Bildschirm (SCADA/TV): nur der Inhalt, ohne Kopf, Fuß, Banner und Widgets
  const pfad = usePathname() || "";
  const { land } = useLand();
  const hauptRef = useRef(null);
  const ersterAufruf = useRef(true);

  // Seitenwechsel (Client-Navigation): Inhalt blendet sanft ein – nur opacity/transform,
  // ohne Füllmodus, damit danach kein transform am <main> hängen bleibt (fixierte Elemente).
  // Dokument-Navigation übernimmt @view-transition in globals.css.
  useEffect(() => {
    if (ersterAufruf.current) {
      ersterAufruf.current = false;
      return;
    }
    const el = hauptRef.current;
    if (!el?.animate || window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    const anim = el.animate(
      [
        { opacity: 0, transform: "translate3d(0, 8px, 0)" },
        { opacity: 1, transform: "none" },
      ],
      { duration: 320, easing: "cubic-bezier(0.22, 1, 0.36, 1)" }
    );
    return () => anim.cancel();
  }, [pfad]);

  if (pfad.startsWith("/portal")) return <main id="main-content">{children}</main>;

  return (
    <>
      {/* <GoogleTagManager gtmId="GTM-WR8PDT7V" /> */}
      <RevealObserver />
      <MotionInteraktion />
      {LESESEITEN.some((r) => r.test(pfad)) && <ScrollFortschritt key={pfad} />}
      {/* Cookie-Banner vor dem Seitenkopf: in der Tab-Reihenfolge direkt nach dem Sprunglink erreichbar (fixiert positioniert) */}
      <CookieComponentLazy />
      <LaenderHinweisLazy />
      <Navbar />
      {/* GRP: Offenlegung „Teil der ÖKOVOLT Gruppe“ – erscheint nur nach Gruppen-/Partnerverweis */}
      <Offenlegung land={land} />
      <main id="main-content" ref={hauptRef}>{children}</main>
      <Footer />
      <aside aria-label="Schnellzugriff">
        <MobileCta />
        <ToTopButtonLazy />
      </aside>
    </>
  );
}
