/** Dataset-JSON-LD (über das escapende JsonLd). */
function datasetSchema(land, serien, url) {
  const q = land ? `?land=${land}` : "";
  const runden = serien.flatMap((s) => rundenDer(s.id));
  const daten = runden.map((x) => x.datum).sort();
  return {
    "@context": "https://schema.org",
    "@type": "Dataset",
    "@id": `${url}#dataset`,
    name: land ? `Ausschreibungs-Monitor Photovoltaik ${LAND_NAME[land]}` : "Ausschreibungs-Monitor Photovoltaik DACH",
    description:
      "Ergebnisse und Termine der Photovoltaik-Ausschreibungen je Gebotstermin (Volumen, Gebots- und Zuschlagsmenge, mengengewichteter, niedrigster und höchster Zuschlagswert, Höchstwert) aus amtlichen Veröffentlichungen: " +
      serien.map((s) => `${s.name} (${s.land}, ${s.stelle})`).join("; ") +
      ".",
    url,
    inLanguage: "de",
    isAccessibleForFree: true,
    license: MONITOR_LIZENZ.url,
    creator: { "@type": "Organization", name: "ÖkoInvest GmbH", url: abs("/") },
    publisher: { "@id": `${abs("/")}/#organization` },
    isBasedOn: quellenFuer(land).map((x) => x.url),
    spatialCoverage: (land ? [land] : ["DE", "AT", "CH"]).map((l) => ({ "@type": "Place", name: LAND_NAME[l] })),
    temporalCoverage: `${daten[0]}/${daten[daten.length - 1]}`,
    dateModified: MONITOR_STAND,
    keywords: ["Ausschreibung", "Photovoltaik", "Zuschlagswert", "EEG", "EAG-Marktprämie", "Pronovo", "Einmalvergütung", "DACH"],
    variableMeasured: [
      { "@type": "PropertyValue", name: "Mengengewichteter durchschnittlicher Zuschlagswert", unitText: [...new Set(serien.map((s) => s.einheit))].join(", ") },
      { "@type": "PropertyValue", name: "Ausgeschriebene Menge, Gebotsmenge, Zuschlagsmenge", unitText: "MW" },
      { "@type": "PropertyValue", name: "Höchstwert je Gebotstermin", unitText: [...new Set(serien.map((s) => s.einheit))].join(", ") },
    ],
    distribution: [
      { "@type": "DataDownload", encodingFormat: "text/csv", contentUrl: abs(`${DATEI}.csv${q}`) },
      { "@type": "DataDownload", encodingFormat: "application/json", contentUrl: abs(`${DATEI}.json${q}`) },
    ],
  };
}
