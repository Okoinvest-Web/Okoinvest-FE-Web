/**
 * Flache Inline-SVG-Illustrationen der Agri-PV-Systeme (Server-Komponente).

 * Maßstab grob: 1 m ≈ 22 px, Boden bei y = 170.

const BODEN = 170;

function Boden() {
  return (
    <g>
      <rect x="0" y={BODEN} width="320" height="30" className="fill-ink-100" />
      <rect x="0" y={BODEN - 3} width="320" height="6" rx="3" className="fill-ov-300" />
    </g>
  );
}

function Traktor({ x, s = 1 }) {
  return (
    <g transform={`translate(${x} ${BODEN}) scale(${s})`}>
      <rect x="0" y="-26" width="34" height="14" rx="3" className="fill-ov-600" />
      <rect x="18" y="-44" width="16" height="20" rx="2" className="fill-ov-700" />
      <rect x="21" y="-41" width="10" height="9" rx="1" className="fill-navy-100" />
      <circle cx="28" cy="-10" r="10" className="fill-ink-800" />
      <circle cx="28" cy="-10" r="4" className="fill-ink-300" />
      <circle cx="6" cy="-7" r="7" className="fill-ink-800" />
      <circle cx="6" cy="-7" r="2.5" className="fill-ink-300" />
    </g>
  );
}

function Getreide({ x, n = 6, abstand = 7 }) {
  return (
    <g>
      {Array.from({ length: n }, (_, i) => (
        <g key={i}>
          <line x1={x + i * abstand} x2={x + i * abstand} y1={BODEN - 2} y2={BODEN - 18} strokeWidth="1.5" className="stroke-ov-600" />
          <ellipse cx={x + i * abstand} cy={BODEN - 20} rx="2" ry="4" className="fill-sun-400" />
        </g>
      ))}
    </g>
  );
}

function Strauch({ x, r = 11 }) {
  return (
    <g>
      <rect x={x - 1.5} y={BODEN - 14} width="3" height="14" className="fill-ink-500" />
      <circle cx={x} cy={BODEN - 20} r={r} className="fill-ov-500" />
      <circle cx={x - 4} cy={BODEN - 22} r="2.2" className="fill-sun-500" />
      <circle cx={x + 5} cy={BODEN - 17} r="2.2" className="fill-sun-500" />
    </g>
  );
}

function Schaf({ x }) {
  return (
    <g>
      <rect x={x + 4} y={BODEN - 10} width="2.5" height="9" className="fill-ink-800" />
      <rect x={x + 18} y={BODEN - 10} width="2.5" height="9" className="fill-ink-800" />
      <ellipse cx={x + 12} cy={BODEN - 14} rx="13" ry="8" fill="#fff" strokeWidth="1.2" className="stroke-ink-300" />
      <ellipse cx={x - 1} cy={BODEN - 17} rx="5" ry="4" className="fill-ink-800" />
    </g>
  );
}

function Modul({ x1, y1, x2, y2, dick = 4 }) {
  return <line x1={x1} y1={y1} x2={x2} y2={y2} strokeWidth={dick} strokeLinecap="round" className="stroke-navy-700" />;
}

export default function AgriBild({ typ = "vertikal", className, titel }) {
  return (
    <svg viewBox="0 0 320 200" className={className} role="img" aria-label={titel || BESCHREIBUNG[typ]}>
      <rect x="0" y="0" width="320" height={BODEN} className="fill-navy-50" />
      <circle cx="286" cy="30" r="14" className="fill-sun-300" />
      {typ === "hoch" && <Hoch />}
      {typ === "vertikal" && <Vertikal />}
      {typ === "tracker" && <Tracker />}
      {typ === "bodennah" && <Bodennah />}
      <Boden />
    </svg>
  );
}

  vertikal: "Vertikale bifaziale Agri-PV: senkrechte Modulzäune mit breiten Bewirtschaftungsgassen, in denen Getreide wächst und ein Traktor fährt.",
  tracker: "Nachgeführte Agri-PV: Module auf Drehachsen folgen der Sonne, zwischen den Reihen wird Ackerbau betrieben.",
  bodennah: "Bodennahe Agri-PV mit Beweidung: schräg aufgeständerte Module mit großem Reihenabstand, darunter und dazwischen weiden Schafe.",
};

"use client";

import { useMemo, useState } from "react";
import { Banknote, Gauge, Info, Sprout, Sun, Trophy, Zap } from "lucide-react";
import Button from "@/components/ui/Button";
import { cn } from "@/components/ui/cn";
import { Auswahl, Belege, Gruppe, Hauptzahl, Kennzahl, Regler, Zahl } from "@/components/Rechner/bausteine";
import { useTeilbarerZustand } from "@/components/Rechner/kontext";
import { ErgebnisKopf, MobilKurz, RechnerRahmen } from "@/components/Rechner/InvestTeile";
import { WEGE } from "./wege";
import { AGRI, ERLOESE, PACHT, REGIONEN } from "@/lib/invest/markt";
import { SYSTEME, agriVergleich, kulturById, systemPasst } from "@/lib/rechner/agriPv";
import { REGION_KURZ } from "@/lib/rechner/pacht";
import { fmt, fmtEur, fmtPct } from "@/lib/rechner/annahmen";

const betrag = (v) => {
  const a = Math.abs(v);
  const s = v < 0 ? "−" : "";
  return a >= 1e6 ? `${s}${fmt(a / 1e6, 2)} Mio. €` : `${s}${fmt(Math.round(a / 1000))} Tsd. €`;
};

const WEG_META = {
  pacht: { weg: WEGE.find((w) => w.id === "investor"), titel: "Verpachten", sub: "Investor baut, Sie ernten weiter" },
  eigen: { weg: WEGE.find((w) => w.id === "eigen"), titel: "Selbst investieren", sub: "Ihre Anlage, wir bauen & vermarkten" },
  genossenschaft: { weg: WEGE.find((w) => w.id === "genossenschaft"), titel: "Genossenschaft", sub: "Pacht + Anteil als Mitglied" },
};

export default function AgriPvRechner() {
  const [ha, setHa] = useState(10);
  const [system, setSystem] = useState("vertikal");
  const [kulturId, setKulturId] = useState("acker");
  const [regionId, setRegionId] = useState("de-sued");
  const [vermarktung, setVermarktung] = useState("eeg");
  const [db, setDb] = useState(kulturById("acker").db);
  const [anteil, setAnteil] = useState(AGRI.beteiligungAnteil.standard * 100);

  const kultur = kulturById(kulturId);
  const land = REGIONEN.find((r) => r.id === regionId)?.land;

  function kulturWaehlen(id) {
    setKulturId(id);
    const k = kulturById(id);
    setDb(k.db);
    if (!systemPasst(k, system)) setSystem("hoch");
  }

  const r = useMemo(
    () => agriVergleich({ ha, system, kulturId, regionId, dbProHa: db, vermarktung, anteil: anteil / 100 }),
    [ha, system, kulturId, regionId, db, vermarktung, anteil]
  );
  const [pacht, eigen, gen] = r.wege;
  const bester = [...r.wege].sort((a, b) => b.gesamtSumme - a.gesamtSumme)[0].id;
  const maxSumme = Math.max(...r.wege.map((w) => w.gesamtSumme), 1);
  const besterWeg = r.wege.find((w) => w.id === bester);

  // Teilen-Link: Eingaben in der Adresszeile (nur Abweichungen vom Standard)
  const SETZER = { ha: setHa, system: setSystem, kulturId: setKulturId, regionId: setRegionId, vermarktung: setVermarktung, db: setDb, anteil: setAnteil };
  useTeilbarerZustand(
    { ha, system, kulturId, regionId, vermarktung, db, anteil },
    (neu) => Object.entries(neu).forEach(([k, v]) => SETZER[k]?.(v)),
    {
      optionen: { system: SYSTEME.map((x) => x.id), kulturId: AGRI.kulturen.map((k) => k.id), regionId: REGIONEN.map((x) => x.id), vermarktung: ["eeg", "ppa"] },
      grenzen: { ha: [1, 50], db: [0, 10000], anteil: [AGRI.beteiligungAnteil.min * 100, AGRI.beteiligungAnteil.max * 100] },
    }
  );

  return (
    <RechnerRahmen
      breiteLinks={390}
      mitlaufend
      eingaben={
        <>
          <MobilKurz
            werte={[
              ["Leistung", `${fmt(r.anlage.kwp / 1000, 1)} MWp`],
              ["Pacht/Jahr", `${fmt(Math.round(pacht.energieJahr / 1000))} Tsd. €`],
              ["IRR eigen", eigen.irr != null ? fmtPct(eigen.irr, 1) : "–"],
            ]}
          />
          <div className="space-y-8">
            <Gruppe titel="Fläche & System">
              <Regler label="Fläche" wert={ha} min={1} max={50} step={0.5} stellen={ha % 1 ? 1 : 0} einheit="ha" onChange={setHa} hinweis={ha <= AGRI.privilegierungHa ? `Bis ${fmt(AGRI.privilegierungHa, 1)} ha hofnah ggf. ohne Bebauungsplan (§ 35 BauGB)` : "Größere Anlagen brauchen in der Regel einen Bebauungsplan"} />
              <Auswahl legende="Kultur" wert={kulturId} onChange={kulturWaehlen} spalten={2} klein optionen={AGRI.kulturen.map((k) => ({ id: k.id, label: k.label, sub: k.sub }))} />
              <Auswahl
                legende="System"
                wert={system}
                onChange={(v) => (systemPasst(kultur, v) ? setSystem(v) : null)}
                optionen={SYSTEME.map((s) => ({ id: s.id, label: s.label, sub: systemPasst(kultur, s.id) ? s.sub : "für Kultur unüblich" }))}
              />
              <Auswahl legende="Region" wert={regionId} onChange={setRegionId} spalten={3} klein optionen={REGIONEN.map((x) => ({ id: x.id, label: REGION_KURZ[x.id] || x.label }))} />
            </Gruppe>

            <Gruppe titel="Vermarktung & Beteiligung">
              {land === "DE" ? (
                <Auswahl
                  legende="Stromvermarktung"
                  wert={vermarktung}
                  onChange={setVermarktung}
                  optionen={[
                    { id: "eeg", label: "EEG-Zuschlag", sub: `${fmt(system === "hoch" ? ERLOESE.eegBesondereCt : ERLOESE.eegFreiflaecheCt, 1)} ct/kWh` },
                    { id: "ppa", label: "PPA", sub: `${fmt(ERLOESE.ppaCt.mittel, 1)} ct/kWh` },
                  ]}
                />
              ) : (
                <p className="rounded-2xl bg-white px-4 py-3 text-[13.5px] leading-relaxed text-ink-600 ring-1 ring-ink-200">
                  In {land === "AT" ? "Österreich" : "der Schweiz"} rechnen wir mit einem Stromliefervertrag (PPA) von {fmt(ERLOESE.ppaCt.mittel, 1)} ct/kWh; Förderungen nach {land === "AT" ? "EAG" : "Einmalvergütung"} prüfen wir im Einzelfall.
                </p>
              )}
              <Regler
                label="Ihr Anteil an der Anlage (Genossenschaft)"
                wert={anteil}
                min={AGRI.beteiligungAnteil.min * 100}
                max={AGRI.beteiligungAnteil.max * 100}
                step={1}
                einheit="%"
                onChange={setAnteil}
                hinweis={`Einlage ≈ ${betrag(gen.invest)}`}
              />
            </Gruppe>

            <Gruppe titel="Landwirtschaft">
              <Regler
                label="Deckungsbeitrag ohne Anlage"
                wert={db}
                min={0}
                max={kulturId === "sonder" ? 10000 : 1500}
                step={kulturId === "sonder" ? 100 : 10}
                format={(v) => `${fmt(v)} €/ha`}
                onChange={setDb}
                hinweis={`Unter dem System ca. ${fmt(r.landwirtschaft.ertragLw * 100)} % davon (inkl. Flächenverlust)`}
              />
            </Gruppe>
          </div>
        </>
      }
      ergebnis={
        <>
          <p className="sr-only" aria-live="polite">
            {`Anlage ${fmt(r.anlage.kwp / 1000, 1)} Megawatt Peak. Verpachten ${Math.round(pacht.gesamtSumme)} Euro, Eigeninvestition ${Math.round(eigen.gesamtSumme)} Euro, Genossenschaft ${Math.round(gen.gesamtSumme)} Euro über ${r.laufzeit} Jahre.`}
          </p>
          <ErgebnisKopf titel="Drei Wege im Vergleich" plakette={`${r.laufzeit} Jahre · vor Steuern`} />

          {/* Hauptzahl: der Weg mit der höchsten Summe */}
          <Hauptzahl className="mt-5" icon={Trophy} label={`Höchste Summe: ${WEG_META[bester].titel}`} zusatz={`${r.laufzeit} Jahre inkl. Landwirtschaft, nominal`}>
            <Zahl wert={besterWeg.gesamtSumme / 1000} suffix=" Tsd. €" />
          </Hauptzahl>

          {/* Anlage */}
          <Belege spalten={4}>
            <Kennzahl icon={Zap} label="Leistung">{`${fmt(r.anlage.kwp / 1000, 2)} MWp`}</Kennzahl>
            <Kennzahl icon={Sun} label="Strom pro Jahr">{`${fmt(r.anlage.kwh / 1000)} MWh`}</Kennzahl>
            <Kennzahl icon={Banknote} label="Investition">{betrag(r.anlage.capex)}</Kennzahl>
            <Kennzahl icon={Gauge} label="Erlös">{`${fmt(r.ctFoerder, 1)} ct/kWh`}</Kennzahl>
          </Belege>
          {kulturId === "sonder" && (
            <p className="mt-3 text-[13px] leading-relaxed text-ink-500">Bei Obst und Beeren ersetzt das Moduldach oft Hagelnetz oder Folientunnel und sichert Qualität – diese Einsparung ist hier nicht eingerechnet und kann die Rechnung deutlich verbessern.</p>
          )}
          {r.systemGewechselt && (
            <p className="mt-3 text-[13px] text-ink-500">Für {kultur.label} rechnen wir mit hoch aufgeständerter Agri-PV (Kategorie I).</p>
          )}

          {/* Drei Wege */}
          <div className="mt-5 grid gap-3 xl:grid-cols-3">
            {r.wege.map((w) => {
              const m = WEG_META[w.id];
              const Icon = m.weg.icon;
              const top = w.id === bester;
              return (
                <div key={w.id} className={cn("relative flex flex-col rounded-3xl p-5 ring-1", top ? "bg-ov-50 ring-2 ring-ov-500" : "bg-white ring-ink-200/70")}>
                  {top && <span className="absolute -top-3 left-5 rounded-full bg-ov-600 px-2.5 py-0.5 text-[11.5px] font-semibold text-white">Höchste Summe</span>}
                  <div className="flex items-center gap-3">
                    <span className={cn("flex h-10 w-10 shrink-0 items-center justify-center rounded-xl", top ? "bg-ov-500 text-white" : "bg-sand-50 text-ov-600 ring-1 ring-ink-200")}>
                      <Icon aria-hidden="true" className="h-5 w-5" />
                    </span>
                    <div className="min-w-0">
                      <p className="font-display text-[16.5px] font-bold leading-tight text-ink-900">{m.titel}</p>
                      <p className="text-[12.5px] leading-snug text-ink-500">{m.sub}</p>
                    </div>
                  </div>
                  <p className="mt-5 text-[12.5px] font-medium text-ink-500">Summe {r.laufzeit} Jahre inkl. Landwirtschaft</p>
                  <p className="ov-num font-display text-[26px] font-extrabold leading-tight tracking-tight text-ink-900">
                    <Zahl wert={w.gesamtSumme / 1000} suffix=" Tsd. €" />
                  </p>
                  <dl className="mt-4 space-y-2 border-t border-ink-200/70 pt-4 text-[13.5px]">
                    <Zeile k="Eigene Investition" v={w.invest ? betrag(w.invest) : "keine"} />
                    <Zeile k="Energie im 1. Jahr" v={fmtEur(w.energieJahr)} />
                    <Zeile k="Projektrendite (IRR)" v={w.irr != null ? fmtPct(w.irr, 1) : "–"} />
                    <Zeile k="Amortisation" v={w.invest ? (w.amortisation ? `${fmt(w.amortisation, 1)} Jahre` : "> Laufzeit") : "sofort"} />
                    <Zeile k="Risiko" v={w.risiko} />
                  </dl>
                </div>
              );
            })}
          </div>

          {/* Balken */}
          <div className="mt-5 rounded-3xl ring-1 ring-ink-200/70 p-5 md:p-6">
            <p className="font-display text-[16px] font-bold text-ink-900">Woraus sich die Summe zusammensetzt</p>
            <div className="mt-4 space-y-3.5">
              {r.wege.map((w) => {
                const lw = r.landwirtschaft.summe;
                const en = w.gesamtSumme - lw;
                return (
                  <div key={w.id}>
                    <div className="mb-1.5 flex justify-between gap-3 text-[13px]">
                      <span className="font-semibold text-ink-800">{WEG_META[w.id].titel}</span>
                      <span className="ov-num text-right text-ink-600">
                        {betrag(w.gesamtSumme)}
                        {en < 0 && <span className="block text-[12px] text-ink-500">Energie-Teil {betrag(en)}</span>}
                      </span>
                    </div>
                    <div className="flex h-3 overflow-hidden rounded-full bg-ink-100">
                      <span className="h-full bg-sun-400" style={{ width: `${(Math.max(0, lw) / maxSumme) * 100}%` }} />
                      <span className="h-full bg-ov-500" style={{ width: `${(Math.max(0, en) / maxSumme) * 100}%` }} />
                    </div>
                  </div>
                );
              })}
            </div>
            <ul className="mt-4 flex flex-wrap gap-x-5 gap-y-1.5 text-[12.5px] text-ink-600">
              <li className="flex items-center gap-2"><span aria-hidden="true" className="h-2.5 w-2.5 rounded-full bg-sun-400" />Landwirtschaft: Deckungsbeitrag + GAP</li>
              <li className="flex items-center gap-2"><span aria-hidden="true" className="h-2.5 w-2.5 rounded-full bg-ov-500" />Energie: Pacht bzw. Überschuss nach Investition</li>
            </ul>
          </div>

          {/* Landwirtschaft */}
          <div className="mt-4 grid gap-3 sm:grid-cols-3">
            <Mini icon={Sprout} k="Deckungsbeitrag/Jahr" v={fmtEur(r.landwirtschaft.dbJahr)} z={`ohne Anlage ${fmtEur(r.landwirtschaft.dbOhneAnlage)}`} />
            <Mini icon={Info} k="GAP-Direktzahlung/Jahr" v={r.land === "CH" ? "nicht gerechnet" : fmtEur(r.landwirtschaft.gapJahr)} z={r.land === "CH" ? "CH-Direktzahlungen eigenes System" : `${fmt(AGRI.gapFoerderfaehig * 100)} % der Fläche förderfähig`} />
            <Mini icon={Zap} k="Agri-Pacht/Jahr" v={fmtEur(r.pachtJahr0)} z={`${fmt(PACHT.agriProHa.mittel)} €/ha, +${fmt(PACHT.indexierung * 100, 1)} % p. a.`} />
          </div>

          <div className="mt-7 flex flex-col gap-3 border-t border-ink-100 pt-6 sm:flex-row sm:items-center sm:justify-between">
            <p className="max-w-md text-[13px] leading-relaxed text-ink-500">
              Projektrendite ohne Fremdkapital und vor Steuern, nominal – eine Orientierung, kein Angebot. Eigenverbrauch am Hof, Speicher und eingesparte Schutznetze sind nicht eingerechnet – sie verbessern Weg B oft deutlich.
            </p>
            <Button href="/landwirte#anfrage" size="lg" pfeil className="shrink-0">
              Mit uns durchrechnen
            </Button>
          </div>
        </>
      }
    />
  );
}

function Zeile({ k, v }) {
  return (
    <div className="flex items-baseline justify-between gap-3">
      <dt className="text-ink-500">{k}</dt>
      <dd className="ov-num text-right font-semibold text-ink-900">{v}</dd>
    </div>
  );
}

export default function WegeEntscheider() {
  const [antworten, setAntworten] = useState({});
  const [schritt, setSchritt] = useState(0);
  const [richtung, setRichtung] = useState("vor");
  const fertig = schritt >= FRAGEN.length;

  const ergebnis = useMemo(() => {
    if (!fertig) return null;
    const punkte = Object.fromEntries(WEGE.map((w) => [w.id, 0]));
    const gruende = Object.fromEntries(WEGE.map((w) => [w.id, []]));
    for (const f of FRAGEN) {
      const a = f.antworten.find((x) => x.id === antworten[f.id]);
      if (!a) continue;
      for (const [w, p] of Object.entries(a.p)) punkte[w] += p;
      for (const [w, g] of Object.entries(a.g || {})) gruende[w].push(g);
    }
    const max = Math.max(...Object.values(punkte));
    const min = Math.min(...Object.values(punkte));
    const liste = WEGE.map((w) => ({ ...w, punkte: punkte[w.id], anteil: max === min ? 1 : (punkte[w.id] - min) / (max - min), gruende: gruende[w.id] })).sort((a, b) => b.punkte - a.punkte);
    return liste;
  }, [fertig, antworten]);

  function waehle(fid, aid) {
    setAntworten((alt) => ({ ...alt, [fid]: aid }));
    setRichtung("vor");
    setTimeout(() => setSchritt((s) => Math.min(s + 1, FRAGEN.length)), 180);
  }

  function zurueck() {
    setRichtung("zurueck");
    setSchritt((s) => Math.max(0, s - 1));
  }

  function neu() {
    setAntworten({});
    setRichtung("zurueck");
    setSchritt(0);
  }

  const f = FRAGEN[schritt];

  return (
    <div className="overflow-hidden rounded-[2rem] bg-white shadow-[0_40px_90px_-50px_rgba(3,18,43,0.55)] ring-1 ring-ink-200/70">
      <div className="flex items-center justify-between gap-4 border-b border-ink-100 bg-sand-50/70 px-5 py-4 sm:px-8">
        <p className="text-[13px] font-semibold uppercase tracking-[0.14em] text-ink-500">
          {fertig ? "Ihre Empfehlung" : `Frage ${schritt + 1} von ${FRAGEN.length}`}
        </p>
        <div className="flex h-2 w-32 overflow-hidden rounded-full bg-ink-100 sm:w-48" role="progressbar" aria-valuemin={0} aria-valuemax={FRAGEN.length} aria-valuenow={Math.min(schritt, FRAGEN.length)} aria-label="Fortschritt">
          <div className="h-full rounded-full bg-ov-500 motion-safe:transition-[width] motion-safe:duration-500" style={{ width: `${(Math.min(schritt, FRAGEN.length) / FRAGEN.length) * 100}%` }} />
        </div>
      </div>

      <div className="p-5 sm:p-8 md:p-10">
        <div key={schritt} className={richtung === "vor" ? "ov-step-vor" : "ov-step-zurueck"}>
          {!fertig && (
            <fieldset>
              <legend className="font-display text-[clamp(1.3rem,1.05rem+0.9vw,1.75rem)] font-extrabold leading-snug tracking-tight text-ink-900">{f.frage}</legend>
              <div className={cn("mt-6 grid gap-3", f.antworten.length === 4 ? "sm:grid-cols-2" : "sm:grid-cols-3")}>
                {f.antworten.map((a) => {
                  const an = antworten[f.id] === a.id;
                  return (
                    <label
                      key={a.id}
                      className={cn(
                        "relative flex cursor-pointer flex-col rounded-3xl p-5 pr-12 transition-all focus-within:ring-2 focus-within:ring-ov-500",
                        an ? "bg-ov-50 ring-2 ring-ov-500" : "bg-sand-50 ring-1 ring-ink-200 hover:bg-white hover:ring-ink-300"
                      )}
                    >
                      <input type="radio" name={f.id} value={a.id} checked={an} onChange={() => waehle(f.id, a.id)} className="sr-only" />
                      <span className="font-display text-[17px] font-bold text-ink-900">{a.label}</span>
                      <span className="mt-1 text-[14px] leading-snug text-ink-500">{a.sub}</span>
                      <span aria-hidden="true" className={cn("absolute right-4 top-5 flex h-6 w-6 items-center justify-center rounded-full", an ? "bg-ov-500 text-white" : "bg-white ring-1 ring-ink-300")}>
                        {an && <Check className="h-3.5 w-3.5" strokeWidth={3} />}
                      </span>
                    </label>
                  );
                })}
              </div>
              {schritt > 0 && (
                <button type="button" onClick={zurueck} className="mt-6 inline-flex h-11 items-center gap-2 rounded-full px-4 text-[14.5px] font-semibold text-ink-700 hover:bg-ink-100">
                  <ArrowLeft aria-hidden="true" className="h-4 w-4" /> Zurück
                </button>
              )}
            </fieldset>
          )}

          {fertig && ergebnis && <Empfehlung liste={ergebnis} onNeu={neu} onZurueck={zurueck} />}
        </div>
      </div>
    </div>
  );
}

function Empfehlung({ liste, onNeu, onZurueck }) {
  const [erster, ...rest] = liste;
  const zweiter = rest[0];
  const knapp = zweiter && erster.punkte - zweiter.punkte <= 1;
  const Icon = erster.icon;
  return (
    <div aria-live="polite">
      <div className="ov-noise relative overflow-hidden rounded-3xl bg-navy-950 p-6 text-white md:p-8">
        <div aria-hidden="true" className="ov-grid-bg absolute inset-0" />
        <div aria-hidden="true" className="absolute -right-16 -top-24 h-72 w-72 rounded-full bg-ov-500/30 blur-[90px]" />
        <div className="relative">
          <p className="flex items-center gap-2 text-[12.5px] font-semibold uppercase tracking-[0.16em] text-ov-300">
            <Sparkles aria-hidden="true" className="h-4 w-4" /> Am besten passt
          </p>
          <div className="mt-4 flex items-start gap-4">
            <span className="hidden h-14 w-14 shrink-0 items-center justify-center rounded-2xl bg-ov-500 text-white sm:flex">
              <Icon aria-hidden="true" className="h-7 w-7" />
            </span>
            <div className="min-w-0">
              <h3 className="font-display text-[clamp(1.4rem,1.1rem+1.2vw,2rem)] font-extrabold leading-tight tracking-tight">

              </h3>
              <p className="mt-1 text-[15.5px] text-white/70">{erster.kurz}</p>
            </div>
          </div>
          {erster.gruende.length > 0 && (
            <ul className="mt-6 space-y-2.5 border-t border-white/10 pt-5">
              {erster.gruende.map((g) => (
                <li key={g} className="flex gap-2.5 text-[15px] leading-relaxed text-white/85">
                  <Check aria-hidden="true" className="mt-1 h-4 w-4 shrink-0 text-ov-300" strokeWidth={3} />
                  {g}
                </li>
              ))}
            </ul>
          )}
          {knapp && (
            <p className="mt-5 rounded-2xl bg-white/[0.07] px-4 py-3 text-[14px] leading-relaxed text-white/75 ring-1 ring-white/10">
              Knapp dahinter: <strong className="text-white">Weg {zweiter.buchstabe} – {zweiter.titel}</strong>. Beide Wege lassen sich oft auch kombinieren, etwa Pacht für einen Teil und Beteiligung für den Rest.
            </p>
          )}
          <div className="mt-6 flex flex-col gap-3 sm:flex-row">
            <Link href={erster.href} className="group inline-flex h-12 items-center justify-center gap-2 rounded-full bg-ov-600 px-6 text-[15px] font-semibold text-white hover:bg-ov-700">
              Mehr zu Weg {erster.buchstabe}
              <ArrowRight aria-hidden="true" className="h-4 w-4 transition-transform group-hover:translate-x-1" />
            </Link>
            <Link href="/rechner/agri-pv" className="inline-flex h-12 items-center justify-center gap-2 rounded-full px-6 text-[15px] font-semibold text-white ring-1 ring-inset ring-white/35 hover:bg-white/10">
              Wege durchrechnen
            </Link>
          </div>
        </div>
      </div>

      <div className="mt-6">
        <p className="text-[13px] font-semibold uppercase tracking-[0.14em] text-ink-500">Alle Wege im Abgleich mit Ihren Antworten</p>
        <ul className="mt-3 space-y-2.5">
          {liste.map((w) => (
            <li key={w.id} className="grid grid-cols-[2rem_1fr] items-center gap-3 sm:grid-cols-[2rem_minmax(0,16rem)_1fr]">
              <span className={cn("flex h-8 w-8 items-center justify-center rounded-full font-display text-[13px] font-extrabold", w.id === erster.id ? "bg-ov-600 text-white" : "bg-ink-100 text-ink-600")}>{w.buchstabe}</span>
              <span className="text-[14.5px] font-semibold text-ink-800">{w.titel}</span>
              <span className="col-span-2 h-2.5 overflow-hidden rounded-full bg-ink-100 sm:col-span-1">
                <span className={cn("block h-full rounded-full motion-safe:transition-[width] motion-safe:duration-700", w.id === erster.id ? "bg-ov-500" : "bg-ov-200")} style={{ width: `${Math.max(6, w.anteil * 100)}%` }} />
              </span>
            </li>
          ))}
        </ul>
      </div>

      <div className="mt-8 flex flex-wrap items-center justify-between gap-3 border-t border-ink-100 pt-5">
        <p className="max-w-xl text-[13px] leading-relaxed text-ink-500">Eine Orientierung, keine Beratung. Welcher Weg wirtschaftlich passt, hängt von Fläche, Netz und Betrieb ab – das rechnen wir gern gemeinsam durch.</p>
        <div className="flex gap-2">
          <button type="button" onClick={onZurueck} className="inline-flex h-11 items-center gap-2 rounded-full px-4 text-[14.5px] font-semibold text-ink-700 hover:bg-ink-100">
            <ArrowLeft aria-hidden="true" className="h-4 w-4" /> Zurück
          </button>
          <button type="button" onClick={onNeu} className="inline-flex h-11 items-center gap-2 rounded-full px-4 text-[14.5px] font-semibold text-ink-700 ring-1 ring-ink-200 hover:bg-ink-50">
            <RotateCcw aria-hidden="true" className="h-4 w-4" /> Neu starten
          </button>
        </div>
      </div>
    </div>
  );
}
