export default async function PortalLayout({ children }) {
  const nutzer = await portalNutzer();
  const b = backend();
  const admin = nutzer.rollen?.includes("admin");
  // Zähler sind nicht kritisch – Fehler dürfen das Portal nicht blockieren
  const [threads, hinweise, aufgaben] = await Promise.all([
    b.nachrichtenThreads(nutzer).catch(() => []),
    b.benachrichtigungen(nutzer, { limit: 8 }).catch(() => ({ eintraege: [], ungelesen: 0 })),
    admin ? b.aufgaben(nutzer, { zustaendigId: nutzer.id, status: "offen" }).catch(() => []) : [],
  ]);
  const ungelesen = threads.reduce((s, t) => s + (t.ungelesen || 0), 0);
  const aufgabenFaellig = aufgaben.filter((a) => a.ueberfaellig || a.heuteFaellig).length;
  return (
    <PortalRahmen
      nutzer={nutzerFuerBrowser(nutzer)}
      ungelesen={ungelesen}
      hinweise={{ eintraege: hinweise.eintraege, ungelesen: hinweise.ungelesen }}
      aufgabenFaellig={aufgabenFaellig}
      demo={istDemo()}
    >
      {children}
    </PortalRahmen>
  );
}

function brotkrumen(pfad) {
  const teile = pfad.split("/").filter(Boolean);
  return teile.map((t, i) => ({ href: "/" + teile.slice(0, i + 1).join("/"), label: LABELS[t] || decodeURIComponent(t) }));
}

function aktiv(pfad, e) {
  return e.exakt ? pfad === e.href : pfad === e.href || pfad.startsWith(e.href + "/");
}

export const KONFIG_FEHLER =
  "Das Plattform-Backend ist nicht konfiguriert. Für den Betrieb PLATTFORM_BACKEND=frappe setzen (siehe docs/PLATTFORM.md) – oder PLATTFORM_DEMO=1 für eine reine Vorführung.";

export function backendName() {
  if (process.env.PLATTFORM_BACKEND === "frappe") return "frappe";
  if (process.env.NODE_ENV === "production" && process.env.PLATTFORM_DEMO !== "1") return null;
  return "demo";
}

/** Liefert die aktive Implementierung oder wirft 503 */
export function backend() {
  const n = backendName();
  if (n === "frappe") return frappe;
  if (n === "demo") return demo;
  throw new PlattformFehler(503, KONFIG_FEHLER);
}

// Gemeinsamer Rahmen aller Plattform-APIs:
//  - CSRF-Schutz: Mutationen nur mit passendem Origin-Header UND Content-Type JSON
//    (Uploads: multipart + Kopfzeile X-Plattform-Upload). Ein fremdes Formular kann weder
//    JSON senden noch eigene Kopfzeilen setzen; fremde fetch-Aufrufe scheitern am Origin.
//  - Sitzung aus dem signierten Cookie, Nutzer frisch aus dem Backend (Rollen, Sperre, Version)
//  - Rollenprüfung vorab (Eigentum prüft das Backend)
//  - Rate-Limits, Größenbegrenzung des Bodys
//  - Fehler → { ok:false, error, code } mit passendem HTTP-Status, Interna nur ins Log

const KEIN_CACHE = { "Cache-Control": "no-store" };

export function antwort(daten, status = 200) {
  return NextResponse.json(daten, { status, headers: KEIN_CACHE });
}

export function fehlerAntwort(e) {
  if (e instanceof PlattformFehler) {
    return antwort({ ok: false, error: e.message, code: e.code }, e.status);
  }
  console.error("[plattform] Unerwarteter Fehler:", e);
  return antwort({ ok: false, error: "Es ist ein Fehler aufgetreten. Bitte versuchen Sie es später erneut.", code: "fehler" }, 500);
}

function hostVon(url) {
  try {
    return new URL(url).host.toLowerCase();
  } catch {
    return "";
  }
}

/** Erlaubte Hosts für den Origin-Abgleich */
function erlaubteHosts(request) {
  const hosts = new Set();
  const h = request.headers.get("x-forwarded-host") || request.headers.get("host");
  if (h) hosts.add(h.split(",")[0].trim().toLowerCase());
  hosts.add(hostVon(SITE.url));
  for (const o of String(process.env.PLATTFORM_ORIGINS || "").split(",")) if (o.trim()) hosts.add(hostVon(o.trim()));
  return hosts;
}

/** CSRF: Origin muss zur eigenen Seite gehören */
export function herkunftPruefen(request) {
  const origin = request.headers.get("origin");
  if (!origin || origin === "null" || !erlaubteHosts(request).has(hostVon(origin))) {
    throw new PlattformFehler(403, "Anfrage von fremder Herkunft abgelehnt.", "herkunft");
  }
}

/** JSON-Body lesen (max. Bytes), muss ein Objekt sein */
export async function jsonLesen(request, maxBytes = 64 * 1024) {
  const typ = request.headers.get("content-type") || "";
  if (!typ.toLowerCase().startsWith("application/json")) throw new PlattformFehler(415, "Erwartet wird JSON.");
  const laenge = Number(request.headers.get("content-length") || 0);
  if (laenge > maxBytes) throw new PlattformFehler(413, "Anfrage zu groß.");
  const text = await request.text();
  if (text.length > maxBytes) throw new PlattformFehler(413, "Anfrage zu groß.");
  let daten;
  try {
    daten = text ? JSON.parse(text) : {};
  } catch {
    throw new PlattformFehler(400, "Ungültiges JSON.");
  }
  if (!daten || typeof daten !== "object" || Array.isArray(daten)) throw new PlattformFehler(400, "Ungültige Daten.");
  return daten;
}

/** Angemeldeten Nutzer aus dem Cookie lesen (oder null) */
export async function nutzerAusAnfrage(request, b = backend()) {
  const token = request.cookies.get(cookieName())?.value;
  if (!token) return null;
  const s = await tokenPruefen(token);
  if (!s) return null;
  const u = await b.nutzerFuerSitzung(s.nutzerId);
  if (!u || !u.aktiv || (u.sitzungVersion || 0) !== s.version) return null;
  return u;
}

/** Sitzungscookie setzen */
export async function sitzungSetzen(res, nutzer) {
  const token = await tokenErstellen({ nutzerId: nutzer.id, version: nutzer.sitzungVersion || 0 });
  res.cookies.set(cookieName(), token, cookieOptionen());
  return res;
}

export function sitzungLoeschen(res) {
  res.cookies.set(cookieName(), "", { ...cookieOptionen(0), maxAge: 0 });
  return res;
}

/** Basis-URL für Links in E-Mails: in Produktion fest (kein Host-Header-Vertrauen) */
export function basisUrl(request) {
  if (process.env.NODE_ENV === "production") return (process.env.PLATTFORM_BASIS_URL || SITE.url).replace(/\/$/, "");
  return request.nextUrl.origin;
}

/** Nutzer für die Antwort an den Browser (ohne interne Felder) */
export function nutzerFuerBrowser(u) {
  if (!u) return null;
  return { id: u.id, email: u.email, vorname: u.vorname, nachname: u.nachname, firma: u.firma, telefon: u.telefon, rollen: u.rollen, emailBestaetigt: !!u.emailBestaetigt, suchprofil: u.suchprofil || null, mailEinstellungen: u.mailEinstellungen || null };
}

/**
 * Route-Handler mit Rahmen.
 *
 * optionen:
 *   auth      true (Pflicht, Standard) | "optional" | false
 *   rollen    [..] – mindestens eine davon (admin darf immer)
 *   admin     true – nur Rolle admin
 *   limit     Name aus GRENZEN (je IP)
 *   json      true – JSON-Body lesen (Standard bei POST/PATCH/PUT/DELETE)
 *   upload    true – multipart statt JSON (eigene Kopfzeile Pflicht)
 *   maxBytes  Größe des JSON-Bodys
 */
export function route(optionen, handler) {
  const o = { auth: true, ...optionen };
  return async (request, ctx) => {
    try {
      const mutation = !["GET", "HEAD", "OPTIONS"].includes(request.method);
      const params = ctx?.params ? await ctx.params : {};
      const ip = ipAdresse(request);

      if (mutation) {
        herkunftPruefen(request);
        if (o.upload && request.headers.get("x-plattform-upload") !== "1") throw new PlattformFehler(403, "Upload nur über das Portal.", "herkunft");
      }
      if (o.limit) rateLimit(`${o.limit}:${ip || "ohne-ip"}`, ...grenze(o.limit));

      const b = backend();
      let nutzer = null;
      if (o.auth) {
        nutzer = await nutzerAusAnfrage(request, b);
        if (!nutzer && o.auth === true) throw new PlattformFehler(401, "Bitte melden Sie sich an.");
        if (nutzer) {
          const admin = nutzer.rollen?.includes("admin");
          if (o.admin && !admin) throw new PlattformFehler(403, "Nur für das Backoffice.");
          if (o.rollen && !admin && !o.rollen.some((r) => nutzer.rollen?.includes(r))) {
            throw new PlattformFehler(403, "Dafür fehlt Ihnen die passende Rolle.");
          }
        }
      }

      const body = mutation && !o.upload && o.json !== false ? await jsonLesen(request, o.maxBytes) : undefined;
      const ergebnis = await handler({ request, params, nutzer, body, backend: b, ip, suche: request.nextUrl.searchParams });
      if (ergebnis instanceof Response) return ergebnis;
      return antwort(ergebnis === undefined ? { ok: true } : ergebnis);
    } catch (e) {
      return fehlerAntwort(e);
    }
  };
}
// src/lib/plattform/server/seite.js

// Sitzung in Server-Komponenten (Seiten, Layouts). Die Middleware prüft vorab nur die
// Signatur; hier wird der Nutzer frisch geladen (Rollen, Sperre, Sitzungsversion).

import { cookies, headers } from "next/headers";
import { redirect, notFound } from "next/navigation";
import { backend } from "../backend";
import { cookieName, tokenPruefen } from "../sitzung";
import { PlattformFehler } from "../fehler";

export async function aktuellerNutzer() {
  const token = (await cookies()).get(cookieName())?.value;
  if (!token) return null;
  const s = await tokenPruefen(token);
  if (!s) return null;
  const u = await backend().nutzerFuerSitzung(s.nutzerId);
  if (!u || !u.aktiv || (u.sitzungVersion || 0) !== s.version) return null;
  return u;
}

/** Aktuellen Pfad (von der Middleware als Kopfzeile gesetzt) für ?weiter= */
async function aktuellerPfad() {
  const h = await headers();
  const p = h.get("x-oi-pfad") || "/portal";
  return p.startsWith("/portal") ? p : "/portal";
}

/**
 * Für Portalseiten: Nutzer oder Weiterleitung zum Login.
 * rollen: mindestens eine davon (admin darf immer), sonst 404 (keine Existenz verraten).
 */
export async function portalNutzer({ rollen, admin } = {}) {
  const u = await aktuellerNutzer();
  if (!u) redirect(`/login?weiter=${encodeURIComponent(await aktuellerPfad())}`);
  const istAdmin = u.rollen.includes("admin");
  if (admin && !istAdmin) notFound();
  if (rollen && !istAdmin && !rollen.some((r) => u.rollen.includes(r))) notFound();
  return u;
}

export const istAdmin = (u) => !!u?.rollen?.includes("admin");
export const hatRolle = (u, ...rollen) => istAdmin(u) || rollen.some((r) => u?.rollen?.includes(r));

/** Backend-Aufruf in Seiten: 403/404 → notFound(), sonst weiterwerfen */
export async function laden(fn) {
  try {
    return await fn();
  } catch (e) {
    if (e instanceof PlattformFehler && [403, 404].includes(e.status)) notFound();
    throw e;
  }
}
// src/lib/plattform/client.js
//
// fetch-Helfer für Client-Komponenten. Mutationen immer als JSON (CSRF-Schutz
// über Content-Type + Origin, siehe server/route.js); Uploads mit Kopfzeile.

export async function senden(url, { method = "POST", body } = {}) {
  let res;
  try {
    res = await fetch(url, {
      method,
      headers: body !== undefined ? { "Content-Type": "application/json" } : undefined,
      body: body !== undefined ? JSON.stringify(body) : undefined,
      credentials: "same-origin",
      cache: "no-store",
    });
  } catch {
    throw new Error("Keine Verbindung – bitte prüfen Sie Ihre Internetverbindung.");
  }
  const json = await res.json().catch(() => ({}));
  if (!res.ok || json.ok === false) {
    const e = new Error(json.error || `Fehler ${res.status}`);
    e.status = res.status;
    throw e;
  }
  return json;
}

export async function hochladen(datei, { bezugArt, bezugId, kategorie = "allgemein", checkliste = "" }) {
  const fd = new FormData();
  fd.append("datei", datei);
  fd.append("bezugArt", bezugArt);
  fd.append("bezugId", bezugId);
  fd.append("kategorie", kategorie);
  if (checkliste) fd.append("checkliste", checkliste);
  const res = await fetch("/api/portal/dokumente", { method: "POST", body: fd, headers: { "X-Plattform-Upload": "1" }, credentials: "same-origin" });
  const json = await res.json().catch(() => ({}));
  if (!res.ok || json.ok === false) throw new Error(json.error || `Upload fehlgeschlagen (${res.status})`);
  return json.dokument;
}

/** Nur interne Pfade als Ziel nach dem Login zulassen (kein Open Redirect) */
export function sicheresZiel(weiter, standard = "/portal") {
  const w = String(weiter || "");
  return w.startsWith("/") && !w.startsWith("//") && !w.startsWith("/\\") && !/^\/api\//.test(w) ? w : standard;
}
