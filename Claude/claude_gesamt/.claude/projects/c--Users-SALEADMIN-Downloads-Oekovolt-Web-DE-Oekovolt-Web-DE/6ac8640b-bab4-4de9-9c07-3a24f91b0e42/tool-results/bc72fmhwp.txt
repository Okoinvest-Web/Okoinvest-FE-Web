    92	/** Standardwerte je Land/Typ (Beispielprojekt 10 MWp Freifläche) – alles aus markt.js */
    93	export function cockpitStandard({ land = "DE", typ = "freiflaeche", mwp = 10 } = {}) {
    94	  const l = LAENDER_FZ.includes(land) ? land : "DE";
    95	  const fk = FINANZIERUNG.foerderkredit[l];
    96	  const kwp = mwp * 1000;
    97	  const capexKwp = FREIFLAECHE.capexProKwp.mittel;
    98	  const capex = Math.round(kwp * capexKwp + mwp * FREIFLAECHE.netzanschlussFixProMwp + 1000 * FREIFLAECHE.kabelProMeter + kwp * FREIFLAECHE.entwicklungProKwp.min);
    99	  const p50 = l === "AT" ? 1120 : l === "CH" ? 1050 : 1100;
   100	  return {
   101	    land: l, typ: TYPEN_FZ[typ] ? typ : "freiflaeche", mwp, capex, p50, p90: Math.round(p50 * 0.92),
   102	    modell: "eeg", preisCt: ERLOESE.eegFreiflaecheCt, vertragJahre: 20, lieferAnteil: 1, indexierung: 0, merchantCt: ERLOESE.merchantCt,

   114	/** Eingaben begrenzen (Server: gespeicherte Szenarien; Client: Formular). Unbekanntes fällt weg. */
   115	export function cockpitBereinigen(e = {}, basis = null) {
   116	  const out = { ...(basis || cockpitStandard({ land: e.land, typ: e.typ, mwp: zahl(e.mwp, 10) || 10 })) };
   117	  if (LAENDER_FZ.includes(e.land)) out.land = e.land;
   118	  if (TYPEN_FZ[e.typ]) out.typ = e.typ;
   119	  for (const f of COCKPIT_FELDER) {
   120	    const v = e[f.id];
   121	    if (v === undefined || v === null || v === "") continue;
   122	    if (f.typ === "auswahl") {
   123	      if (f.optionen.some(([id]) => id === v)) out[f.id] = v;
   124	    } else if (f.typ === "schalter") {
   125	      out[f.id] = v === true || v === "true" || v === 1 || v === "1";
   126	    } else {
   127	      const n = typeof v === "number" ? v : Number(String(v).replace(",", "."));
   128	      if (Number.isFinite(n)) out[f.id] = Math.min(f.max, Math.max(f.min, n));
   129	    }
   130	  }
   131	  if (out.p90 > out.p50) out.p90 = out.p50;
   132	  out.laufzeitKredit = Math.min(out.laufzeitKredit, out.laufzeit);
   133	  out.foerderLaufzeit = Math.min(out.foerderLaufzeit, out.laufzeit);
   134	  out.leasingLaufzeit = Math.min(out.leasingLaufzeit, out.laufzeit);
   135	  out.mietkaufLaufzeit = Math.min(out.mietkaufLaufzeit, out.laufzeit);
   136	  out.tilgungsfrei = Math.min(out.tilgungsfrei, Math.max(0, out.laufzeitKredit - 1));
   137	  out.foerderAnlauf = Math.min(out.foerderAnlauf, Math.max(0, out.foerderLaufzeit - 1));
   138	  return out;
   139	}

   143	/** Operative Jahreswerte P50 und P90 (Index 0 = Jahr 1) */
   144	export function betriebsJahre(e) {
   145	  const kwp = e.mwp * 1000;
   146	  const jahre = [];
   147	  const vertrag = e.modell === "merchant" ? 0 : Math.min(e.vertragJahre, e.laufzeit);
   148	  for (let t = 1; t <= e.laufzeit; t++) {
   149	    const deg = Math.pow(1 - FREIFLAECHE.degradation, t - 1);
   150	    const marktCt = e.merchantCt * Math.pow(1 + ERLOESE.merchantSteigerung, t - 1);
   151	    const imVertrag = t <= vertrag;
   152	    let preisCt;
   153	    if (!imVertrag) preisCt = marktCt;
   154	    else if (e.modell === "eeg") preisCt = Math.max(e.preisCt, marktCt);
   155	    else if (e.modell === "contracting") {
   156	      const liefer = e.preisCt * Math.pow(1 + (e.indexierung || 0), t - 1);
   157	      preisCt = (e.lieferAnteil ?? 1) * liefer + (1 - (e.lieferAnteil ?? 1)) * marktCt;
   158	    } else preisCt = e.preisCt * Math.pow(1 + (e.indexierung || 0), t - 1);
   159	    const e50 = kwp * e.p50 * deg;
   160	    const e90 = kwp * e.p90 * deg;
   161	    const opex = kwp * e.opexKwp * Math.pow(1 + FREIFLAECHE.inflation, t - 1);
   162	    const pacht = e.pachtJahr * Math.pow(1 + PACHT.indexierung, t - 1);
   163	    const rueckbau = (kwp * FREIFLAECHE.rueckbauProKwp) / e.laufzeit;
   164	    const kosten = opex + pacht + rueckbau;
   165	    const erloes50 = (e50 * preisCt) / 100;
   166	    const erloes90 = (e90 * preisCt) / 100;
   167	    jahre.push({ jahr: t, preisCt, imVertrag, erzeugung50: e50, erzeugung90: e90, erloes50, erloes90, opex, pacht, rueckbau, kosten, cfads50: erloes50 - kosten, cfads90: erloes90 - kosten });
   168	  }
   169	  return jahre;
   170	}

   174	/**
   175	 * Tilgungsplan über T Jahre (Arrays Länge T, Index 0 = Jahr 1).
   176	 * profil: annuitaet | linear | endfaellig | skulptiert (dienst = Schuldendienst je Jahr vorgegeben)
   177	 */
   178	export function tilgungsplan({ betrag, zins, jahre, profil = "annuitaet", anlauf = 0, dienst = null, T = jahre }) {
   179	  const zinsen = new Array(T).fill(0);
   180	  const tilgung = new Array(T).fill(0);
   181	  const rest = new Array(T).fill(0);
   182	  let r = Math.max(0, betrag);
   183	  const n = Math.max(1, Math.min(jahre, T));
   184	  const a = Math.min(anlauf, n - 1);
   185	  const rate = profil === "annuitaet" ? annuitaet(r, zins, n - a) : 0;
   186	  for (let t = 1; t <= T; t++) {
   187	    const i = t - 1;
   188	    if (r <= 0.005 || t > n) {
   189	      rest[i] = Math.max(r, 0);
   190	      continue;
   191	    }
   192	    zinsen[i] = r * zins;
   193	    let tg = 0;
   194	    if (t <= a) tg = 0;
   195	    else if (profil === "annuitaet") tg = rate - zinsen[i];
   196	    else if (profil === "linear") tg = betrag / (n - a);
   197	    else if (profil === "endfaellig") tg = t === n ? r : 0;
   198	    else if (profil === "skulptiert") tg = (dienst?.[i] || 0) - zinsen[i];
   199	    if (t === n) tg = r; // Restschuld im letzten Jahr vollständig
   200	    tg = Math.min(tg, r);
   201	    tilgung[i] = tg;
   202	    r -= tg;
   203	    rest[i] = Math.max(r, 0);
   204	  }
   205	  return { zinsen, tilgung, rest, dienst: zinsen.map((z, i) => z + tilgung[i]) };
   206	}

   208	/**
   209	 * DSCR-Skulptierung: Schuldendienst_t = CFADS_t ÷ Ziel-DSCR über den Tenor; Darlehen = Barwert dieses Schuldendienstes zum
   210	 * Kreditzins (so ist die Restschuld am Laufzeitende genau null). Obergrenze fkMax × Investition (dann proportional gekürzt).
   211	 */
   212	export function skulptieren({ cfads, zins, jahre, dscrZiel, maxBetrag = Infinity }) {
   213	  const n = Math.min(jahre, cfads.length);
   214	  const ds = cfads.slice(0, n).map((c) => Math.max(0, c) / dscrZiel);
   215	  const bw = ds.reduce((s, d, i) => s + d / Math.pow(1 + zins, i + 1), 0);
   216	  const faktor = bw > maxBetrag ? maxBetrag / bw : 1;
   217	  return { betrag: bw * faktor, dienst: ds.map((d) => d * faktor), begrenzt: faktor < 1, faktor };
   218	}

   222	/** AfA-Reihe (Länge T): linear bzw. degressiv mit Wechsel zur linearen AfA, ab Jahr `start` (1-basiert) */
   223	export function afaReihe(basis, jahre, T, { degressivSatz = 0, start = 1 } = {}) {
   224	  const out = new Array(T).fill(0);
   225	  let rest = basis;
   226	  for (let t = start; t <= T && rest > 0.5; t++) {
   227	    const verbleibend = Math.max(1, jahre - (t - start));
   228	    const linear = rest / verbleibend;
   229	    const deg = degressivSatz > 0 ? rest * Math.min(degressivSatz, 3 / jahre) : 0;
   230	    const a = Math.min(rest, Math.max(linear, deg));
   231	    out[t - 1] = a;
   232	    rest -= a;
   233	  }
   234	  return out;
   235	}

   237	function steuernRechnen({ e, jahre, zinsen, abzug, afa, ifbJahr1 = 0 }) {
   238	  const T = jahre.length;
   239	  const steuer = new Array(T).fill(0);
   240	  if (!e.mitSteuern || !e.steuersatz) return steuer;
   241	  let vortrag = ifbJahr1; // IFB wirkt als zusätzlicher Abzug im Anschaffungsjahr
   242	  for (let i = 0; i < T; i++) {
   243	    let gewinn = jahre[i].cfads50 - (zinsen?.[i] || 0) - (abzug?.[i] || 0) - (afa?.[i] || 0);
   244	    if (i === 0) gewinn -= vortrag, (vortrag = 0);
   245	    if (gewinn < 0) {
   246	      vortrag += -gewinn;
   247	      continue;
   248	    }
   249	    const verrechnet = Math.min(vortrag, gewinn);
   250	    vortrag -= verrechnet;
   251	    steuer[i] = (gewinn - verrechnet) * e.steuersatz;
   252	  }
   253	  return steuer;
   254	}

   261	/** Kredit auf Mindest-DSCR begrenzen (Schuldendienst ist linear im Betrag) */
   262	function dscrFaktor(cf, dienst, ziel) {
   263	  let min = Infinity;
   264	  dienst.forEach((d, i) => {
   265	    if (d > 0.5) min = Math.min(min, cf[i] / d);
   266	  });
   267	  if (!Number.isFinite(min)) return 1;
   268	  return min >= ziel ? 1 : Math.max(0, min / ziel);
   269	}

   271	/** Aufbau einer Variante: { fk, tranchen:[{betrag,zins,plan}], dienst, zinsen, einmal0, leasing? } */
   272	function varianteAufbauen(art, e, jahre) {
   273	  const T = jahre.length;
   274	  const C = e.capex;
   275	  const cfBasis = jahre.map((j) => (e.dscrBasis === "p50" ? j.cfads50 : j.cfads90));
   276	  const tranche = (betrag, zins, laufzeit, profil, anlauf = 0, dienst = null) => ({ betrag, zins, laufzeit, profil, anlauf, plan: tilgungsplan({ betrag, zins, jahre: laufzeit, profil, anlauf, dienst, T }) });
   277	  const skaliert = (tranchen) => {
   278	    const dienst = nullen(T);
   279	    for (const tr of tranchen) tr.plan.dienst.forEach((d, i) => (dienst[i] += d));
   280	    const f = dscrFaktor(cfBasis, dienst, e.dscrZiel);
   281	    if (f >= 1) return { tranchen, begrenzt: false };
   282	    return { tranchen: tranchen.map((tr) => tranche(tr.betrag * f, tr.zins, tr.laufzeit, tr.profil, tr.anlauf, tr.plan && tr.profil === "skulptiert" ? tr.plan.dienst.map((d) => d * f) : null)), begrenzt: true };
   283	  };
   284	  if (art === "ek") return { tranchen: [], begrenzt: false, ekAnteil: C };
   285	  if (art === "annuitaet") return { ...skaliert([tranche(C * e.fkQuote, e.zins, e.laufzeitKredit, "annuitaet", e.tilgungsfrei)]), dsra: true, gebuehr: true };
   286	  if (art === "skulptiert") {
   287	    const s = skulptieren({ cfads: cfBasis, zins: e.zins, jahre: e.laufzeitKredit, dscrZiel: e.dscrZiel, maxBetrag: C * Math.max(e.fkQuote, 0) });
   288	    return { tranchen: [tranche(s.betrag, e.zins, e.laufzeitKredit, "skulptiert", 0, s.dienst)], begrenzt: s.begrenzt, dsra: true, gebuehr: true, skulptur: s };
   289	  }
   290	  if (art === "foerder") {
   291	    const fkL = FINANZIERUNG.foerderkredit[e.land] || FINANZIERUNG.foerderkredit.DE;
   292	    const betrag = C * Math.min(e.fkQuote, fkL.anteilMax);
   293	    return { ...skaliert([tranche(betrag, e.foerderZins, e.foerderLaufzeit, fkL.tilgung, e.foerderAnlauf)]), dsra: false, gebuehr: false };
   294	  }
   295	  if (art === "misch") {
   296	    const fkL = FINANZIERUNG.foerderkredit[e.land] || FINANZIERUNG.foerderkredit.DE;
   297	    const a = C * Math.min(e.foerderAnteil, fkL.anteilMax, e.fkQuote);
   298	    const b = Math.max(0, C * e.fkQuote - a);
   299	    return { ...skaliert([tranche(a, e.foerderZins, e.foerderLaufzeit, fkL.tilgung, e.foerderAnlauf), tranche(b, e.zins, e.laufzeitKredit, "annuitaet", e.tilgungsfrei)]), dsra: true, gebuehr: true, gebuehrBasis: b };
   300	  }
   301	  if (art === "mietkauf") return { tranchen: [tranche(C * (1 - e.mietkaufAnzahlung), e.mietkaufZins, e.mietkaufLaufzeit, "annuitaet")], begrenzt: false };
   302	  if (art === "leasing") {
   303	    const L = e.leasingLaufzeit;
   304	    const rw = C * e.leasingRestwert;
   305	    const bar = C * (1 - e.leasingSonderzahlung) - rw / Math.pow(1 + e.leasingZins, L);
   306	    const rate = annuitaet(Math.max(0, bar), e.leasingZins, L);
   307	    return { tranchen: [], begrenzt: false, leasing: { rate, L, rw, sonder: C * e.leasingSonderzahlung, barwert: bar } };
   308	  }
   309	  throw new Error(`Unbekannte Variante ${art}`);
   310	}

   312	function amortisation(cf) {
   313	  let kum = cf[0];
   314	  if (kum >= 0) return 0;
   315	  for (let t = 1; t < cf.length; t++) {
   316	    const vorher = kum;
   317	    kum += cf[t];
   318	    if (kum >= 0 && cf[t] > 0) return r2(t - 1 + -vorher / cf[t]);
   319	  }
   320	  return null;
   321	}

   323	/**
   324	 * Eine Finanzierungsvariante vollständig rechnen.
   325	 * @returns Kennzahlen + Reihen (Index 0 = Jahr 0 bei cfEk und liquiditaet, sonst Jahr 1)
   326	 */
   327	export function varianteRechnen(art, e, jahre = betriebsJahre(e)) {
   328	  const T = jahre.length;
   329	  const C = e.capex;
   330	  const v = varianteAufbauen(art, e, jahre);
   331	  const dienst = nullen(T);
   332	  const zinsen = nullen(T);
   333	  const tilgung = nullen(T);
   334	  const rest = nullen(T);
   335	  for (const tr of v.tranchen) {
   336	    for (let i = 0; i < T; i++) {
   337	      dienst[i] += tr.plan.dienst[i];
   338	      zinsen[i] += tr.plan.zinsen[i];
   339	      tilgung[i] += tr.plan.tilgung[i];
   340	      rest[i] += tr.plan.rest[i];
   341	    }
   342	  }
   343	  const fk = summe(v.tranchen.map((t) => t.betrag));
   344	  const gebuehren = v.gebuehr ? (v.gebuehrBasis ?? fk) * e.gebuehrPct : 0;
   345	  const erstDienst = dienst.find((d) => d > 0.5) || 0;
   346	  const dsra = v.dsra && fk > 0 ? (erstDienst * e.dsraMonate) / 12 : 0;
   347	  const kreditEnde = v.tranchen.length ? Math.max(...v.tranchen.map((t) => Math.min(t.laufzeit, T))) : 0;
   348	
   349	  // Leasing: Raten statt Schuldendienst, Kaufoption am Ende (Restwert als Einmalzahlung)
   350	  const einmal = nullen(T);
   351	  const leasingRate = nullen(T);
   352	  let ek0;
   353	  if (v.leasing) {
   354	    for (let t = 1; t <= v.leasing.L; t++) leasingRate[t - 1] = v.leasing.rate;
   355	    einmal[v.leasing.L - 1] += v.leasing.rw;
   356	    ek0 = v.leasing.sonder;
   357	  } else {
   358	    ek0 = C - fk + gebuehren + dsra;
   359	  }
   360	  if (dsra > 0 && kreditEnde > 0) einmal[kreditEnde - 1] -= dsra; // Freigabe der DSRA im letzten Kreditjahr
   361	  const schuldendienst = dienst.map((d, i) => d + leasingRate[i]);
   362	
   363	  // Steuern (Näherung): Eigentümer schreibt ab; beim Leasing sind die Raten abzugsfähig, AfA erst nach Ausübung der Kaufoption
   364	  const degr = e.afaDegressiv ? FINANZIERUNG.afaDegressiv[e.land] || 0 : 0;
   365	  const afa = v.leasing
   366	    ? afaReihe(v.leasing.rw, Math.max(1, e.afaJahre - v.leasing.L), T, { start: v.leasing.L + 1 })
   367	    : afaReihe(C, e.afaJahre, T, { degressivSatz: degr });
   368	  const ifbJahr1 = !v.leasing && e.ifb && e.land === "AT" ? Math.min(C, FINANZIERUNG.ifbAt.maxBemessung) * FINANZIERUNG.ifbAt.satz : 0;
   369	  const abzug = leasingRate.map((r, i) => r + (i === 0 ? gebuehren : 0));
   370	  const steuer = steuernRechnen({ e, jahre, zinsen, abzug, afa, ifbJahr1 });
   371	
   372	  const cfEk50 = [-ek0, ...jahre.map((j, i) => j.cfads50 - schuldendienst[i] - einmal[i])];
   373	  const cfEk90 = [-ek0, ...jahre.map((j, i) => j.cfads90 - schuldendienst[i] - einmal[i])];
   374	  const cfEkSt = cfEk50.map((c, t) => (t === 0 ? c : c - steuer[t - 1]));
   375	  const liquiditaet = [];
   376	  cfEkSt.reduce((s, c) => (liquiditaet.push(s + c), s + c), 0);
   377	
   378	  const dscr = (cfKey) => {
   379	    const w = [];
   380	    jahre.forEach((j, i) => {
   381	      if (schuldendienst[i] > 0.5) w.push(j[cfKey] / schuldendienst[i]);
   382	    });
   383	    return w.length ? { min: Math.min(...w), mittel: summe(w) / w.length, reihe: w } : { min: null, mittel: null, reihe: [] };
   384	  };
   385	  const d50 = dscr("cfads50");
   386	  const d90 = dscr("cfads90");
   387	  // LLCR zu Beginn: Barwert CFADS (Bemessungsbasis) über die Kreditlaufzeit zum gewichteten Kreditzins ÷ Darlehen
   388	  let llcr = null;
   389	  if (fk > 0) {
   390	    const zinsMittel = summe(v.tranchen.map((t) => t.betrag * t.zins)) / fk;
   391	    const cf = jahre.slice(0, kreditEnde).map((j) => (e.dscrBasis === "p50" ? j.cfads50 : j.cfads90));
   392	    llcr = (npv(zinsMittel, [0, ...cf]) + dsra) / fk;
   393	  }
   394	  const gesamtkosten = v.leasing
   395	    ? summe(leasingRate) + v.leasing.sonder + v.leasing.rw - C
   396	    : summe(zinsen) + gebuehren;
   397	  const ekEingesetzt = Math.max(ek0, 1);
   398	  const ekIrr = ek0 > 0 ? irr(cfEk50) : null;
   399	  return {
   400	    art,
   401	    label: VARIANTEN[art].label,
   402	    kurz: VARIANTEN[art].kurz,
   403	    kapitalbedarfEk: ek0,
   404	    fremdkapital: fk,
   405	    fkQuote: C > 0 ? fk / C : 0,
   406	    fkBegrenzt: !!v.begrenzt,
   407	    gebuehren,
   408	    dsra,
   409	    tranchen: v.tranchen.map((t) => ({ betrag: t.betrag, zins: t.zins, laufzeit: t.laufzeit, profil: t.profil, anlauf: t.anlauf })),
   410	    leasing: v.leasing ? { rate: v.leasing.rate, laufzeit: v.leasing.L, restwert: v.leasing.rw, sonderzahlung: v.leasing.sonder } : null,
   411	    rateJahr1: schuldendienst[0] || 0,
   412	    schuldendienstMittel: kreditEnde || v.leasing ? summe(schuldendienst) / (v.leasing ? v.leasing.L : kreditEnde) : 0,
   413	    dscrMin: d50.min, dscrMittel: d50.mittel, dscrMinP90: d90.min, dscrMittelP90: d90.mittel,
   414	    llcr,
   415	    ekIrr,
   416	    ekIrrP90: ek0 > 0 ? irr(cfEk90) : null,
   417	    ekIrrNachSteuer: e.mitSteuern && ek0 > 0 ? irr(cfEkSt) : null,
   418	    npvEk: npv(e.diskontEk, e.mitSteuern ? cfEkSt : cfEk50),
   419	    amortisationEk: amortisation(e.mitSteuern ? cfEkSt : cfEk50),
   420	    roiEk: (summe((e.mitSteuern ? cfEkSt : cfEk50).slice(1)) - ek0) / ekEingesetzt,
   421	    gesamtkostenFinanzierung: gesamtkosten,
   422	    steuernSumme: summe(steuer),
   423	    ifb: ifbJahr1,
   424	    minLiquiditaet: Math.min(...liquiditaet),
   425	    reihen: { schuldendienst, zinsen, tilgung, restschuld: rest, steuer, afa, cfEk: cfEk50, cfEkNachSteuer: cfEkSt, liquiditaet, dscr: jahre.map((j, i) => (schuldendienst[i] > 0.5 ? j.cfads50 / schuldendienst[i] : null)) },
   426	  };
   427	}

   429	/** Projektkennzahlen (unverschuldet, P50): Projekt-IRR, NPV zum WACC, Amortisation, LCOE, ROI */
   430	export function projektKennzahlen(e, jahre = betriebsJahre(e)) {
   431	  const cf = [-e.capex, ...jahre.map((j) => j.cfads50)];
   432	  const cf90 = [-e.capex, ...jahre.map((j) => j.cfads90)];
   433	  const wacc = e.fkQuote * e.zins + (1 - e.fkQuote) * e.diskontEk;
   434	  const bwKosten = e.capex + jahre.reduce((s, j) => s + j.kosten / Math.pow(1 + wacc, j.jahr), 0);
   435	  const bwEnergie = jahre.reduce((s, j) => s + j.erzeugung50 / Math.pow(1 + wacc, j.jahr), 0);
   436	  return {
   437	    projektIrr: irr(cf),
   438	    projektIrrP90: irr(cf90),
   439	    npv: npv(wacc, cf),
   440	    wacc,
   441	    amortisation: amortisation(cf),
   442	    lcoeCt: bwEnergie > 0 ? (bwKosten / bwEnergie) * 100 : null,
   443	    roi: (summe(cf.slice(1)) - e.capex) / e.capex,
   444	    ertragJahr1Mwh: (jahre[0]?.erzeugung50 || 0) / 1000,
   445	    erloesJahr1: jahre[0]?.erloes50 || 0,
   446	    cfadsJahr1: jahre[0]?.cfads50 || 0,
   447	    capexProKwp: e.capex / (e.mwp * 1000),
   448	  };
   449	}

   451	/** Stresstest der gewählten Variante: P90, Zins +1 Pp., Capex +10 %, Preis −10 % */
   452	export function szenarienRechnen(e, art = "skulptiert") {
   453	  const fall = (label, aend) => {
   454	    const x = { ...e, ...aend };
   455	    const r = varianteRechnen(art, x);
   456	    return { label, ekIrr: r.ekIrr, dscrMin: r.dscrMin, kapitalbedarfEk: r.kapitalbedarfEk, npvEk: r.npvEk, fkBegrenzt: r.fkBegrenzt };
   457	  };
   458	  return [
   459	    fall("Basis (P50)", {}),
   460	    fall("Ertrag P90 dauerhaft", { p50: e.p90 }),
   461	    fall("Zins +1 Prozentpunkt", { zins: e.zins + 0.01, foerderZins: e.foerderZins + 0.01, leasingZins: e.leasingZins + 0.01, mietkaufZins: e.mietkaufZins + 0.01 }),
   462	    fall("Investition +10 %", { capex: e.capex * 1.1 }),
   463	    fall("Erlöspreis −10 %", { preisCt: e.preisCt * 0.9, merchantCt: e.merchantCt * 0.9 }),
   464	  ];
   465	}
