   104	/** Stammdaten einer Gesellschaft für Belege und SEPA: {id, name, anschrift, land, uid, firmenbuch, iban, bic, erpnextCompany, spv} */
   105	export function gesellschaftFuer(db, id) {
   106	  const e = einstellungen(db);
   107	  // Nur Name bekannt (Feld „Eigentümergesellschaft“ der Anlage, noch ohne Verzeichnis-Eintrag): Anschrift/UID fehlen → Pflichtangaben-Warnung
   108	  if (String(id || "").startsWith("NAME:")) {
   109	    const name = String(id).slice(5);
   110	    return { id, name, anschrift: "", land: "", uid: "", firmenbuch: "", iban: "", bic: "", beispielIban: false, erpnextCompany: name, spv: true, bekannt: true, nurName: true };
   111	  }
   112	  const eintrag = id && id !== OEKOINVEST_ID ? verzeichnis(db).find((x) => x.id === id) : null;
   113	  if (istOekoInvest(eintrag)) id = OEKOINVEST_ID; // ÖkoInvest GmbH steht auch im Verzeichnis – Stammdaten aus den Finanzeinstellungen
   114	  if (!id || id === OEKOINVEST_ID) return { id: OEKOINVEST_ID, name: e.firma.name, anschrift: e.firma.anschrift, land: e.firma.land, uid: e.firma.uid, firmenbuch: e.firma.firmenbuch, iban: e.auftraggeber.iban, bic: e.auftraggeber.bic, beispielIban: e.auftraggeber.beispiel, erpnextCompany: "ÖkoInvest GmbH", spv: false, bekannt: true };
   115	  const g = eintrag || {};
   116	  const bank = g.bank || g.bankverbindung || {};
   117	  return {
   118	    id, name: g.name || g.firma || g.titel || id, anschrift: adresseText(g.anschrift || g.sitz || g.adresse) || "", land: g.land || g.sitzLand || "",
   119	    uid: g.uid || g.uidNummer || "", firmenbuch: [g.firmenbuch || g.registernummer || g.fn, g.gericht].filter(Boolean).join(", "),
   120	    iban: g.iban || bank.iban || "", bic: g.bic || bank.bic || "", beispielIban: false, erpnextCompany: g.erpCompany || g.erpnextCompany || g.company || g.name || "", spv: true, bekannt: !!g.id,
   121	  };
   122	}

   124	/**
   125	 * Eigentümergesellschaft einer Anlage. Felder der Fachprüfung: `eigentuemer` (Name, Standard „ÖkoInvest GmbH“) und
   126	 * `eigentuemerId` (Kennung im Gesellschaftsverzeichnis, folgt). Reihenfolge: Verzeichnis-Eintrag → Name → ÖkoInvest GmbH.
   127	 */
   128	export function anlageGesellschaft(db, a) {
   129	  if (!a) return gesellschaftFuer(db, OEKOINVEST_ID);
   130	  const roh = a.eigentuemerId ?? a.eigentuemerGesellschaftId ?? a.gesellschaftId ?? null;
   131	  const id = (typeof roh === "object" ? roh?.id : roh) || null;
   132	  const name = (typeof a.eigentuemer === "string" ? a.eigentuemer : a.eigentuemer?.name || "").trim();
   133	  const standard = !name || name === "ÖkoInvest GmbH";
   134	  if (id && id !== OEKOINVEST_ID) {
   135	    const g = gesellschaftFuer(db, id);
   136	    if (g.bekannt) return g;
   137	    if (standard) return g;
   138	  }
   139	  if (standard) return gesellschaftFuer(db, OEKOINVEST_ID);
   140	  // Name ohne Kennung: Eintrag gleichen Namens im Verzeichnis (Anschrift, UID), sonst nur der Name (Pflichtangaben-Warnung)
   141	  const treffer = verzeichnis(db).find((x) => (x.name || x.firma) === name && x.status !== "aufgeloest");
   142	  return treffer ? gesellschaftFuer(db, treffer.id) : gesellschaftFuer(db, `NAME:${name}`);
   143	}

   146	/** Stromabnehmer (Rechnungsempfänger Contracting): eigenes Feld der Fachprüfung, sonst Kunde/Organisation der Anlage */
   147	export function stromabnehmerVon(roh, a) {
   148	  const s = roh?.stromabnehmer;
   149	  const id = typeof s === "string" ? s : s?.id || null;
   150	  const kontoId = roh?.stromabnehmerId || roh?.stromabnehmerKontoId || s?.kontoId || (id && id.startsWith("U-") ? id : null);
   151	  const organisationId = roh?.stromabnehmerOrganisationId || s?.organisationId || (id && id.startsWith("O-") ? id : null);
   152	  if (kontoId || organisationId) return { kontoId: kontoId || null, organisationId: organisationId || null, quelle: "stromabnehmer" };
   153	  return { kontoId: a?.kunde?.id || roh?.kundeId || null, organisationId: a?.organisation?.id || roh?.organisationId || null, quelle: "kunde" };

   158	/**
   159	 * Freigabe einer Auszahlung an ein Konto (Pacht-Gutschrift) – verbindlich über das KYC-Modul: kycFreigabeDb(db, "auszahlung",
   160	 * {kontoId, flaecheId}) (= kycFreigabe/kycSperre des KYC-Teams: Verifizierung + geprüfte IBAN + Eigentumsnachweis der Fläche;
   161	 * Geltung nach PLATTFORM_KYC_GATING bzw. Compliance-Einstellung – in der Entwicklung nur Pilot-Konten). Zusätzlich gilt immer:
   162	 * Die Empfänger-IBAN stammt ausschließlich aus der Verifizierung – ohne gültige IBAN keine Auszahlung.
   163	 * Nur wenn das KYC-Modul in dieser Demo-Datenbank (noch) keine Daten hat, gelten die Demo-Stammdaten db.finanzen.kycDemo.
   164	 */
   165	export function kycAuszahlung(db, { kontoId, organisationId = null, flaecheId = null }) {
   166	  if (db.kyc) {
   167	    const fr = kycFreigabeDb(db, "auszahlung", { kontoId, organisationId, flaecheId });
   168	    const bank = kycFallFuer(db, { kontoId, organisationId })?.bank || {};
   169	    const gruende = [...fr.gruende];
   170	    if (!F.ibanPruefen(bank.iban)) gruende.push(bank.iban ? "Die IBAN der Bankverbindung ist ungültig." : "In der Verifizierung ist keine Bankverbindung hinterlegt – ohne IBAN keine Auszahlung.");
   171	    return { quelle: "kyc", aktiv: fr.aktiv !== false, status: fr.status, verifiziert: fr.status === "verifiziert", ok: gruende.length === 0, gruende, iban: bank.iban || "", bic: bank.bic || "", kontoinhaber: bank.kontoinhaber || "" };
   172	  }
   173	  const d = db.finanzen?.kycDemo?.[kontoId];
   174	  const gruende = [];
   175	  if (!d || d.status !== "verifiziert") gruende.push("Die Verifizierung ist nicht abgeschlossen.");
   176	  if (!d?.iban || !d?.bankGeprueft) gruende.push("Es fehlt eine geprüfte Bankverbindung (IBAN, Kontoinhaber = verifizierte Person/Firma).");
   177	  else if (!F.ibanPruefen(d.iban)) gruende.push("Die IBAN der geprüften Bankverbindung ist ungültig.");
   178	  if (flaecheId && d?.status === "verifiziert" && !(d.eigentum || []).includes(flaecheId)) gruende.push("Für diese Fläche liegt kein geprüfter Eigentumsnachweis vor.");
   179	  return { quelle: "demo", status: d?.status || "offen", verifiziert: d?.status === "verifiziert", ok: gruende.length === 0, gruende, iban: d?.iban || "", bic: d?.bic || "", kontoinhaber: d?.kontoinhaber || "" };
   180	}

   182	/** Rechnungsempfänger bzw. Gutschriftsempfänger (Name, Anschrift, Land, UID) – aus KYC, sonst aus dem Konto */
   183	export function parteiFuer(db, { kontoId = null, organisationId = null }) {
   184	  const u = kontoId ? nutzerById(db, kontoId) : null;
   185	  const org = organisationId ? (db.organisationen || []).find((o) => o.id === organisationId) : null;
   186	  const fall = db.kyc ? kycFallFuer(db, { kontoId, organisationId }) : null;
   187	  const demo = kontoId ? db.finanzen?.kycDemo?.[kontoId] : null;
   188	  const firma = fall?.firma?.name || org?.name || demo?.firma || u?.firma || "";
   189	  const person = u ? nameVon(db, u.id) : "";
   190	  return {
   191	    name: firma || person || "–",
   192	    zuHaenden: firma && person ? person : "",
   193	    anschrift: adresseText(fall?.firma?.anschrift || fall?.firma?.sitz || fall?.person?.anschrift || fall?.anschrift || demo?.anschrift) || "",
   194	    land: fall?.firma?.land || fall?.person?.land || demo?.land || org?.land || "",
   195	    uid: fall?.firma?.uid || fall?.steuer?.uid || demo?.uid || "",
   196	    email: u?.email || "",
   197	    kleinunternehmer: !!demo?.kleinunternehmer,
   198	  };
   199	}

   219	/** Klärfall anlegen bzw. offenen Klärfall zum selben Schlüssel aktualisieren */
   220	export function klaerfallMelden(db, { art, schluessel, bezug, bezugText, gruende, laufId }) {
   221	  const offen = db.finanzen.klaerfaelle.find((k) => k.schluessel === schluessel && k.status === "offen");
   222	  if (offen) {
   223	    Object.assign(offen, { gruende, zuletztAm: jetzt(), anzahl: (offen.anzahl || 1) + 1, laufId });
   224	    return { klaerfall: offen, neu: false };
   225	  }
   226	  const k = { id: neueId("KL"), art, schluessel, bezug, bezugText, gruende, status: "offen", erstelltAm: jetzt(), zuletztAm: jetzt(), anzahl: 1, laufId, kommentar: "", verlauf: [] };
   227	  db.finanzen.klaerfaelle.push(k);
   228	  return { klaerfall: k, neu: true };

   233	/** Beleg erzeugen (bzw. im Trockenlauf nur berechnen) */
   234	export function belegAnlegen(db, n, e, { trockenlauf = false, melden = true, heute = F.heute(), erstelltAm = null } = {}) {
   235	  const art = F.ABRECHNUNG_ARTEN[e.art];
   236	  const netto = F.summe(e.positionen, "betrag");
   237	  const steuer = F.steuerBerechnen(netto, e.steuerVorlage);
   238	  const ges = gesellschaftFuer(db, e.gesellschaftId);
   239	  const beleg = {
   240	    id: trockenlauf ? null : neueId("AR"),
   241	    nummer: trockenlauf ? "(Vorschau)" : naechsteNummer(db, art.kreis, F.jahrVon(heute)),
   242	    art: e.art,
   243	    beleg: e.art === "pacht" ? "gutschrift" : "rechnung",
   244	    titel: TITEL[e.art],
   245	    empfaengerId: e.empfaengerId || null,
   246	    organisationId: e.organisationId || null,
   247	    partei: e.partei,
   248	    // Gutschrift: ausstellende Gesellschaft = Pächter; Rechnung: leistende Gesellschaft (Contracting: Eigentümergesellschaft)
   249	    gesellschaftId: ges.id,
   250	    aussteller: { name: ges.name, anschrift: ges.anschrift, land: ges.land, uid: ges.uid, firmenbuch: ges.firmenbuch },
   251	    bezug: e.bezug,
   252	    bezugText: e.bezugText,
   253	    schluessel: e.schluessel,
   254	    zeitraum: e.zeitraum,
   255	    rechnungsdatum: heute,
   256	    faelligAm: e.faelligAm,
   257	    positionen: e.positionen,
   258	    netto,
   259	    steuer: { vorlage: steuer.vorlage, label: steuer.label, satz: steuer.satz, art: steuer.art, betrag: steuer.betrag, text: steuer.text, rechtsgrundlage: steuer.rechtsgrundlage },
   260	    brutto: steuer.brutto,
   261	    waehrung: "EUR",
   262	    erklaerung: e.erklaerung || [],
   263	    zaehlerstaende: e.zaehlerstaende || [],
   264	    hinweise: e.hinweise || [],
   265	    kennzahlen: e.kennzahlen || {},
   266	    status: "offen",
   267	    bezahltAm: null,
   268	    zahlungslaufId: null,
   269	    mahnungen: [],
   270	    erpnext: { doctype: art.erpnext, name: null },
   271	    laufId: e.laufId || null,
   272	    erstelltAm: erstelltAm || jetzt(),
   273	    erstelltVon: n?.id || null,
   274	  };
   275	  beleg.pflichtFehlt = F.pflichtangabenPruefen({ ...beleg, empfaenger: beleg.partei });
   276	  if (trockenlauf) return beleg;
   277	  db.finanzen.belege.push(beleg);
   278	  klaerfallSchliessen(db, e.schluessel, `Beleg ${beleg.nummer} erzeugt.`);
   279	  protokoll(db, n?.id === SYSTEM_FINANZEN.id ? null : n, "abrechnung_erzeugt", "abrechnung", beleg.id, `${art.label} ${beleg.nummer}: ${F.euro(beleg.brutto)} (${e.bezugText})`);
   280	  if (melden) belegMelden(db, beleg, "neu");
   281	  return beleg;
   282	}

   306	/** Provisionsstatus aus dem Zahlungsstatus der Rechnung nachziehen (offen → in_rechnung → bezahlt) */
   307	export function provisionNachziehen(db, n, b) {
   308	  if (b.art !== "provision") return;
   309	  const pv = db.provisionen.find((p) => p.id === b.bezug?.provisionId);
   310	  if (!pv || pv.status === "storniert") return;
   311	  const ziel = F.provisionStatusAusRechnung(b);
   312	  const reihenfolge = ["offen", "in_rechnung", "bezahlt"];
   313	  if (reihenfolge.indexOf(ziel) <= reihenfolge.indexOf(pv.status)) return;
   314	  const von = pv.status;
   315	  pv.status = ziel;
   316	  pv.statusVerlauf = pv.statusVerlauf || [];
   317	  pv.statusVerlauf.push({ status: ziel, am: jetzt(), vonId: n?.id && n.id !== SYSTEM_FINANZEN.id ? n.id : null, vonRolle: "system", kommentar: ziel === "bezahlt" ? `Zahlung zu Rechnung ${b.nummer} verbucht.` : `Rechnung ${b.nummer} gestellt.` });
   318	  pv.rechnung = { id: b.id, nummer: b.nummer, faelligAm: b.faelligAm };
   319	  protokoll(db, n?.id === SYSTEM_FINANZEN.id ? null : n, "status", "provision", pv.id, `${von} → ${ziel} (Rechnung ${b.nummer})`);
   320	  backofficeEreignis(db, { typ: "status", art: "provision", objekt: pv, von, nach: ziel }); // Automationen
   321	}

   331	export function pachtSteuerVorlage(pt) {
   332	  return pt.steuervorlage && F.STEUERVORLAGEN[pt.steuervorlage] ? pt.steuervorlage : F.steuerVorschlag("pacht", pt.land, { option: !!pt.steuerOption, kleinunternehmer: !!pt.kleinunternehmer }).vorlage;

   335	/**
   336	 * Gesperrte Auszahlung sichtbar machen: je Eigentümer und Vertrag ein Klärfall „KYC unvollständig“ (nicht stillschweigend
   337	 * überspringen); ist die Freigabe inzwischen erteilt, wird der Klärfall geschlossen.
   338	 */
   339	export function kycKlaerfall(db, lauf, pt, partei, kyc, trocken = false) {
   340	  const schluessel = `KYC:${pt.eigentuemerId}:${pt.id}`;
   341	  if (kyc.ok) {
   342	    if (!trocken) klaerfallSchliessenKyc(db, schluessel);
   343	    return;
   344	  }
   345	  const e = { art: "pacht", schluessel, bezug: { pachtvertragId: pt.id, kontoId: pt.eigentuemerId }, bezugText: `Auszahlung an ${partei?.name || pt.eigentuemerId} (${pt.bezeichnung})`, gruende: [`KYC unvollständig: ${kyc.gruende.join(" ")}`] };
   346	  if (trocken) return;
   347	  const vorher = db.finanzen.klaerfaelle.find((k) => k.schluessel === schluessel && k.status === "offen");
   348	  if (vorher && vorher.laufId === lauf.id) return;
   349	  const { klaerfall, neu } = klaerfallMelden(db, { ...e, laufId: lauf.id });
   350	  lauf.klaerfaelle.push({ id: klaerfall.id, art: "pacht", bezugText: e.bezugText, gruende: e.gruende, neu });
   351	}
   352	
   353	export function klaerfallSchliessenKyc(db, schluessel, text = "KYC-Freigabe für Auszahlungen erteilt.") {
   354	  for (const k of db.finanzen.klaerfaelle.filter((x) => x.schluessel === schluessel && x.status === "offen")) Object.assign(k, { status: "erledigt", erledigtAm: jetzt(), kommentar: text });
   355	}

   357	/**
   358	 * Lauf ausführen (innerhalb einer Schreibtransaktion).
   359	 * opt: { art: pacht|contracting|provision|alle, stichtag, monat (Contracting), trockenlauf, vertragIds[], anlageIds[], vorab: { anlagen: {id: {anlage, vertrag, messpunkte, zeilen}} }, melden }
   360	 */
   361	export function laufAusfuehren(db, n, opt) {
   362	  const art = opt.art || "alle";
   363	  const stichtag = opt.stichtag || F.heute();
   364	  const monat = opt.monat || F.vormonat(stichtag.slice(0, 7));
   365	  const trocken = !!opt.trockenlauf;
   366	  const lauf = { id: trocken ? null : neueId("LA"), art, stichtag, monat, trockenlauf: trocken, am: jetzt(), vonId: n?.id && n.id !== SYSTEM_FINANZEN.id ? n.id : null, vonName: n?.id === SYSTEM_FINANZEN.id ? "Scheduler" : nameVon(db, n?.id) || "", erzeugt: [], klaerfaelle: [], vorhanden: 0, uebersprungen: 0 };
   367	  const erzeugen = (e) => {
   368	    const b = belegAnlegen(db, n, { ...e, laufId: lauf.id }, { trockenlauf: trocken, melden: opt.melden !== false, heute: opt.rechnungsdatum || F.heute() });
   369	    lauf.erzeugt.push({ id: b.id, nummer: b.nummer, art: b.art, empfaenger: b.partei?.name, zeitraum: b.zeitraum, netto: b.netto, brutto: b.brutto, bezugText: b.bezugText });
   370	    return b;
   371	  };
   372	  const klaeren = (e) => {
   373	    if (trocken) {
   374	      lauf.klaerfaelle.push({ id: null, art: e.art, bezugText: e.bezugText, gruende: e.gruende, neu: true });
   375	      return;
   376	    }
   377	    const { klaerfall, neu } = klaerfallMelden(db, { ...e, laufId: lauf.id });
   378	    lauf.klaerfaelle.push({ id: klaerfall.id, art: e.art, bezugText: e.bezugText, gruende: e.gruende, neu });
   379	  };
   380	  const anlagen = opt.vorab?.anlagen || {};
   381	
   382	  // ---------------------------------------------------------------- Pacht
   383	  if (art === "pacht" || art === "alle") {
   384	    for (const pt of db.finanzen.pachtvertraege) {
   385	      if (!["aktiv", "gekuendigt"].includes(pt.status)) continue;
   386	      if (opt.vertragIds?.length && !opt.vertragIds.includes(pt.id)) continue;
   387	      const eigentuemer = nutzerById(db, pt.eigentuemerId);
   388	      for (const f of F.abzurechnen(vertragFuerFaelligkeit(db, pt), stichtag)) {
   389	        if (aktiverBeleg(db, f.schluessel)) {
   390	          lauf.vorhanden++;
   391	          continue;
   392	        }
   393	        const bezugText = `${pt.bezeichnung} (${pt.id}) ${F.datumDe(f.von)} – ${F.datumDe(f.bis)}`;
   394	        let ertrag = { kwh: null, fehlt: [] };
   395	        if (pt.modell === "umsatz" && f.phase === "pacht") {
   396	          const a = anlagen[pt.anlageId];
   397	          ertrag = a ? F.ertragAusMonatswerten(a.zeilen, a.messpunkte, f.von, f.bis) : { kwh: null, fehlt: ["keine Anlage/Monatswerte verknüpft"] };
   398	        }
   399	        const r = F.pachtPositionen(pt, f, { indexTabelle: db.finanzen.vpi, ertragKwh: ertrag.kwh, ertragFehlt: ertrag.fehlt });
   400	        if (r.klaerung || !eigentuemer) {
   401	          klaeren({ art: "pacht", schluessel: f.schluessel, bezug: { pachtvertragId: pt.id }, bezugText, gruende: [r.klaerung || "Eigentümer-Konto nicht gefunden."] });
   402	          continue;
   403	        }
   404	        if (!(r.netto > 0)) {
   405	          lauf.uebersprungen++;
   406	          continue;
   407	        }
   408	        const partei = parteiFuer(db, { kontoId: pt.eigentuemerId });
   409	        const kyc = kycAuszahlung(db, { kontoId: pt.eigentuemerId, flaecheId: pt.flaecheId || null });
   410	        kycKlaerfall(db, lauf, pt, partei, kyc, trocken);
   411	        const pachtGesellschaft = pt.gesellschaftId || (pt.anlageId && anlagen[pt.anlageId] ? anlageGesellschaftId(db, anlagen[pt.anlageId].roh || anlagen[pt.anlageId].anlage) : OEKOINVEST_ID);
   412	        erzeugen({
   413	          art: "pacht", gesellschaftId: pachtGesellschaft, empfaengerId: pt.eigentuemerId, partei, schluessel: f.schluessel,
   414	          bezug: { pachtvertragId: pt.id, flaecheId: pt.flaecheId || null, anlageId: pt.anlageId || null, phase: f.phase },
   415	          bezugText, zeitraum: { von: f.von, bis: f.bis }, faelligAm: f.faelligAm, positionen: r.positionen, steuerVorlage: pachtSteuerVorlage(pt),
   416	          erklaerung: r.erklaerung, kennzahlen: { ...r.basis, phase: f.phase, zahlweise: f.zahlweise },
   417	          hinweise: kyc.ok ? [] : [`Auszahlung gesperrt: ${kyc.gruende.join(" ")}`],
   418	        });
   419	      }
   420	    }
   421	  }
   422	
   423	  // ---------------------------------------------------------------- Contracting (Monat)
   424	  if (art === "contracting" || art === "alle") {
   425	    const z = F.monatsZeitraum(monat);
   426	    for (const [anlageId, a] of Object.entries(anlagen)) {
   427	      if (!a.vertrag || a.vertrag.status !== "aktiv") continue;
   428	      if (opt.anlageIds?.length && !opt.anlageIds.includes(anlageId)) continue;
   429	      if (!a.vertrag.beginn || a.vertrag.beginn > z.bis) continue;
   430	      const schluessel = `CV:${anlageId}:${monat}`;
   431	      if (aktiverBeleg(db, schluessel)) {
   432	        lauf.vorhanden++;
   433	        continue;
   434	      }
   435	      const bezugText = `${a.anlage.name} (${anlageId}) ${monat.slice(5)}/${monat.slice(0, 4)}`;
   436	      const r = F.contractingAbrechnen({ vertrag: a.vertrag, anlage: a.anlage, monat, messpunkte: a.messpunkte, zeilen: a.zeilen, indexTabelle: db.finanzen.vpi });
   437	      const { kontoId, organisationId } = stromabnehmerVon(a.roh, a.anlage);
   438	      const ges = anlageGesellschaft(db, a.roh || a.anlage);
   439	      const gesellschaftId = ges.id;
   440	      if (r.ok && !ges.bekannt) {
   441	        klaeren({ art: "contracting", schluessel, bezug: { anlageId, monat }, bezugText, gruende: [`Eigentümergesellschaft ${gesellschaftId} nicht im Gesellschaftsverzeichnis – Rechnungsaussteller unklar.`] });
   442	        continue;
   443	      }
   444	      if (!r.ok || (!kontoId && !organisationId)) {
   445	        klaeren({ art: "contracting", schluessel, bezug: { anlageId, monat }, bezugText, gruende: r.ok ? ["Kein Rechnungsempfänger (Stromabnehmer bzw. Kunde/Organisation) an der Anlage."] : r.klaerung });
   446	        continue;
   447	      }
   448	      erzeugen({
   449	        art: "contracting", gesellschaftId, empfaengerId: kontoId, organisationId, partei: parteiFuer(db, { kontoId, organisationId }), schluessel,
   450	        bezug: { anlageId, vertragId: a.vertrag.id || null, monat }, bezugText,
   451	        zeitraum: z, faelligAm: F.plusTage(opt.rechnungsdatum || F.heute(), einstellungen(db).zahlungszielTage),
   452	        positionen: r.positionen, steuerVorlage: a.vertrag.steuervorlage || F.steuerVorschlag("contracting", a.anlage.land).vorlage,
   453	        zaehlerstaende: r.zaehlerstaende, hinweise: r.hinweise, kennzahlen: { gelieferteKwh: r.gelieferteKwh, arbeitspreisCt: r.arbeitspreisCt, monat, kwp: a.anlage.kwp },
   454	        erklaerung: r.gelieferteKwh !== null ? [`Gelieferter PV-Strom = Erzeugung (geeichter Erzeugungszähler) abzüglich Einspeisung ins Netz. Grundlage: Zählerstand-Monatswerte des Messstellenbetreibers.`] : ["Abrechnung der monatlichen Grundgebühr laut Contracting-Vertrag."],
   455	      });
   456	    }
   457	  }
   458	
   459	  // ---------------------------------------------------------------- Provision
   460	  if (art === "provision" || art === "alle") {
   461	    for (const pv of db.provisionen) {
   462	      if (!["offen", "in_rechnung"].includes(pv.status)) continue;
   463	      const schluessel = `PV:${pv.id}`;
   464	      if (aktiverBeleg(db, schluessel)) {
   465	        lauf.vorhanden++;
   466	        continue;
   467	      }
   468	      const p = db.projekte.find((x) => x.id === pv.projektId);
   469	      const bezugText = `Provision ${pv.id} – ${p?.titel || pv.projektId}`;
   470	      if (!nutzerById(db, pv.anbieterId)) {
   471	        klaeren({ art: "provision", schluessel, bezug: { provisionId: pv.id }, bezugText, gruende: ["Anbieter-Konto nicht gefunden."] });
   472	        continue;
   473	      }
   474	      const partei = parteiFuer(db, { kontoId: pv.anbieterId, organisationId: p?.organisationId || null });
   475	      const land = partei.land || p?.land || "AT";
   476	      const vorschlag = F.steuerVorschlag("provision", land, { grundstuecksbezug: true });
   477	      const b = erzeugen({
   478	        art: "provision", gesellschaftId: OEKOINVEST_ID, empfaengerId: pv.anbieterId, organisationId: p?.organisationId || null, partei, schluessel,
   479	        bezug: { provisionId: pv.id, projektId: pv.projektId }, bezugText,
   480	        zeitraum: { von: String(pv.erstelltAm).slice(0, 10), bis: String(pv.erstelltAm).slice(0, 10) },
   481	        faelligAm: F.plusTage(opt.rechnungsdatum || F.heute(), einstellungen(db).zahlungszielTage),
   482	        positionen: F.provisionPositionen(pv, p?.titel || ""), steuerVorlage: vorschlag.vorlage,
   483	        hinweise: partei.land ? vorschlag.hinweise : [...vorschlag.hinweise, "Sitz des Anbieters unbekannt – Land des Projekts verwendet."],
   484	        erklaerung: [`Provision laut Provisionsvereinbarung: ${F.rund(pv.satz * 100, 2).toLocaleString("de-DE")} % des Kaufpreises, fällig bei Abschluss.`],
   485	        kennzahlen: { kaufpreis: pv.kaufpreis, satz: pv.satz },
   486	      });
   487	      if (!trocken) provisionNachziehen(db, n, b);
   488	    }
   489	  }
   490	
   491	  if (!trocken) {
   492	    db.finanzen.laeufe.unshift({ ...lauf, erzeugt: lauf.erzeugt.slice(0, 200) });
   493	    db.finanzen.laeufe = db.finanzen.laeufe.slice(0, 100);
   494	    protokoll(db, n?.id === SYSTEM_FINANZEN.id ? null : n, "abrechnungslauf", "finanzen", lauf.id, `Lauf ${F.LAUF_ARTEN[art]} zum ${F.datumDe(stichtag)}: ${lauf.erzeugt.length} Belege, ${lauf.klaerfaelle.length} Klärfälle`);
   495	  }
   496	  return lauf;
   497	}
