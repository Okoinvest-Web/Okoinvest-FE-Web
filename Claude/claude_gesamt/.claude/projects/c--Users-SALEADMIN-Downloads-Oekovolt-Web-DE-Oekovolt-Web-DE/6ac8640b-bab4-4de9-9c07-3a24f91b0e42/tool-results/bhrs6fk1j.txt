    84	/** EPC-Spanne (€/kWp) je Typ aus markt.js */
    85	export function capexSpanne(typ) {
    86	  if (typ === "agri") return AGRI.capexProKwp.vertikal;
    87	  if (typ === "carport") return CARPORT.capexProKwp;
    88	  if (typ === "dach") return DACH.capexProKwp;
    89	  return FREIFLAECHE.capexProKwp;
    90	}
    91	const anteileFuer = (typ) => (typ === "carport" ? PK.epcAnteile.carport : typ === "dach" ? PK.epcAnteile.dach : PK.epcAnteile.standard);
    92	const hatFlaeche = (typ) => ["freiflaeche", "agri", "hybrid", "speicher"].includes(typ);

    94	/** Parameter vervollständigen (Standardwerte aus markt.js) */
    95	export function parameterVoll(p = {}) {
    96	  const typ = p.typ || "freiflaeche";
    97	  const kostenstand = KOSTENSTAENDE[p.kostenstand] ? p.kostenstand : "schaetzung";
    98	  const q = p.trasse?.querungen || {};
    99	  return {
   100	    typ,
   101	    land: p.land || "DE",
   102	    mwp: Math.max(0, Number(p.mwp) || 0),
   103	    mwh: Math.max(0, Number(p.mwh) || 0),
   104	    flaecheHa: Math.max(0, Number(p.flaecheHa) || 0),
   105	    flaechenAnzahl: Math.max(1, Math.round(Number(p.flaechenAnzahl) || 1)),
   106	    zaunM: Math.max(0, Number(p.zaunM) || 0),
   107	    trasse: {
   108	      abschnitte: (Array.isArray(p.trasse?.abschnitte) ? p.trasse.abschnitte : []).filter((a) => VERLEGEARTEN[a?.art] && Number(a.laengeM) > 0).map((a) => ({ art: a.art, laengeM: Number(a.laengeM) })),
   109	      querungen: { strasse: Math.max(0, Math.round(Number(q.strasse) || 0)), bahn: Math.max(0, Math.round(Number(q.bahn) || 0)), gewaesser: Math.max(0, Math.round(Number(q.gewaesser) || 0)) },
   110	      quelle: p.trasse?.quelle || "",
   111	    },
   112	    speicherNetzbezug: !!p.speicherNetzbezug,
   113	    kostenstand,
   114	    contingency: Number.isFinite(Number(p.contingency)) && p.contingency !== null && p.contingency !== "" ? Math.max(0, Math.min(0.5, Number(p.contingency))) : PK.contingency[kostenstand],
   115	    fkQuote: Number.isFinite(Number(p.fkQuote)) && p.fkQuote !== null && p.fkQuote !== "" ? Math.max(0, Math.min(1, Number(p.fkQuote))) : FINANZIERUNG.fremdkapitalQuote,
   116	    zinsBau: Number.isFinite(Number(p.zinsBau)) && p.zinsBau !== null && p.zinsBau !== "" ? Math.max(0, Math.min(0.2, Number(p.zinsBau))) : (FINANZIERUNG.projekt.referenzzinsPct + FINANZIERUNG.projekt.margePct) / 100,
   117	    bauzeitMonate: Math.max(1, Math.min(36, Math.round(Number(p.bauzeitMonate) || PK.bauzeitMonate[typ] || 6))),
   118	    gutachten: (Array.isArray(p.gutachten) ? p.gutachten : []).filter((g) => g?.schluessel).map((g) => ({ schluessel: String(g.schluessel), titel: String(g.titel || g.schluessel) })),
   119	  };
   120	}

   124	/** Kabelsysteme 20 kV nach Anschlussleistung (netzanschluss.js) */
   125	export function kabelSysteme(mw) {
   126	  return Math.max(1, Math.ceil((Number(mw) || 0) / FREIFLAECHE.netz.msSystemMw - 1e-9));

   129	/**
   130	 * Vorschlagspositionen aus den Parametern. Jede Position: { schluessel, gewerk, bezeichnung, art, menge, einheit, ep,
   131	 * satz, basis, quelle }.
   132	 */
   133	export function vorschlag(parameter) {
   134	  const p = parameterVoll(parameter);
   135	  const kwp = p.mwp * 1000;
   136	  const pos = [];
   137	  const add = (x) => pos.push({ art: "menge", satz: null, basis: null, ...x });
   138	  const spanne = capexSpanne(p.typ);
   139	  const anteile = anteileFuer(p.typ);
   140	  const pv = kwp > 0 && p.typ !== "speicher";
   141	
   142	  // ---- Entwicklung
   143	  const gutachtenSumme = p.gutachten.reduce((s, g) => s + (PK.gutachtenEur[g.schluessel] ?? PK.gutachtenEur.standard), 0);
   144	  const entwGesamt = kwp * FREIFLAECHE.entwicklungProKwp.min > 0 ? (kwp * (FREIFLAECHE.entwicklungProKwp.min + FREIFLAECHE.entwicklungProKwp.max)) / 2 : 0;
   145	  const rest = Math.max(kwp * FREIFLAECHE.entwicklungProKwp.min * 0.5, entwGesamt - gutachtenSumme);
   146	  if (hatFlaeche(p.typ) && p.flaecheHa > 0) {
   147	    const proHa = (PACHT.reservierungProHa.min + PACHT.reservierungProHa.max) / 2;
   148	    add({ schluessel: "ent_pacht", gewerk: "ent_pacht", bezeichnung: `Reservierungsentgelt ${PACHT.reservierungJahre.ueblich} Jahre bis Baubeginn`, menge: r2(p.flaecheHa * PACHT.reservierungJahre.ueblich), einheit: "ha·a", ep: proHa, quelle: "markt.js PACHT.reservierungProHa (Mitte)" });
   149	  }
   150	  for (const k of ["planung", "gebuehren", "recht", "netzpruefung"]) {
   151	    if (!rest || !PK.entwicklungAnteile[k]) continue;
   152	    add({ schluessel: `ent_${k}`, gewerk: `ent_${k}`, bezeichnung: GEWERKE[`ent_${k}`].label, menge: 1, einheit: "psch", ep: r2(rest * PK.entwicklungAnteile[k]), quelle: "markt.js FREIFLAECHE.entwicklungProKwp (Mitte) abzgl. Gutachten" });
   153	  }
   154	  for (const g of p.gutachten) {
   155	    add({ schluessel: `gutachten:${g.schluessel}`, gewerk: "ent_gutachten", bezeichnung: g.titel, menge: 1, einheit: "Stk", ep: PK.gutachtenEur[g.schluessel] ?? PK.gutachtenEur.standard, quelle: "markt.js PROJEKTKOSTEN.gutachtenEur" });
   156	  }
   157	
   158	  // ---- Grundstück
   159	  const tl = trassenLaenge(p);
   160	  if (tl > 0) add({ schluessel: "gr_dienstbarkeit", gewerk: "gr_dienstbarkeit", bezeichnung: "Entschädigung Leitungsrechte entlang der Trasse", menge: r2(tl), einheit: "m", ep: PK.dienstbarkeitProM, quelle: "markt.js PROJEKTKOSTEN.dienstbarkeitProM" });
   161	  if (hatFlaeche(p.typ) || tl > 0) {
   162	    const vorgaenge = p.flaechenAnzahl + (tl > 0 ? Math.ceil(tl / 250) : 0);
   163	    add({ schluessel: "gr_notar", gewerk: "gr_notar", bezeichnung: "Notar/Grundbuch (Dienstbarkeiten, Vormerkungen)", menge: vorgaenge, einheit: "Vorgang", ep: PK.notarJeFlaeche, quelle: "markt.js PROJEKTKOSTEN.notarJeFlaeche" });
   164	  }
   165	
   166	  // ---- Technik (EPC)
   167	  const epc = spanne.mittel;
   168	  if (pv) {
   169	    for (const k of ["module", "wechselrichter", "unterkonstruktion", "dc", "ac", "bos"]) {
   170	      if (!anteile[k]) continue;
   171	      add({ schluessel: k, gewerk: k, bezeichnung: GEWERKE[k].label, menge: kwp, einheit: "kWp", ep: r2(epc * anteile[k]), quelle: `markt.js EPC ${epc} €/kWp × Anteil ${Math.round(anteile[k] * 100)} %` });
   172	    }
   173	    if (anteile.zaun) {
   174	      if (p.zaunM > 0 || p.flaecheHa > 0) {
   175	        const m = p.zaunM > 0 ? p.zaunM : Math.round(4 * Math.sqrt(p.flaecheHa * 10000) * 1.15);
   176	        add({ schluessel: "zaun", gewerk: "zaun", bezeichnung: `Zaun inkl. Tore${p.zaunM > 0 ? "" : " (Länge geschätzt aus Fläche)"}`, menge: m, einheit: "m", ep: PK.zaunProM, quelle: p.zaunM > 0 ? "Zaunlänge aus Parametern/DXF" : "Umfang ≈ 4 × √Fläche × 1,15" });
   177	        if (p.flaecheHa > 0) add({ schluessel: "kamera", gewerk: "zaun", bezeichnung: "Kamera / Einbruchmeldeanlage", menge: r2(p.flaecheHa), einheit: "ha", ep: PK.kameraProHa, quelle: "markt.js PROJEKTKOSTEN.kameraProHa" });
   178	      } else add({ schluessel: "zaun", gewerk: "zaun", bezeichnung: GEWERKE.zaun.label, menge: kwp, einheit: "kWp", ep: r2(epc * anteile.zaun), quelle: "EPC-Anteil (ohne Zaunlänge)" });
   179	    }
   180	    if (anteile.wege) {
   181	      if (p.flaecheHa > 0 && hatFlaeche(p.typ)) add({ schluessel: "wege", gewerk: "wege", bezeichnung: GEWERKE.wege.label, menge: r2(p.flaecheHa), einheit: "ha", ep: PK.wegeProHa, quelle: "markt.js PROJEKTKOSTEN.wegeProHa" });
   182	      else add({ schluessel: "wege", gewerk: "wege", bezeichnung: p.typ === "carport" ? "Belag, Entwässerung, Erdarbeiten" : GEWERKE.wege.label, menge: kwp, einheit: "kWp", ep: r2(epc * anteile.wege), quelle: "EPC-Anteil" });
   183	    }
   184	    if (hatFlaeche(p.typ) && p.flaecheHa > 0) add({ schluessel: "ausgleich", gewerk: "ausgleich", bezeichnung: GEWERKE.ausgleich.label, menge: r2(p.flaecheHa), einheit: "ha", ep: PK.ausgleichProHa, quelle: "markt.js PROJEKTKOSTEN.ausgleichProHa" });
   185	  }
   186	  if (p.mwh > 0) add({ schluessel: "speicher", gewerk: "speicher", bezeichnung: "Batteriespeicher schlüsselfertig", menge: p.mwh * 1000, einheit: "kWh", ep: SPEICHER.capexProKwh.mittel, quelle: "markt.js SPEICHER.capexProKwh (Mitte)" });
   187	  if (pv && anteile.bauleitung) pos.push({ schluessel: "bauleitung", gewerk: "bauleitung", bezeichnung: GEWERKE.bauleitung.label, art: "prozent", menge: null, einheit: "%", ep: null, satz: Math.round((anteile.bauleitung / (1 - anteile.bauleitung)) * 1e6) / 1e6, basis: "technik", quelle: "EPC-Anteil als Aufschlag auf die Gewerke" });
   188	
   189	  // ---- Netz
   190	  const speicherMw = p.mwh > 0 ? p.mwh / 2 : 0;
   191	  const anschlussMw = p.typ === "speicher" ? Math.max(p.mwp, speicherMw) : p.mwp;
   192	  const ebene = ebeneFuer(anschlussMw);
   193	  if (p.typ === "speicher") {
   194	    if (anschlussMw > 0) add({ schluessel: "station", gewerk: "station", bezeichnung: "Umspannung / Schaltanlage Speicher", menge: anschlussMw * 1000, einheit: "kW", ep: ebene === "hs" ? SPEICHER.netzanschlussProKw.hs : SPEICHER.netzanschlussProKw.ms, quelle: "markt.js SPEICHER.netzanschlussProKw" });
   195	  } else if (p.mwp > 0) add({ schluessel: "station", gewerk: "station", bezeichnung: GEWERKE.station.label, menge: r2(p.mwp), einheit: "MWp", ep: FREIFLAECHE.netzanschlussFixProMwp, quelle: "markt.js FREIFLAECHE.netzanschlussFixProMwp" });
   196	  if (tl > 0) {
   197	    if (ebene === "hs") add({ schluessel: "trasse:hs", gewerk: "trasse", bezeichnung: "Kabeltrasse 110 kV", menge: r2(tl), einheit: "m", ep: FREIFLAECHE.netz.hsKabelProKm.mittel / 1000, quelle: "markt.js FREIFLAECHE.netz.hsKabelProKm (Mitte)" });
   198	    else {
   199	      const sys = kabelSysteme(anschlussMw);
   200	      const faktor = 1 + (sys - 1) * FREIFLAECHE.netz.zusatzSystemFaktor;
   201	      for (const a of p.trasse.abschnitte) {
   202	        add({ schluessel: `trasse:${a.art}`, gewerk: "trasse", bezeichnung: `MS-Trasse ${VERLEGEARTEN[a.art]}${sys > 1 ? ` (${sys} Systeme)` : ""}`, menge: r2(a.laengeM), einheit: "m", ep: r2(PK.trasseProM[a.art] * faktor), quelle: `markt.js PROJEKTKOSTEN.trasseProM${sys > 1 ? " × Systemfaktor" : ""}` });
   203	      }
   204	    }
   205	  }
   206	  for (const [k, n] of Object.entries(p.trasse.querungen)) {
   207	    if (n > 0) add({ schluessel: `querung:${k}`, gewerk: "querung", bezeichnung: `Querung ${QUERUNGEN[k]}`, menge: n, einheit: "Stk", ep: PK.querung[k], quelle: "markt.js PROJEKTKOSTEN.querung" });
   208	  }
   209	  if (anschlussMw > 0) {
   210	    if (ebene === "msLeitung") add({ schluessel: "netzanschluss", gewerk: "netzanschluss", bezeichnung: "Einschleifen in die MS-Leitung (Schaltanlage, Schutz)", menge: 1, einheit: "psch", ep: FREIFLAECHE.netz.einschleifen, quelle: "markt.js FREIFLAECHE.netz.einschleifen" });
   211	    else if (ebene === "uwMs") {
   212	      const sys = kabelSysteme(anschlussMw);
   213	      add({ schluessel: "netzanschluss", gewerk: "netzanschluss", bezeichnung: "Schaltfeld(er) im Umspannwerk", menge: sys, einheit: "Feld", ep: FREIFLAECHE.netz.uwFeld.mittel, quelle: "markt.js FREIFLAECHE.netz.uwFeld (Mitte)" });
   214	    } else add({ schluessel: "netzanschluss", gewerk: "netzanschluss", bezeichnung: "Eigenes Umspannwerk 110/20 kV", menge: 1, einheit: "psch", ep: FREIFLAECHE.netz.kundenUwFix + FREIFLAECHE.netz.kundenUwProMva * anschlussMw, quelle: "markt.js FREIFLAECHE.netz.kundenUw*" });
   215	  }
   216	  if (speicherMw > 0 && p.speicherNetzbezug) {
   217	    const satz = ebene === "hs" ? SPEICHER.bkzProKw.hs : SPEICHER.bkzProKw.ms;
   218	    add({ schluessel: "bkz", gewerk: "bkz", bezeichnung: p.land === "AT" ? "Netzbereitstellungsentgelt Speicher (Näherung wie BKZ)" : "Baukostenzuschuss Netzbezug Speicher", menge: speicherMw * 1000, einheit: "kW", ep: satz, quelle: "markt.js SPEICHER.bkzProKw" });
   219	  }
   220	
   221	  // ---- Nebenkosten, Reserve, Finanzierung, Sicherheiten
   222	  pos.push({ schluessel: "versicherung", gewerk: "versicherung", bezeichnung: GEWERKE.versicherung.label, art: "prozent", menge: null, einheit: "%", ep: null, satz: PK.versicherungBau, basis: "bau", quelle: "markt.js PROJEKTKOSTEN.versicherungBau" });

   230	/**
   231	 * Gespeicherte Positionen mit dem aktuellen Vorschlag zusammenführen: manuelle Positionen bleiben, nicht manuelle
   232	 * folgen dem Vorschlag (Ist und Notiz bleiben), freie Positionen („frei:…“) bleiben immer.
   233	 */
   234	export function zusammenfuehren(gespeichert = [], vorschlaege = []) {
   235	  const alt = new Map(gespeichert.map((x) => [x.schluessel, x]));
   236	  const out = [];
   237	  for (const v of vorschlaege) {
   238	    const a = alt.get(v.schluessel);
   239	    const vor = { menge: v.menge, ep: v.ep, satz: v.satz };
   240	    if (a?.manuell) out.push({ ...v, ...pick(a), manuell: true, vorschlag: vor });
   241	    else out.push({ ...v, id: a?.id, istEur: a?.istEur ?? null, notiz: a?.notiz || "", angebot: a?.angebot || "", manuell: false, vorschlag: vor });
   242	    alt.delete(v.schluessel);
   243	  }
   244	  for (const a of alt.values()) {
   245	    if (String(a.schluessel).startsWith("frei:") || a.manuell || Number(a.istEur) > 0) out.push({ ...a, vorschlag: a.schluessel.startsWith("frei:") ? null : a.vorschlag || null, entfallen: !a.schluessel.startsWith("frei:") });
   246	  }
   247	  return out;
   248	}
   249	const pick = (a) => ({ id: a.id, bezeichnung: a.bezeichnung, menge: a.menge, ep: a.ep, satz: a.satz, einheit: a.einheit, istEur: a.istEur ?? null, notiz: a.notiz || "", angebot: a.angebot || "" });

   251	/**
   252	 * Berechnung: Beträge je Position, Summen je Gruppe/Gewerk, Kennzahlen, Benchmark-Ampeln.
   253	 * Ergebnis { positionen[+betrag, eurProKwp, abweichung, ampel], gruppen{}, gewerke{}, summen{…}, benchmark{…} }
   254	 */
   255	export function berechnen(positionen, parameter) {
   256	  const p = parameterVoll(parameter);
   257	  const kwp = p.mwp * 1000;
   258	  const liste = positionen.map((x) => ({ ...x, gruppe: GEWERKE[x.gewerk]?.gruppe || "neben" }));
   259	  const betragMenge = (x) => r2((Number(x.menge) || 0) * (Number(x.ep) || 0));
   260	  const summeWo = (f) => liste.filter(f).reduce((s, x) => s + (x.betrag || 0), 0);
   261	  // 1. Mengenpositionen
   262	  for (const x of liste) if (x.art === "menge" || !x.art) x.betrag = betragMenge(x);
   263	  // 2. Aufschläge auf Technik (Bauleitung/EPC-Marge)
   264	  const technikBasis = summeWo((x) => x.gruppe === "technik" && x.art !== "prozent" && x.gewerk !== "speicher");
   265	  for (const x of liste) if (x.art === "prozent" && x.basis === "technik") x.betrag = r2(technikBasis * (Number(x.satz) || 0));
   266	  // 3. Bezug Bau (Technik + Netz): Versicherung, Unvorhergesehenes
   267	  const bau = summeWo((x) => x.gruppe === "technik" || x.gruppe === "netz");
   268	  for (const x of liste) if (x.art === "prozent" && x.basis === "bau") x.betrag = r2(bau * (Number(x.satz) || 0));
   269	  // 4. Bauzeitzinsen: FK-Quote × (Bau + Neben + Reserve) × Zins × Monate/12 × 0,5
   270	  const bauGesamt = summeWo((x) => ["technik", "netz", "neben", "reserve"].includes(x.gruppe) && x.art !== "zins");
   271	  for (const x of liste) if (x.art === "zins") x.betrag = r2(p.fkQuote * bauGesamt * (Number(x.satz) || 0) * (p.bauzeitMonate / 12) * 0.5);
   272	  // 5. Bezug Investition (alles außer Finanzierung/Sicherheit): Gebühren
   273	  const invest = summeWo((x) => !["finanzierung", "sicherheit"].includes(x.gruppe));
   274	  for (const x of liste) if (x.art === "prozent" && x.basis === "invest") x.betrag = r2(invest * (Number(x.satz) || 0));
   275	
   276	  const gruppen = {};
   277	  const gewerke = {};
   278	  for (const g of Object.keys(GRUPPEN)) gruppen[g] = { soll: 0, ist: 0, eurProKwp: null };
   279	  for (const x of liste) {
   280	    gruppen[x.gruppe].soll += x.betrag || 0;
   281	    gruppen[x.gruppe].ist += Number(x.istEur) || 0;
   282	    const gw = (gewerke[x.gewerk] ||= { soll: 0, ist: 0, vorschlag: 0, eurProKwp: null });
   283	    gw.soll += x.betrag || 0;
   284	    gw.ist += Number(x.istEur) || 0;
   285	    // Vorschlagswert der Position (für Abweichung)
   286	    if (x.vorschlag && (x.art === "menge" || !x.art)) x.vorschlagBetrag = r2((Number(x.vorschlag.menge) || 0) * (Number(x.vorschlag.ep) || 0));
   287	    else if (x.vorschlag && x.satz && x.vorschlag.satz) x.vorschlagBetrag = r2((x.betrag || 0) * (x.vorschlag.satz / x.satz));
   288	    else x.vorschlagBetrag = null;
   289	    x.abweichung = x.vorschlagBetrag ? (x.betrag - x.vorschlagBetrag) / x.vorschlagBetrag : null;
   290	    x.ampel = x.abweichung === null || !x.manuell ? null : Math.abs(x.abweichung) <= 0.1 ? "gruen" : Math.abs(x.abweichung) <= 0.25 ? "gelb" : "rot";
   291	    x.eurProKwp = kwp > 0 ? r2(x.betrag / kwp) : null;
   292	    gw.vorschlag += x.vorschlagBetrag ?? x.betrag ?? 0;
   293	  }
   294	  for (const g of Object.values(gruppen)) {
   295	    g.soll = r2(g.soll);
   296	    g.ist = r2(g.ist);
   297	    g.eurProKwp = kwp > 0 ? r2(g.soll / kwp) : null;
   298	  }
   299	  for (const g of Object.values(gewerke)) {
   300	    g.soll = r2(g.soll);
   301	    g.ist = r2(g.ist);
   302	    g.vorschlag = r2(g.vorschlag);
   303	    g.eurProKwp = kwp > 0 ? r2(g.soll / kwp) : null;
   304	  }
   305	  const gesamt = r2(Object.entries(gruppen).filter(([k]) => k !== "sicherheit").reduce((s, [, g]) => s + g.soll, 0));
   306	  const capex = r2(gesamt - gruppen.finanzierung.soll);
   307	  const finanzierungsbedarf = r2(gesamt + gruppen.sicherheit.soll);
   308	  const epcOhneSpeicher = r2(gruppen.technik.soll - (gewerke.speicher?.soll || 0));
   309	  const summen = {
   310	    gesamt, capex, finanzierungsbedarf, bau: r2(bau), invest: r2(invest),
   311	    ist: r2(liste.reduce((s, x) => s + (Number(x.istEur) || 0), 0)),
   312	    eurProKwp: kwp > 0 ? r2(gesamt / kwp) : null,
   313	    capexProKwp: kwp > 0 ? r2(capex / kwp) : null,
   314	    epcProKwp: kwp > 0 && p.typ !== "speicher" ? r2(epcOhneSpeicher / kwp) : null,
   315	    eurProKwh: p.mwh > 0 ? r2((gewerke.speicher?.soll || 0) / (p.mwh * 1000)) : null,
   316	    kwp,
   317	  };
   318	  return { positionen: liste, gruppen, gewerke, summen, benchmark: benchmark(summen, gruppen, p), parameter: p };
   319	}

   321	/** Benchmark gegen markt.js: EPC-Spanne des Typs, Entwicklungskosten, Netzanteil an der Anlage */
   322	export function benchmark(summen, gruppen, p) {
   323	  const out = [];
   324	  const kwp = summen.kwp;
   325	  if (kwp > 0 && p.typ !== "speicher") {
   326	    const sp = capexSpanne(p.typ);
   327	    const w = summen.epcProKwp;
   328	    out.push({ id: "epc", label: "Technik/EPC €/kWp", wert: w, min: sp.min, max: sp.max, mittel: sp.mittel, ampel: w <= sp.max && w >= sp.min ? "gruen" : w <= sp.max * 1.1 && w >= sp.min * 0.9 ? "gelb" : "rot", text: `Spanne ${sp.min}–${sp.max} €/kWp (markt.js)` });
   329	    const e = gruppen.entwicklung.eurProKwp;
   330	    const es = FREIFLAECHE.entwicklungProKwp;
   331	    out.push({ id: "entwicklung", label: "Entwicklung €/kWp", wert: e, min: es.min, max: es.max, mittel: (es.min + es.max) / 2, ampel: e <= es.max && e >= es.min ? "gruen" : e <= es.max * 1.25 && e >= es.min * 0.5 ? "gelb" : "rot", text: `Spanne ${es.min}–${es.max} €/kWp bis Baureife` });
   332	    const anteil = summen.epcProKwp > 0 ? gruppen.netz.soll / (summen.epcProKwp * kwp) : 0;
   333	    out.push({ id: "netz", label: "Netzanschluss in % der Anlage", wert: r2(anteil * 10000) / 100, einheit: "%", ampel: anteil <= 0.15 ? "gruen" : anteil <= 0.3 ? "gelb" : "rot", text: "bis 15 % unkritisch, über 30 % drückt die Rendite spürbar (Netzanschluss-Check)" });
   334	  }
   335	  if (p.mwh > 0) {
   336	    const w = summen.eurProKwh;
   337	    const sp = SPEICHER.capexProKwh;
   338	    out.push({ id: "speicher", label: "Speicher €/kWh", wert: w, min: sp.min, max: sp.max, mittel: sp.mittel, ampel: w <= sp.max && w >= sp.min ? "gruen" : w <= sp.max * 1.1 ? "gelb" : "rot", text: `Spanne ${sp.min}–${sp.max} €/kWh` });
   339	  }
   340	  const schlecht = out.some((b) => b.ampel === "rot") ? "rot" : out.some((b) => b.ampel === "gelb") ? "gelb" : "gruen";
   341	  return { eintraege: out, ampel: out.length ? schlecht : null };
   342	}

   344	/**
   345	 * Soll/Ist je Kostenart des Finanzmoduls (nur Entwicklungs-/Grundstückspositionen haben eine Kostenart). istFinanzen =
   346	 * { jeArt: { planung: 1234, … } } aus finanzen.entwicklungIst bzw. ERPNext; sonst manuelle Ist-Werte der Positionen.
   347	 */
   348	export function sollIst(ergebnis, istFinanzen = null) {
   349	  const zeilen = {};
   350	  for (const x of ergebnis.positionen) {
   351	    const art = GEWERKE[x.gewerk]?.kostenart || (x.gruppe === "entwicklung" ? "sonstiges" : null);
   352	    const key = art || `gruppe:${x.gruppe}`;
   353	    const z = (zeilen[key] ||= { schluessel: key, kostenart: art, gruppe: x.gruppe, soll: 0, istManuell: 0, istFinanzen: null });
   354	    z.soll += x.betrag || 0;
   355	    z.istManuell += Number(x.istEur) || 0;
   356	  }
   357	  const jeArt = istFinanzen?.jeArt || null;
   358	  if (jeArt) {
   359	    for (const [art, betrag] of Object.entries(jeArt)) {
   360	      const z = (zeilen[art] ||= { schluessel: art, kostenart: art, gruppe: "entwicklung", soll: 0, istManuell: 0, istFinanzen: null });
   361	      z.istFinanzen = r2((z.istFinanzen || 0) + (Number(betrag) || 0));
   362	    }
   363	  }
   364	  return Object.values(zeilen).map((z) => {
   365	    const ist = z.istFinanzen !== null ? z.istFinanzen : z.istManuell;
   366	    return { ...z, soll: r2(z.soll), istManuell: r2(z.istManuell), ist: r2(ist), quelle: z.istFinanzen !== null ? istFinanzen?.quelle || "finanzen" : "manuell", abweichung: z.soll ? (ist - z.soll) / z.soll : null };
   367	  }).sort((a, b) => (GRUPPEN[a.gruppe]?.reihenfolge || 9) - (GRUPPEN[b.gruppe]?.reihenfolge || 9));
   368	}
