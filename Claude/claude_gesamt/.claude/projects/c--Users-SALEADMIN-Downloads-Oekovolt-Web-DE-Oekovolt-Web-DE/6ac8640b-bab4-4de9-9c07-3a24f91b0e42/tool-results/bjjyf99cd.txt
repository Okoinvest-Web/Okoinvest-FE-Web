    89	/** Wartezeit in Sekunden nach `fehl` Fehlversuchen (0 = keine Sperre) */
    90	export function sperrDauer(fehl) {
    91	  if (fehl < SPERRE.schwelle) return 0;
    92	  return Math.min(SPERRE.maxSekunden, SPERRE.basisSekunden * 2 ** (fehl - SPERRE.schwelle));
    93	}

    95	const sperrText = (sek) => {
    96	  const min = Math.ceil(sek / 60);
    97	  return `Zu viele fehlgeschlagene Anmeldeversuche. Das Konto ist vorübergehend gesperrt – bitte versuchen Sie es in ${sek < 60 ? `${sek} Sekunden` : `${min} Minute${min === 1 ? "" : "n"}`} erneut oder melden Sie sich per E-Mail-Link an.`;
    98	};

   100	/** Wirft 429 (code „gesperrt“), solange die E-Mail-Adresse gesperrt ist */
   101	export function sperrePruefen(db, email) {
   102	  const s = db.anmeldeSperren?.[email];
   103	  if (!s?.bis) return;
   104	  const rest = Math.ceil((new Date(s.bis).getTime() - Date.now()) / 1000);
   105	  if (rest > 0) throw new PlattformFehler(429, sperrText(rest), "gesperrt");
   106	}

   108	/** Fehlversuch zählen (Passwort oder 2FA-Code) – ab der Schwelle wächst die Sperre exponentiell */
   109	export function fehlversuchBuchen(db, { email, nutzer, ip, userAgent, land, grund }) {
   110	  sammlungen(db);
   111	  if (email) {
   112	    const s = db.anmeldeSperren[email] || { fehl: 0, bis: null };
   113	    s.fehl += 1;
   114	    s.zuletzt = jetzt();
   115	    const dauer = sperrDauer(s.fehl);
   116	    s.bis = dauer ? new Date(Date.now() + dauer * 1000).toISOString() : null;
   117	    db.anmeldeSperren[email] = s;
   118	    const schluessel = Object.keys(db.anmeldeSperren);
   119	    if (schluessel.length > 2000) {
   120	      schluessel.sort((a, b) => String(db.anmeldeSperren[a].zuletzt).localeCompare(String(db.anmeldeSperren[b].zuletzt)));
   121	      for (const k of schluessel.slice(0, schluessel.length - 2000)) delete db.anmeldeSperren[k];
   122	    }
   123	    if (nutzer && dauer && s.fehl === SPERRE.schwelle) {
   124	      sicherheitMelden(db, nutzer, {
   125	        titel: "Konto vorübergehend gesperrt",
   126	        text: `${SPERRE.schwelle} fehlgeschlagene Anmeldeversuche – die Anmeldung ist kurz gesperrt.`,
   127	        betreff: "Mehrere fehlgeschlagene Anmeldeversuche – ÖkoInvest",
   128	        mailText: `Für Ihr Konto gab es ${SPERRE.schwelle} fehlgeschlagene Anmeldeversuche. Die Anmeldung ist deshalb vorübergehend gesperrt; jede weitere Fehleingabe verlängert die Sperre. Über den E-Mail-Link auf der Anmeldeseite kommen Sie jederzeit in Ihr Konto.`,
   129	      });
   130	    }
   131	  }
   132	  if (nutzer) historie(db, { nutzer, email, ok: false, methode: grund === "2fa" ? "zwei-faktor" : "passwort", grund, ip, userAgent, land });
   133	}

   137	/** Nutzer um Sitzung, Organisation, 2FA-Status und Komfort ergänzen */
   138	export function nutzerAnreichern(db, u, s = null) {
   139	  sammlungen(db);
   140	  const orgs = organisationenVon(db, u.id);
   141	  const org = s?.organisationId ? orgs.find((o) => o.id === s.organisationId) || null : null;
   142	  const z = zf(db, u.id);
   143	  const pflicht = pflichtFuer(db, u);
   144	  const out = {
   145	    ...u,
   146	    sitzung: s ? { id: s.id, organisationId: org?.id || null, zweiFaktor: !!s.zweiFaktor } : null,
   147	    organisation: org,
   148	    organisationen: orgs,
   149	    zweiFaktorAktiv: !!z?.aktiv,
   150	    zweiFaktorPflicht: pflicht,
   151	    zweiFaktorPflichtOffen: pflicht && !z?.aktiv,
   152	    komfort: komfortBereinigen(u.komfort || {}),
   153	  };
   154	  if (org) {
   155	    // Im Organisationskontext gilt das Team-Suchprofil
   156	    out.suchprofilPersoenlich = u.suchprofil || null;
   157	    out.suchprofil = db.organisationen.find((o) => o.id === org.id)?.suchprofil || null;
   158	  }
   159	  return out;
   160	}

   162	function sitzungGueltig(s, u) {
   163	  return !!s && s.nutzerId === u.id && !s.widerrufenAm && new Date(s.ablauf) > new Date() && (s.version || 0) === (u.sitzungVersion || 0);

   166	/**
   167	 * Sitzung prüfen und Nutzer anreichern. Ohne Sitzungskennung (Token aus der Zeit vor den Sitzungslisten)
   168	 * gilt nur die Sitzungsversion. Letzte Aktivität wird höchstens einmal pro Minute geschrieben.
   169	 */
   170	export async function sitzungNutzer(db, u, { sitzungId } = {}) {
   171	  sammlungen(db);
   172	  if (!sitzungId) return nutzerAnreichern(db, u, null);
   173	  const s = db.sitzungen.find((x) => x.id === sitzungId);
   174	  if (!sitzungGueltig(s, u)) return null;
   175	  if (Date.now() - new Date(s.letzteAktivitaet).getTime() > AKTIVITAET_MS) {
   176	    // nicht abwarten: die Anfrage soll nie auf die Schreibsperre warten
   177	    aendern((d) => {
   178	      const x = (d.sitzungen || []).find((y) => y.id === sitzungId);
   179	      if (x) x.letzteAktivitaet = jetzt();
   180	    }).catch(() => {});
   181	  }
   182	  return nutzerAnreichern(db, u, s);
   183	}

   190	/**
   191	 * Neue Sitzung nach vollständiger Anmeldung (Passwort/E-Mail-Link + ggf. 2FA). Setzt die Sperre zurück,
   192	 * schreibt die Historie, erkennt neue Geräte (Benachrichtigung + E-Mail) und merkt das Gerät optional 30 Tage.
   193	 * Liefert { sitzung: {id}, geraetToken (nur wenn neu erzeugt), neuesGeraet }.
   194	 */
   195	export async function sitzungAnlegen(n, { ip = "", userAgent = "", land = "", methode = "passwort", zweiFaktor = false, geraetToken = "", geraetMerken = false, ersetzt = null } = {}) {
   196	  if (!n) throw nichtAngemeldet();
   197	  return aendern((db) => {
   198	    sammlungen(db);
   199	    orgSeed(db);
   200	    const u = nutzerById(db, n.id);
   201	    if (!u || !u.aktiv) throw nichtAngemeldet();
   202	    delete db.anmeldeSperren[u.email];
   203	    const g = geraetAusUserAgent(userAgent);
   204	    const grob = ipGrob(ip);
   205	    const ort = ortGrob(ip, land);
   206	    const tokenOk = /^[A-Za-z0-9_-]{30,100}$/.test(String(geraetToken || ""));
   207	    let geraet = tokenOk ? db.geraete.find((x) => x.nutzerId === u.id && x.hash === sha256(geraetToken)) : null;
   208	    const bekannt = !!geraet || db.anmeldungen.some((a) => a.nutzerId === u.id && a.ok && a.geraet === g.label && a.ipGrob === grob);
   209	    let neuesToken = null;
   210	    if (!geraet) {
   211	      const token = tokenOk ? geraetToken : zufallsToken();
   212	      if (!tokenOk) neuesToken = token;
   213	      geraet = { id: neueId("G"), nutzerId: u.id, hash: sha256(token), label: g.label, mobil: g.mobil, ersteAnmeldung: jetzt(), letzteAnmeldung: jetzt(), vertrautBis: null, zweiFaktorVersion: null };
   214	      db.geraete.push(geraet);
   215	    }
   216	    geraet.letzteAnmeldung = jetzt();
   217	    geraet.label = g.label;
   218	    const z = zf(db, u.id);
   219	    if (geraetMerken && zweiFaktor && z?.aktiv) {
   220	      geraet.vertrautBis = new Date(Date.now() + GERAET_MERKEN_TAGE * 86400e3).toISOString();
   221	      geraet.zweiFaktorVersion = z.version;
   222	    }
   223	
   224	    // Sitzungsrotation: eine ersetzte Sitzung (z. B. nach 2FA-Einrichtung) wird beendet, Kontext bleibt
   225	    const alt = ersetzt ? db.sitzungen.find((x) => x.id === ersetzt && x.nutzerId === u.id) : null;
   226	    if (alt && !alt.widerrufenAm) {
   227	      alt.widerrufenAm = jetzt();
   228	      alt.widerrufenGrund = "rotation";
   229	    }
   230	    const s = {
   231	      id: neueId("S", 10), nutzerId: u.id, version: u.sitzungVersion || 0, erstelltAm: jetzt(),
   232	      ablauf: new Date(Date.now() + SITZUNG_DAUER_S * 1000).toISOString(), letzteAktivitaet: jetzt(),
   233	      ipGrob: grob, ort, userAgent: v.text(userAgent, 300), geraet: g.label, mobil: g.mobil, geraetId: geraet.id,
   234	      methode, zweiFaktor: !!zweiFaktor, organisationId: alt?.organisationId || null, widerrufenAm: null,
   235	    };
   236	    db.sitzungen.push(s);
   237	    // Aufräumen: abgelaufene/widerrufene Einträge nach 30 Tagen, höchstens 30 aktive Sitzungen je Konto
   238	    const grenze = Date.now() - 30 * 86400e3;
   239	    db.sitzungen = db.sitzungen.filter((x) => !((x.widerrufenAm && new Date(x.widerrufenAm).getTime() < grenze) || new Date(x.ablauf).getTime() < grenze));
   240	    const aktive = db.sitzungen.filter((x) => sitzungGueltig(x, u)).sort((a, b) => a.letzteAktivitaet.localeCompare(b.letzteAktivitaet));
   241	    for (const x of aktive.slice(0, Math.max(0, aktive.length - 30))) {
   242	      x.widerrufenAm = jetzt();
   243	      x.widerrufenGrund = "grenze";
   244	    }
   245	
   246	    u.letzterLogin = jetzt();
   247	    historie(db, { nutzer: u, ok: true, methode: `${methode}${zweiFaktor ? " + 2FA" : ""}`, ip, userAgent, land });
   248	    if (!bekannt && !["registrierung"].includes(methode)) {
   249	      sicherheitMelden(db, u, {
   250	        titel: "Anmeldung von einem neuen Gerät",
   251	        text: `${g.label} · ${ort || "Ort unbekannt"}`,
   252	        betreff: "Neue Anmeldung bei Ihrem Konto – ÖkoInvest",
   253	        mailText: `Ihr Konto im Portal von ÖkoInvest wurde soeben von einem neuen Gerät verwendet: ${g.label}, ${ort || "Ort unbekannt"}. Wenn Sie das waren, ist nichts zu tun.`,
   254	      });
   255	    }
   256	    if (methode !== "passwort" || zweiFaktor) internSicherheit.protokoll(db, u, "login", "nutzer", u.id, `Anmeldung (${methode}${zweiFaktor ? ", 2FA" : ""}) · ${g.label}`, ip);
   257	    return { sitzung: { id: s.id }, geraetToken: neuesToken, neuesGeraet: !bekannt };
   258	  });
   259	}

   261	/** Aktive Sitzungen des Kontos (aktuelle markiert) */
   262	export async function sitzungen(n) {
   263	  if (!n) throw nichtAngemeldet();
   264	  const db = sammlungen(await lesen());
   265	  const u = nutzerById(db, n.id);
   266	  return db.sitzungen
   267	    .filter((s) => sitzungGueltig(s, u))
   268	    .sort((a, b) => b.letzteAktivitaet.localeCompare(a.letzteAktivitaet))
   269	    .map((s) => ({
   270	      id: s.id, geraet: s.geraet, mobil: !!s.mobil, ort: s.ort, erstelltAm: s.erstelltAm, letzteAktivitaet: s.letzteAktivitaet, methode: s.methode,
   271	      zweiFaktor: !!s.zweiFaktor, aktuell: s.id === n.sitzung?.id, organisation: s.organisationId ? db.organisationen.find((o) => o.id === s.organisationId)?.name || null : null,
   272	    }));
   273	}

   275	/**
   276	 * Sitzung(en) beenden: { id } eine eigene Sitzung (auch die aktuelle = Abmelden) oder { alleAnderen: true }.
   277	 * „Alle anderen“ erhöht zusätzlich die Sitzungsversion (beendet auch ältere Token) – die aktuelle Sitzung
   278	 * übernimmt die neue Version; die Website setzt dafür ein neues Cookie.
   279	 */
   280	export async function sitzungBeenden(n, { id, alleAnderen = false } = {}) {
   281	  if (!n) throw nichtAngemeldet();
   282	  return aendern((db) => {
   283	    sammlungen(db);
   284	    const u = nutzerById(db, n.id);
   285	    let beendet = 0;
   286	    if (alleAnderen) {
   287	      for (const s of db.sitzungen) {
   288	        if (s.nutzerId !== u.id || s.id === n.sitzung?.id || s.widerrufenAm) continue;
   289	        s.widerrufenAm = jetzt();
   290	        s.widerrufenGrund = "abgemeldet";
   291	        beendet++;
   292	      }
   293	      u.sitzungVersion = (u.sitzungVersion || 0) + 1;
   294	      const aktuell = db.sitzungen.find((s) => s.id === n.sitzung?.id);
   295	      if (aktuell) aktuell.version = u.sitzungVersion;
   296	      internSicherheit.protokoll(db, u, "sitzung", "nutzer", u.id, `Alle anderen Sitzungen beendet (${beendet})`);
   297	      return { beendet, sitzungVersion: u.sitzungVersion, nutzer: internSicherheit.ohneGeheimnis(u) };
   298	    }
   299	    const s = db.sitzungen.find((x) => x.id === v.kennung(id));
   300	    if (!s || s.nutzerId !== u.id) throw nichtGefunden("Sitzung nicht gefunden.");
   301	    if (!s.widerrufenAm) {
   302	      s.widerrufenAm = jetzt();
   303	      s.widerrufenGrund = s.id === n.sitzung?.id ? "abmeldung" : "abgemeldet";
   304	      beendet = 1;
   305	      if (s.id !== n.sitzung?.id) internSicherheit.protokoll(db, u, "sitzung", "nutzer", u.id, `Sitzung beendet: ${s.geraet}`);
   306	    }
   307	    return { beendet, sitzungVersion: u.sitzungVersion || 0 };
   308	  });
   309	}

   322	export async function geraete(n, { geraetToken = "" } = {}) {
   323	  if (!n) throw nichtAngemeldet();
   324	  const db = sammlungen(await lesen());
   325	  const z = zf(db, n.id);
   326	  const h = geraetToken ? sha256(String(geraetToken)) : "";
   327	  return db.geraete
   328	    .filter((g) => g.nutzerId === n.id)
   329	    .sort((a, b) => b.letzteAnmeldung.localeCompare(a.letzteAnmeldung))
   330	    .map((g) => ({
   331	      id: g.id, label: g.label, mobil: !!g.mobil, ersteAnmeldung: g.ersteAnmeldung, letzteAnmeldung: g.letzteAnmeldung,
   332	      vertraut: !!(z?.aktiv && g.vertrautBis && new Date(g.vertrautBis) > new Date() && g.zweiFaktorVersion === z.version),
   333	      vertrautBis: g.vertrautBis, aktuell: !!h && g.hash === h,
   334	    }));
   335	}

   337	/** Gerät vergessen: Eintrag löschen (2FA wird dort wieder abgefragt, neue Anmeldung gilt als neues Gerät) */
   338	export async function geraetVergessen(n, id) {
   339	  if (!n) throw nichtAngemeldet();

   435	/** Code prüfen (TOTP mit Replay-Schutz oder Wiederherstellungscode); verbraucht Wiederherstellungscodes */
   436	function codeEinloesen(db, u, code, { wiederherstellung = true } = {}) {
   437	  const z = zf(db, u.id);
   438	  if (!z?.aktiv) return null;
   439	  if (wiederherstellung && istWiederherstellungscode(code)) {
   440	    const h = codeHash(code);
   441	    const idx = (z.codes || []).indexOf(h);
   442	    if (idx < 0) return null;
   443	    z.codes.splice(idx, 1);
   444	    return { methode: "wiederherstellung", codesUebrig: z.codes.length };
   445	  }
   446	  const schritt = totpPruefen(entschluesseln(z.geheimnis), code, { letzterSchritt: z.letzterSchritt ?? -1 });
   447	  if (schritt === null) return null;
   448	  z.letzterSchritt = schritt;
   449	  return { methode: "totp", codesUebrig: (z.codes || []).length };
   450	}

   700	      zweiFaktor: { aktiv: !!z?.aktiv, aktiviertAm: z?.aktiviertAm || null, wiederherstellungscodesUebrig: z?.aktiv ? (z.codes || []).length : 0 },
   701	      sitzungen: db.sitzungen.filter((s) => s.nutzerId === u.id).map((s) => ohne(s, ["nutzerId", "version", "geraetId"])),
   702	      geraete: db.geraete.filter((g) => g.nutzerId === u.id).map((g) => ohne(g, ["hash", "nutzerId", "zweiFaktorVersion"])),
   703	      anmeldungen: db.anmeldungen.filter((a) => a.nutzerId === u.id).map((a) => ohne(a, ["nutzerId"])),
   704	    },
   705	    organisationen: db.mitgliedschaften
   706	      .filter((m) => m.nutzerId === u.id)
   707	      .map((m) => {
   708	        const o = db.organisationen.find((x) => x.id === m.organisationId);
   709	        return { organisation: o?.name || m.organisationId, art: o?.art || "", rolle: teamRolleLabel(m.rolle), seit: m.seit };
   710	      }),
   711	    vorgaenge: eigene.vorgaenge.map((x) => ({ ...ohne(x, intern), statusVerlauf: verlauf(x.statusVerlauf) })),
   712	    flaechen: eigene.flaechen.map((x) => ({ ...ohne(x, intern), statusVerlauf: verlauf(x.statusVerlauf) })),
   713	    projekte: eigene.projekte.map((x) => ({ ...ohne(x, intern), statusVerlauf: verlauf(x.statusVerlauf) })),
   714	    interessen: eigene.interessen.map((x) => ({ ...ohne(x, intern), statusVerlauf: verlauf(x.statusVerlauf) })),
   715	    provisionen: db.provisionen.filter((x) => x.anbieterId === u.id).map((x) => ({ id: x.id, projektId: x.projektId, satz: x.satz, betrag: x.betrag, status: x.status, erstelltAm: x.erstelltAm })),
   716	    nachrichten: db.nachrichten
   717	      .filter((m) => bezuege.has(`${m.bezugArt}:${m.bezugId}`))
   718	      .map((m) => ({ id: m.id, bezug: `${m.bezugArt}:${m.bezugId}`, am: m.am, von: m.vonId === u.id && !m.system ? "Sie" : m.system ? "System" : m.vonAdmin ? "ÖkoInvest" : "Teammitglied", text: m.text })),
   719	    dokumente: db.dokumente
   720	      .filter((d) => d.hochgeladenVon === u.id || bezuege.has(`${d.bezugArt}:${d.bezugId}`))

   722	    benachrichtigungen: (db.benachrichtigungen || []).filter((b) => b.nutzerId === u.id).map((b) => ohne(b, ["nutzerId"])),
   723	    einladungen: db.einladungen.filter((e) => e.email === u.email).map((e) => ({ organisation: db.organisationen.find((o) => o.id === e.organisationId)?.name || "", rolle: teamRolleLabel(e.rolle), status: e.status, erstelltAm: e.erstelltAm })),
   724	    aktivitaeten: db.aktivitaeten.filter((a) => a.nutzerId === u.id).map((a) => ohne(a, ["nutzerId", "prevHash", "hash"])),
   725	  };
   726	  await aendern((d) => internSicherheit.protokoll(d, u, "datenexport", "nutzer", u.id, "Datenexport (Art. 15/20 DSGVO) heruntergeladen"));
   727	  // E-Mail-Adressen Dritter (z. B. in Protokolltexten „Einladung an …“) nur maskiert ausgeben
   728	  const maske = (x) => (x.toLowerCase() === u.email ? x : `${x[0]}•••@${x.split("@")[1]}`);
   729	  return JSON.parse(JSON.stringify(daten).replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+[.][A-Za-z]{2,}/g, maske));

   732	/**
   733	 * Konto löschen (Art. 17 DSGVO): Passwort + Bestätigung „LÖSCHEN“. Das Konto wird anonymisiert und gesperrt,
   734	 * Sitzungen, Geräte, 2FA, Token, Benachrichtigungen und Mitgliedschaften entfallen, Entwürfe werden gelöscht.
   735	 * Unterlagen mit Aufbewahrungspflicht (Provisionen, NDA-/Einwilligungsnachweise, Vorgänge) bleiben – ohne
   736	 * Zugang – bis zum Fristablauf erhalten; das Backoffice wird informiert.
   737	 */
   738	export async function kontoLoeschen(n, { passwort, bestaetigung } = {}) {
   739	  if (!n) throw nichtAngemeldet();
   740	  v.pflicht(["LÖSCHEN", "LOESCHEN"].includes(String(bestaetigung || "").trim().toUpperCase()), "Bitte bestätigen Sie mit dem Wort LÖSCHEN.");
   741	  const db0 = await lesen();
   742	  const u0 = nutzerById(db0, n.id);
   743	  if (!(await passwortPruefen(String(passwort || ""), u0?.passwortHash))) throw ungueltig("Das Passwort stimmt nicht.");
   744	  return aendern((db) => {
   745	    sammlungen(db);
   746	    const u = nutzerById(db, n.id);
   747	    if (u.rollen.includes("admin") && db.nutzer.filter((x) => x.aktiv && x.rollen.includes("admin")).length <= 1) {
   748	      throw konflikt("Das letzte Backoffice-Konto kann nicht gelöscht werden.");
   749	    }
   750	    const inhaberMitTeam = db.mitgliedschaften.filter((m) => m.nutzerId === u.id && m.rolle === "inhaber").filter((m) => db.mitgliedschaften.some((x) => x.organisationId === m.organisationId && x.nutzerId !== u.id));
   751	    if (inhaberMitTeam.length) throw konflikt("Sie sind Inhaber einer Organisation mit weiteren Mitgliedern. Bitte übertragen Sie zuerst die Inhaberschaft.");
   752	    // Organisationen, in denen das Konto allein ist, entfallen (Objekte werden wieder persönlich)
   753	    for (const m of db.mitgliedschaften.filter((x) => x.nutzerId === u.id && x.rolle === "inhaber")) {
   754	      for (const liste of [db.flaechen, db.projekte, db.interessen]) for (const o of liste) if (o.organisationId === m.organisationId) o.organisationId = null;
   755	      db.organisationen = db.organisationen.filter((o) => o.id !== m.organisationId);
   756	      db.einladungen = db.einladungen.filter((e) => e.organisationId !== m.organisationId);
   757	    }
   758	    db.mitgliedschaften = db.mitgliedschaften.filter((m) => m.nutzerId !== u.id && db.organisationen.some((o) => o.id === m.organisationId));
   759	    const entwuerfeF = db.flaechen.filter((f) => f.nutzerId === u.id && f.status === "entwurf" && !f.organisationId).map((f) => f.id);
   760	    const entwuerfeP = db.projekte.filter((p) => p.anbieterId === u.id && p.status === "entwurf" && !p.organisationId).map((p) => p.id);
   761	    db.flaechen = db.flaechen.filter((f) => !entwuerfeF.includes(f.id));
   762	    db.projekte = db.projekte.filter((p) => !entwuerfeP.includes(p.id));
   763	    const alteEmail = u.email;
   764	    db.einladungen = db.einladungen.filter((e) => e.email !== alteEmail);
   765	    db.zweiFaktor = db.zweiFaktor.filter((z) => z.nutzerId !== u.id);
   766	    db.sitzungen = db.sitzungen.filter((s) => s.nutzerId !== u.id);
   767	    db.geraete = db.geraete.filter((g) => g.nutzerId !== u.id);
   768	    db.anmeldungen = db.anmeldungen.filter((a) => a.nutzerId !== u.id);
   769	    db.tokens = db.tokens.filter((t) => t.nutzerId !== u.id);
   770	    db.benachrichtigungen = (db.benachrichtigungen || []).filter((b) => b.nutzerId !== u.id);
   771	    delete db.anmeldeSperren[alteEmail];
   772	    Object.assign(u, {
   773	      email: `geloescht-${u.id.toLowerCase()}@geloescht.invalid`, vorname: "Gelöschtes", nachname: "Konto", firma: "", telefon: "", passwortHash: null,
   774	      aktiv: false, emailBestaetigt: false, suchprofil: null, komfort: null, mailEinstellungen: null, sitzungVersion: (u.sitzungVersion || 0) + 1, geloeschtAm: jetzt(),
   775	    });
   776	    for (const a of db.nutzer.filter((x) => x.aktiv && x.rollen.includes("admin"))) {
   777	      internSicherheit.benachrichtigen(db, a, { art: "status", titel: "Konto gelöscht", text: `Konto ${u.id} wurde auf eigenen Wunsch gelöscht (${entwuerfeF.length + entwuerfeP.length} Entwürfe entfernt). Offene Vorgänge bitte prüfen.`, link: "/portal/admin/nutzer", objektArt: "nutzer", objektId: u.id }, { mail: false });
   778	    }
   779	    internSicherheit.protokoll(db, { id: u.id }, "konto_geloescht", "nutzer", u.id, `Konto gelöscht (anonymisiert), ${entwuerfeF.length} Flächen- und ${entwuerfeP.length} Projekt-Entwürfe entfernt`);
   780	    return { ok: true };
   781	  });
