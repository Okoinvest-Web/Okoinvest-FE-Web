MAX_BASE64 = (UPLOAD_MAX_BYTES + 2) // 3 * 4 + 16
FELDER = ["name", "bezug_art", "bezug_id", "dateiname", "mime", "groesse", "kategorie", "am", "hochgeladen_von", "datei_id"]


def mime_aus_bytes(b):
	"""Dateityp an den ersten Bytes erkennen (nie dem Dateinamen/Browser-MIME trauen)."""
	if len(b) >= 5 and b[:5] == b"%PDF-":
		return "application/pdf"
	if len(b) >= 3 and b[:3] == b"\xff\xd8\xff":
		return "image/jpeg"
	if len(b) >= 8 and b[:8] == b"\x89PNG\r\n\x1a\n":
		return "image/png"
	return ""


def darf_dokument(n, d):
	if ist_admin(n):
		return True
	if d.hochgeladen_von == n.id:
		return True
	o = objekt_laden(d.bezug_art, d.bezug_id)
	if not o:
		return False
	if besitzer(d.bezug_art, o) == n.id:
		return True
	if d.bezug_art == "projekt" and d.kategorie == "datenraum" and o.status in (*BOERSE_STATUS, "verkauft"):
		status = frappe.db.get_value("Plattform Interesse", {"projekt_id": o.name, "investor_id": n.id}, "status")
		return status in DATENRAUM_STATUS
	return False


def _mit_bezug(meta, n, cache):
	schluessel = (meta["bezugArt"], meta["bezugId"])
	if schluessel not in cache:
		cache[schluessel] = thread_info(meta["bezugArt"], meta["bezugId"], n) or {}
	info = cache[schluessel]
	return {**meta, "bezugTitel": info.get("titel") or meta["bezugId"], "bezugLink": info.get("link")}


@frappe.whitelist(methods=["POST"])
def liste(bezugArt=None, bezugId=None, als_benutzer=None):
	n = nutzer(als_benutzer)
	if bezugArt:
		if bezugArt not in BEZUG_ARTEN or not v.kennung(bezugId):
			return []
		kandidaten = frappe.get_all("Plattform Dokument", filters={"bezug_art": bezugArt, "bezug_id": bezugId}, fields=FELDER, limit_page_length=0)
	elif ist_admin(n):
		kandidaten = frappe.get_all("Plattform Dokument", fields=FELDER, order_by="am desc", limit_page_length=500)
	else:
		# Übersicht: nur Dokumente an eigenen Objekten bzw. eigene Uploads (kein Datenraum fremder Projekte)
		from oekoinvest_app.plattform.api.nachricht import eigene_bezuege

		gesehen = {}
		for d in frappe.get_all("Plattform Dokument", filters={"hochgeladen_von": n.id}, fields=FELDER, limit_page_length=0):
			gesehen[d.name] = d
		for art, namen in eigene_bezuege(n).items():
			if namen:
				for d in frappe.get_all("Plattform Dokument", filters={"bezug_art": art, "bezug_id": ("in", namen)}, fields=FELDER, limit_page_length=0):
					gesehen[d.name] = d
		kandidaten = list(gesehen.values())
	erlaubt = [d for d in kandidaten if darf_dokument(n, d)]
	erlaubt.sort(key=lambda d: str(d.am or ""), reverse=True)
	cache = {}
	return [_mit_bezug(dokument_meta(d), n, cache) for d in erlaubt[:500]]


@frappe.whitelist(methods=["POST"])
def lesen(id=None, als_benutzer=None):  # noqa: A002
	n = nutzer(als_benutzer)
	k = v.kennung(id)
	if not k or not frappe.db.exists("Plattform Dokument", k):
		nicht_gefunden("Dokument nicht gefunden.")
	d = frappe.get_doc("Plattform Dokument", k)
	if not darf_dokument(n, d):
		verboten()
	if not d.datei_id or not frappe.db.exists("File", d.datei_id):
		nicht_gefunden("Dokument nicht gefunden.")
	datei = frappe.get_doc("File", d.datei_id)
	with open(datei.get_full_path(), "rb") as fh:
		inhalt = fh.read()
	if d.sha256 and sha256(inhalt) != d.sha256:
		frappe.log_error(title="ÖkoInvest: Prüfsumme eines Dokuments stimmt nicht", message=f"Dokument {d.name}")
		ungueltig("Das Dokument ist beschädigt. Bitte wenden Sie sich an das Backoffice.")
	# Download-Protokoll (Datenraum: wer hat wann welche Unterlage geöffnet)
	protokoll(n, "download", d.bezug_art, d.bezug_id, f"Dokument {d.dateiname} geöffnet")
	return {"meta": dokument_meta(d), "inhaltBase64": base64.b64encode(inhalt).decode("ascii")}


@frappe.whitelist(methods=["POST"])
def speichern(bezugArt=None, bezugId=None, name=None, mime=None, kategorie=None, inhaltBase64=None, als_benutzer=None):
	n = nutzer(als_benutzer)
	if mime not in UPLOAD_TYPEN:
		ungueltig("Erlaubt sind PDF, JPG und PNG.")
	roh = inhaltBase64 if isinstance(inhaltBase64, str) else ""
	if not roh or len(roh) > MAX_BASE64:
		ungueltig("Die Datei ist leer oder größer als 10 MB.")
	try:
		daten = base64.b64decode(roh, validate=True)
	except (binascii.Error, ValueError):
		ungueltig("Die Datei konnte nicht gelesen werden.")
	if not daten or len(daten) > UPLOAD_MAX_BYTES:
		ungueltig("Die Datei ist leer oder größer als 10 MB.")
	if mime_aus_bytes(daten) != mime:
		ungueltig("Erlaubt sind PDF, JPG und PNG.")
	kat = v.auswahl(kategorie, DOKUMENT_KATEGORIEN, "allgemein")
	if kat == "datenraum" and bezugArt != "projekt":
		ungueltig("Datenraum-Dokumente gehören zu einem Projekt.")
	endung = UPLOAD_TYPEN[mime]
	sauber = re.sub(r'[\\/:*?"<>|]', "_", v.text(name, 120)) or f"dokument.{endung}"

	o = darf_bezug(n, bezugArt, bezugId)  # Rechte vor dem Schreiben der Datei
	eigen = ist_eigen(bezugArt, o, n)
	als_admin = ist_admin(n) and not eigen

	d = frappe.get_doc(
		{
			"doctype": "Plattform Dokument",
			"bezug_art": bezugArt,
			"bezug_doctype": DOCTYPE[bezugArt],  # vor dem Insert nötig: Dynamic Link wird vor validate() geprüft
			"bezug_id": o.name,
			"dateiname": sauber,
			"mime": mime,
			"groesse": len(daten),
			"kategorie": kat,
			"hochgeladen_von": n.id,
			"am": jetzt(),
			"sha256": sha256(daten),
		}
	).insert(ignore_permissions=True)
	datei = frappe.get_doc(
		{
			"doctype": "File",
			"file_name": f"{d.name}-{secrets.token_hex(8)}.{endung}",
			"is_private": 1,
			"content": daten,
			"attached_to_doctype": "Plattform Dokument",
			"attached_to_name": d.name,
		}
	)
	datei.flags.ignore_permissions = True
	datei.insert(ignore_permissions=True)
	d.datei = datei.file_url
	d.datei_id = datei.name
	d.save(ignore_permissions=True)

	nachricht_anlegen(
		bezugArt,
		o.name,
		von_id=n.id,
		von_admin=als_admin,
		system=True,
		text=f"Dokument hochgeladen: {sauber}",
		gelesen_nutzer=None if als_admin else True,
		gelesen_admin=None if als_admin else False,
	)
	protokoll(n, "upload", bezugArt, o.name, f"Dokument {sauber} ({len(daten)} B)")
	return dokument_meta(d)


@frappe.whitelist(methods=["POST"])
def loeschen(id=None, als_benutzer=None):  # noqa: A002
	n = nutzer(als_benutzer)
	k = v.kennung(id)
	if not k or not frappe.db.exists("Plattform Dokument", k):
		nicht_gefunden("Dokument nicht gefunden.")
	d = frappe.get_doc("Plattform Dokument", k)
	if not ist_admin(n) and d.hochgeladen_von != n.id:
		verboten("Nur eigene Uploads können gelöscht werden.")
	protokoll(n, "loeschen", d.bezug_art, d.bezug_id, f"Dokument {d.dateiname} gelöscht")
	datei_id = d.datei_id
	if datei_id and frappe.db.exists("File", datei_id):
		frappe.delete_doc("File", datei_id, ignore_permissions=True, force=True)
	frappe.delete_doc("Plattform Dokument", d.name, ignore_permissions=True, force=True)
	return {"ok": True}

def _suchprofil(n):
	if not n:
		return None
	return aus_json(frappe.db.get_value("Plattform Konto", n.id, "suchprofil"))


def boerse_eintrag(p, n, profil=None):
	pj = projekt_json(p)
	o = projekt_oeffentlich(pj)
	if n and hat_rolle(n, ["investor"]):
		i = frappe.db.get_value("Plattform Interesse", {"projekt_id": p.name, "investor_id": n.id}, ["name", "status", "gemerkt"], as_dict=True)
		o["meinInteresse"] = {"id": i.name, "status": i.status, "gemerkt": bool(cint(i.gemerkt))} if i else None
		if profil:
			o["matching"] = matching(pj, profil)
	return o


@frappe.whitelist(methods=["POST"])
def liste(typ=None, land=None, phase=None, mwpMin=None, mwpMax=None, als_benutzer=None):
	n = nutzer(als_benutzer, pflicht=False)
	filters = {"status": ("in", BOERSE_STATUS)}
	t = v.auswahl(typ, PROJEKT_TYPEN)
	if t:
		filters["typ"] = t
	la = v.auswahl(land, LAENDER)
	if la:
		filters["land"] = la
	ph = v.auswahl(phase, PHASEN_IDS)
	if ph:
		filters["phase"] = ph
	mn = v.zahl(mwpMin, min=0)
	mx = v.zahl(mwpMax, min=0)
	if mn is not None and mx is not None:
		filters["mwp"] = ("between", [mn, mx])
	elif mn is not None:
		filters["mwp"] = (">=", mn)
	elif mx is not None:
		filters["mwp"] = ("<=", mx)
	namen = frappe.get_all("Plattform Projekt", filters=filters, pluck="name", order_by="veroeffentlicht_am desc", limit_page_length=500)
	profil = _suchprofil(n) if n and hat_rolle(n, ["investor"]) else None
	projekte = [boerse_eintrag(frappe.get_doc("Plattform Projekt", x), n, profil) for x in namen]

	sichtbar = frappe.get_all("Plattform Projekt", filters={"status": ("in", BOERSE_STATUS)}, fields=["mwp", "land", "demo"], limit_page_length=0)
	return {
		"projekte": projekte,
		"kennzahlen": {
			"anzahl": len(sichtbar),
			"mwp": round(sum(float(x.mwp or 0) for x in sichtbar) * 10) / 10,
			"laender": len({x.land for x in sichtbar if x.land}),
			"verkauft": frappe.db.count("Plattform Projekt", {"status": "verkauft"}),
		},
		# Hinweis „Vorführdaten“ nur, wenn Demo-Projekte in der Börse stehen
		"demo": any(cint(x.demo) for x in sichtbar),
	}


@frappe.whitelist(methods=["POST"])
def detail(schluessel=None, zaehlen=None, als_benutzer=None):
	n = nutzer(als_benutzer, pflicht=False)
	s = v.slug(schluessel) or v.kennung(schluessel)
	name = None
	if s:
		name = frappe.db.get_value("Plattform Projekt", {"slug": s}, "name") or (s if frappe.db.exists("Plattform Projekt", s) else None)
	if not name:
		nicht_gefunden("Projekt nicht gefunden.")
	p = frappe.get_doc("Plattform Projekt", name)
	eigentuemer = bool(n) and p.anbieter_id == n.id
	admin = ist_admin(n)
	if p.status not in BOERSE_STATUS and not eigentuemer and not admin:
		nicht_gefunden("Projekt nicht gefunden.")

	if v.wahr(zaehlen) and not eigentuemer and not admin:
		frappe.db.sql("update `tabPlattform Projekt` set aufrufe = aufrufe + 1 where name = %s", (p.name,))
		p.aufrufe = cint(p.aufrufe) + 1

	interesse = None
	if n:
		iname = frappe.db.get_value("Plattform Interesse", {"projekt_id": p.name, "investor_id": n.id}, "name")
		interesse = frappe.get_doc("Plattform Interesse", iname) if iname else None
	nda_frei = bool(interesse) and interesse.status in DATENRAUM_STATUS
	zugriff = "admin" if admin else "eigentuemer" if eigentuemer else "nda" if nda_frei else "oeffentlich"

	profil = _suchprofil(n) if n and hat_rolle(n, ["investor"]) else None
	out = {
		"projekt": boerse_eintrag(p, n, profil),
		"zugriff": zugriff,
		"interesse": interesse_json(interesse, n) if interesse else None,
		"vertraulich": None,
		"dokumente": [],
	}
	if zugriff != "oeffentlich":
		out["vertraulich"] = projekt_vertraulich(projekt_json(p))
		filters = {"bezug_art": "projekt", "bezug_id": p.name}
		if zugriff == "nda":
			filters["kategorie"] = "datenraum"
		out["dokumente"] = [dokument_meta(d) for d in frappe.get_all("Plattform Dokument", filters=filters, fields=["*"], order_by="am desc")]
	return out

def vorgang_json(vg):
	return {
		"id": vg.name,
		"typ": vg.typ,
		"status": vg.status,
		"statusVerlauf": verlauf_json(vg),
		"nutzerId": vg.nutzer_id or None,
		"email": vg.email,
		"kontakt": aus_json(vg.kontakt, {}),
		"daten": aus_json(vg.daten, {}),
		"herkunft": aus_json(vg.herkunft, {}),
		"ip": vg.ip or "",
		"einwilligung": aus_json(vg.einwilligung),
		"zustaendigId": vg.zustaendig_id or None,
		"flaecheId": vg.flaeche_id or None,
		"ablehnGrund": vg.ablehn_grund or "",
		"erstelltAm": iso(vg.creation),
		"geaendertAm": iso(vg.modified),
	}


def vorgang_ausgabe(vg, n):
	out = vorgang_json(vg)
	out["zustaendig"] = kurz(vg.zustaendig_id)
	if ist_admin(n):
		out["nutzer"] = kurz(vg.nutzer_id)
	else:
		out.pop("ip", None)
		out.pop("herkunft", None)
	return out

import frappe

from oekoinvest_app.plattform.basis import PlattformDokument

MELDUNG = "Das Aktivitätsprotokoll kann nicht verändert werden."


class PlattformAktivitaet(PlattformDokument):
	def validate(self):
		if not self.is_new():
			frappe.throw(MELDUNG, frappe.PermissionError)

	def on_trash(self):
		frappe.throw("Einträge des Aktivitätsprotokolls können nicht gelöscht werden.", frappe.PermissionError)

	@classmethod
	def setUpClass(cls):
		super().setUpClass()

		cls.admin = konto(["admin"])

		frappe.set_user("Administrator")

	def _eintrag(self):
		frappe.set_user("Administrator")
		protokoll(frappe._dict(id=self.admin), "test", "nutzer", self.admin, "Testeintrag", "127.0.0.1")
		return frappe.get_all("Plattform Aktivitaet", filters={"aktion": "test", "objekt_id": self.admin}, pluck="name", order_by="creation desc", limit=1)[0]

	def test_anlegen_erlaubt(self):
		name = self._eintrag()
		doc = frappe.get_doc("Plattform Aktivitaet", name)
		self.assertTrue(doc.name.startswith("A-"))
		self.assertEqual(doc.text, "Testeintrag")

		doc = frappe.get_doc("Plattform Aktivitaet", self._eintrag())
		doc.text = "manipuliert"
		with self.assertRaises(frappe.PermissionError):
			doc.save(ignore_permissions=True)
		self.assertEqual(frappe.db.get_value("Plattform Aktivitaet", doc.name, "text"), "Testeintrag")

		name = self._eintrag()

			frappe.delete_doc("Plattform Aktivitaet", name, ignore_permissions=True, force=True)

# Das Konto erhält KEIN Passwort – es kommt eine E-Mail mit Link „Neues Passwort festlegen“.

import frappe

from oekoinvest_app.plattform.konstanten import ROLLE_API, ROLLE_DESK_ADMIN, ROLLEN

DESK_ROLLEN = {ROLLE_DESK_ADMIN: 1, ROLLE_API: 0}


def rollen_anlegen():
	for rolle, desk in [*DESK_ROLLEN.items(), *[(r[0], 0) for r in ROLLEN.values()]]:
		if not frappe.db.exists("Role", rolle):
			frappe.get_doc({"doctype": "Role", "role_name": rolle, "desk_access": desk}).insert(ignore_permissions=True)


def stammdaten_anlegen():
	rollen_anlegen()
	for rolle_id, (frappe_rolle, bezeichnung, waehlbar) in ROLLEN.items():
		werte = {"bezeichnung": bezeichnung, "frappe_rolle": frappe_rolle, "selbst_waehlbar": waehlbar}
		if frappe.db.exists("Plattform Rolle", rolle_id):
			doc = frappe.get_doc("Plattform Rolle", rolle_id)
			doc.update(werte)
			doc.save(ignore_permissions=True)
		else:
			frappe.get_doc({"doctype": "Plattform Rolle", "rolle_id": rolle_id, **werte}).insert(ignore_permissions=True)


def nach_installation():
	stammdaten_anlegen()
	frappe.db.commit()


def nach_migration():
	stammdaten_anlegen()


def admin_konto_anlegen(email, vorname, nachname):
	"""Erstes Backoffice-Konto (Rolle admin) anlegen und Passwort-Link senden."""
	from oekoinvest_app.plattform import mail
	from oekoinvest_app.plattform import validierung as v
	from oekoinvest_app.plattform.kern import token_anlegen

	stammdaten_anlegen()
	e = v.email(email)
	if not e:
		frappe.throw("Ungültige E-Mail-Adresse.")
	if frappe.db.exists("Plattform Konto", {"email": e}):
		frappe.throw("Für diese E-Mail-Adresse besteht bereits ein Konto.")
	if not frappe.db.exists("User", e):
		user = frappe.get_doc({"doctype": "User", "email": e, "first_name": vorname, "last_name": nachname, "user_type": "Website User", "send_welcome_email": 0})
		user.flags.no_welcome_mail = True
		user.insert(ignore_permissions=True)
	konto = frappe.get_doc(
		{
			"doctype": "Plattform Konto",
			"user": e,
			"email": e,
			"vorname": v.text(vorname, 80),
			"nachname": v.text(nachname, 80),
			"firma": "ÖkoInvest GmbH",
			"rollen": [{"rolle": "admin"}],
			"aktiv": 1,
			"email_bestaetigt": 1,
		}
	).insert(ignore_permissions=True)
	token = token_anlegen("reset", konto.name, 24)
	if not mail.passwort_link(konto, token):
		print("Achtung: E-Mail nicht versendet – plattform_basis_url und ausgehendes E-Mail-Konto prüfen.")
	frappe.db.commit()
	print(f"Backoffice-Konto {konto.name} ({e}) angelegt. Passwort-Link ist 24 Stunden gültig.")
	return konto.name
