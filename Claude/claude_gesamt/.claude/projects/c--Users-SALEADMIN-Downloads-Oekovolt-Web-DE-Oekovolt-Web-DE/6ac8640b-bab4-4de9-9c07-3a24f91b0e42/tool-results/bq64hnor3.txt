    99	    }
   100	  });
   101	  return lesen();

   138	/** Online-Identifizierung starten → { sitzung, url } */
   139	export async function identStarten(n, { basisUrl = "" } = {}) {
   140	  angemeldet(n);
   141	  const k = identKonfig();
   142	  if (!k.aktiv) throw new PlattformFehler(503, "Die Online-Identifizierung ist derzeit nicht verfügbar. Bitte laden Sie Ihre Ausweiskopie hoch – unser Team prüft manuell.");
   143	  const vorhanden = await lesenBereit();
   144	  const heute = jetzt().slice(0, 10);
   145	  if (D(vorhanden).identitaet.filter((s) => s.kontoId === n.id && s.erstelltAm.slice(0, 10) === heute).length >= 5) throw new PlattformFehler(429, "Heute wurden bereits mehrere Identifizierungen gestartet. Bitte versuchen Sie es morgen erneut.");
   146	  const offen = D(vorhanden).identitaet.find((s) => s.kontoId === n.id && s.status === "ausstehend" && Date.now() - Date.parse(s.erstelltAm) < 30 * 60000);
   147	  if (offen) return { sitzung: sitzungAusgabe(offen), url: offen.url, wiederverwendet: true };
   148	  const id = neueId("IP");
   149	  const r = await sitzungStarten(k, { id, vorname: n.vorname, nachname: n.nachname, email: n.email, rueckUrl: `${basisUrl}/portal/verifizierung?ident=${id}` }).catch((e) => {
   150	    throw new PlattformFehler(e.status || 502, e.message);
   151	  });
   152	  return aendernBereit((db) => {
   153	    const s = { id, kontoId: n.id, anbieter: k.anbieter, status: "ausstehend", url: r.url, externeId: r.externeId || null, erstelltAm: jetzt(), ereignisse: [] };
   154	    D(db).identitaet.push(s);
   155	    if (D(db).identitaet.length > 5000) D(db).identitaet.splice(0, D(db).identitaet.length - 5000);
   156	    protokoll(db, n, "ident_start", "ident", id, `Online-Identifizierung gestartet (${k.label})`);
   157	    return { sitzung: sitzungAusgabe(s), url: r.url };
   158	  });
   159	}

   163	/**
   164	 * Webhook verarbeiten (Systemaufruf ohne Nutzer). kopf: (name) => Wert, roh: Roh-Body.
   165	 * Liefert { status: 200|401|400, antwort }. Signatur falsch → 401 ohne Details. Wiederholte Ereignisse → 200 (idempotent).
   166	 */
   167	export async function identWebhook({ anbieter, kopf, roh }) {
   168	  const k = identKonfig();
   169	  if (!k.anbieter || anbieter !== k.anbieter) return { status: 404, antwort: { ok: false } };
   170	  const sig = webhookSignaturPruefen(anbieter, kopf, roh, { geheimnis: k.geheimnis });
   171	  if (!sig.ok) return { status: 401, antwort: { ok: false, error: "Signatur ungültig." } };
   172	  let p;
   173	  try {
   174	    p = JSON.parse(roh);
   175	  } catch {
   176	    return { status: 400, antwort: { ok: false, error: "Ungültiges JSON." } };
   177	  }
   178	  const e = webhookNormalisieren(anbieter, p);
   179	  if (!e) return { status: 200, antwort: { ok: true, verarbeitet: false } };
   180	  const r = await aendernBereit((db) => {
   181	    const s = D(db).identitaet.find((x) => x.id === v.kennung(e.sitzungId) && x.anbieter === anbieter);
   182	    if (!s) return { unbekannt: true };
   183	    const ereignisId = e.ereignisId || sha256(roh).slice(0, 24);
   184	    if ((D(db).ereignisse || []).includes(`${anbieter}:${ereignisId}`)) return { doppelt: true };
   185	    D(db).ereignisse.push(`${anbieter}:${ereignisId}`);
   186	    if (D(db).ereignisse.length > 5000) D(db).ereignisse.splice(0, 1000);
   187	    s.status = e.ergebnis;
   188	    s.verfahren = e.verfahren;
   189	    s.grund = e.grund;
   190	    s.externeId = e.externeId || s.externeId;
   191	    s.abgeschlossenAm = jetzt();
   192	    s.ereignisse = [...(s.ereignisse || []), { am: jetzt(), ergebnis: e.ergebnis, ereignisId }].slice(-20);
   193	    s.url = null;
   194	    protokoll(db, null, "ident_ergebnis", "ident", s.id, `Identitätsanbieter ${IDENT_ANBIETER[anbieter]?.label}: ${IDENT_ERGEBNIS[e.ergebnis]?.label}`);
   195	    const u = nutzerById(db, s.kontoId);
   196	    if (u) benachrichtigen(db, u, { art: "status", titel: "Online-Identifizierung abgeschlossen", text: "Das Ergebnis liegt unserem Team vor. Die abschließende Prüfung Ihrer Verifizierung erfolgt durch die Compliance.", link: "/portal/verifizierung", objektArt: "kyc", objektId: s.id }, { mail: false });
   197	    return { sitzung: { ...s }, ereignis: e };
   198	  });
   199	  if (r.unbekannt) return { status: 200, antwort: { ok: true, verarbeitet: false } };
   200	  if (r.doppelt) return { status: 200, antwort: { ok: true, verarbeitet: false, doppelt: true } };
   201	  const eintrag = await anKycHaengen({ kontoId: r.sitzung.kontoId }, {
   202	    quelle: `Identitätsanbieter ${IDENT_ANBIETER[anbieter]?.label || anbieter}`, art: "identitaet", ergebnis: KYC_ERGEBNIS[r.ereignis.ergebnis] || "hinweis",
   203	    zusammenfassung: `${IDENT_ERGEBNIS[r.ereignis.ergebnis]?.label}${r.ereignis.verfahren ? ` · ${r.ereignis.verfahren}` : ""}${r.ereignis.grund ? ` · ${r.ereignis.grund}` : ""}`,
   204	    externeId: r.ereignis.externeId || r.sitzung.id, rohdatenRef: `Sitzung ${r.sitzung.id}`, daten: { sitzung: r.sitzung.id, ...r.ereignis.daten },
   205	  });
   206	  if (eintrag?.id) {
   207	    await aendern((db) => {
   208	      const s = D(db).identitaet.find((x) => x.id === r.sitzung.id);
   209	      if (s) s.kycEintragId = eintrag.id;
   210	    });
   211	  }
   212	  return { status: 200, antwort: { ok: true, verarbeitet: true } };
   213	}

   215	/** Demo-Simulator: eigenes Ergebnis wählen → signierter Webhook über denselben Prüfpfad */
   216	export async function identDemoAbschluss(n, { sitzungId, ergebnis } = {}) {
   217	  angemeldet(n);
   218	  const k = identKonfig();
   219	  if (!k.demo) throw verboten("Nur im Demo-Betrieb verfügbar.");
   220	  const db = await lesenBereit();
   221	  const s = D(db).identitaet.find((x) => x.id === v.kennung(sitzungId));
   222	  if (!s || s.kontoId !== n.id) throw nichtGefunden("Sitzung nicht gefunden.");
   223	  if (s.status !== "ausstehend") throw konflikt("Diese Sitzung ist bereits abgeschlossen.");
   224	  const e = v.auswahl(ergebnis, ["bestanden", "pruefen", "abgelehnt", "abgebrochen"], "");
   225	  if (!e) throw ungueltig("Bitte ein Ergebnis wählen.");
   226	  const roh = JSON.stringify({ sitzungId: s.id, externeId: `DEMO-${s.id}`, ergebnis: e, verfahren: "Simulation (Demo)", ereignisId: `demo-${s.id}-${e}`, nfc: e === "bestanden", liveness: e !== "abgebrochen" });
   227	  const kopfWerte = webhookSignieren("demo", roh, { geheimnis: k.geheimnis });
   228	  const r = await identWebhook({ anbieter: "demo", kopf: (h) => kopfWerte[h.toLowerCase()] ?? null, roh });
   229	  return { ok: r.status === 200, ...r.antwort };
   230	}

   232	/** MRZ-Vorprüfung: Prüfziffern, Ablauf, Abgleich mit Konto/KYC-Angaben → Hinweis im KYC-Fall (keine Entscheidung) */
   233	export async function mrzPruefen(n, { zeilen = "" } = {}) {
   234	  angemeldet(n);
   235	  const text = String(zeilen || "").slice(0, 400);
   236	  if (text.replace(/\s/g, "").length < 60) throw ungueltig("Bitte geben Sie alle Zeilen der maschinenlesbaren Zone ein.");
   237	  let fall = null;
   238	  try {
   239	    fall = (await kycDemo.kycFall(n))?.fall || null;
   240	  } catch {
   241	    fall = null;
   242	  }
   243	  const p = fall?.person || {};
   244	  const soll = { vorname: p.vorname || n.vorname, nachname: p.nachname || n.nachname, geburtsdatum: p.geburtsdatum || "", ausweisNummer: p.ausweis?.nummer || "" };
   245	  const r = mrzVorpruefung(text, soll);
   246	  const f = r.mrz.felder || {};
   247	  const eintrag = {
   248	    id: neueId("MZ"), kontoId: n.id, am: jetzt(), ergebnis: r.ergebnis, format: r.mrz.format, dokumentart: f.dokumentart || "", land: f.ausstellerstaat || "",
   249	    gueltigBis: f.gueltigBis || null, nummerMaskiert: nummerMaskiert(f.dokumentnummer), abgleich: r.abgleich, pruefziffern: `${r.mrz.pruefungen.filter((x) => x.ok).length}/${r.mrz.pruefungen.length}`,
   250	  };
   251	  await aendernBereit((db) => {
   252	    D(db).mrz.push(eintrag);
   253	    if (D(db).mrz.length > 5000) D(db).mrz.splice(0, 1000);
   254	    protokoll(db, n, "mrz_vorpruefung", "ident", eintrag.id, `MRZ-Vorprüfung: ${r.ergebnis} (${eintrag.pruefziffern} Prüfziffern)`);
   255	  });
   256	  const kyc = fall
   257	    ? await anKycHaengen({ kontoId: n.id }, {
   258	        quelle: "MRZ-Vorprüfung (automatisch)", art: "mrz", ergebnis: r.ergebnis === "plausibel" ? "ok" : r.ergebnis === "ungueltig" ? "fehler" : "auffaellig",
   259	        zusammenfassung: `${r.ergebnis === "plausibel" ? "Automatisch plausibel" : r.ergebnis === "ungueltig" ? "MRZ nicht stimmig" : "Abweichungen"} – ${f.dokumentart || "Dokument"} ${f.ausstellerstaat || ""}, Nr. ${eintrag.nummerMaskiert}, gültig bis ${f.gueltigBis || "–"}. ${r.hinweise.join(" ")}`.trim(),
   260	        externeId: eintrag.id,
   261	        daten: { format: r.mrz.format || "", pruefziffern: eintrag.pruefziffern, gueltigBis: f.gueltigBis || "", nameStimmt: r.abgleich.name ?? "", geburtsdatumStimmt: r.abgleich.geburtsdatum ?? "", nummerStimmt: r.abgleich.ausweisNummer ?? "" },

   282	export async function registerAbruf(n, { land, nummer, kycFallId = null, zweck = "" } = {}) {
   283	  registerSchreiben(n);
   284	  const l = v.auswahl(land, Object.keys(REGISTER_LAENDER), "");
   285	  if (!l) throw ungueltig("Bitte wählen Sie ein Land.");
   286	  const fallId = kycFallId ? v.kennung(kycFallId) : null;
   287	  if (kycFallId && !fallId) throw ungueltig("Ungültige Kennung des KYC-Falls.");
   288	  if (fallId && stufe(n, "compliance") < 2) throw verboten("Registerauszüge im KYC-Fall hinterlegt nur die Compliance.");
   289	  let r;
   290	  try {
   291	    r = await registerAbrufen({ land: l, nummer });
   292	  } catch (e) {
   293	    throw new PlattformFehler(e.status || 502, e.message);
   294	  }
   295	  const a = await aendernBereit((db) => {
   296	    const x = { id: neueId("RA"), land: l, nummer: registernummerPruefen(l, nummer).nummer, firma: r.firma, quelle: r.quelle, abrufAm: r.abrufAm, vonId: n.id, von: nameVon(db, n.id), zweck: v.text(zweck, 160), kycFallId: fallId };
   297	    D(db).register.unshift(x);
   298	    if (D(db).register.length > 3000) D(db).register.length = 3000;
   299	    protokoll(db, n, "register_abruf", "register", x.id, `Registerabruf ${REGISTER_LAENDER[l].register} ${x.nummer} (${r.quelle})`);
   300	    return x;
   301	  });
   302	  if (fallId) {
   303	    const f = a.firma;
   304	    const kritisch = !f || ["geloescht", "in_loeschung", "in_liquidation"].includes(f.status);
   305	    const e = await anKycHaengen(fallId, {
   306	      quelle: `${REGISTER_LAENDER[l].register}${r.quelle === "demo" ? " (Demo)" : ""}`, art: "register", ergebnis: kritisch ? "auffaellig" : "ok",
   307	      zusammenfassung: f ? `${f.name} · ${a.nummer} · ${f.rechtsform || ""} · Status ${f.status || "–"} · Sitz ${f.sitz || "–"} · abgerufen ${a.abrufAm.slice(0, 10)}` : `Unter ${a.nummer} wurde keine Firma gefunden.`,
   308	      externeId: a.id, rohdatenRef: `/api/admin/register/${a.id}/pdf`, daten: { nummer: a.nummer, status: f?.status || "nicht_gefunden", abrufAm: a.abrufAm, quelle: r.quelle },
   309	    });
   310	    if (!e) throw nichtGefunden("KYC-Fall nicht gefunden – der Abruf wurde gespeichert, aber nicht hinterlegt.");
   311	  }
   312	  return { auskunft: { ...auskunftKurz(a), firma: a.firma } };
   313	}

   374	/** Übersicht: Auskünfte, offene LOI-Voraussetzungen, Einstellungen */
   375	export async function bonitaetUebersicht(n, { q = "" } = {}) {
   376	  bonitaetLesen(n);
   377	  const db = await lesenBereit();
   378	  const e = bonitaetEinstellungenDb(db);
   379	  const such = v.text(q, 80).toLowerCase();
   380	  const auskuenfte = D(db).bonitaet.filter((a) => !such || `${a.subjekt?.name} ${a.id}`.toLowerCase().includes(such)).slice(0, 200).map((a) => auskunftAusgabe(db, a));
   381	  // Offene Voraussetzungen: Interessen mit freigegebenem Datenraum (nächster Schritt LOI/Gebot), Pflicht aktiv, keine gültige Auskunft
   382	  const offen = [];
   383	  const gesehen = new Set();
   384	  for (const i of db.interessen || []) {
   385	    if (!["nda_freigegeben", "loi_abgegeben"].includes(i.status)) continue;
   386	    const ziel = { kontoId: i.investorId, organisationId: i.organisationId || null };
   387	    const schluessel = ziel.organisationId || ziel.kontoId;
   388	    if (gesehen.has(schluessel)) continue;
   389	    const f = bonitaetFreigabeDb(db, "loi", ziel);
   390	    if (!f.aktiv || f.ok) continue;
   391	    gesehen.add(schluessel);
   392	    const s = subjektAufloesen(db, ziel.organisationId ? { art: "organisation", id: ziel.organisationId } : { art: "konto", id: ziel.kontoId });
   393	    offen.push({ interesseId: i.id, projektId: i.projektId, subjekt: { art: ziel.organisationId ? "organisation" : "konto", id: schluessel, name: s.name }, grund: f.grund });
   394	  }
   395	  const k = bonitaetKonfig();
   396	  const kandidaten = [
   397	    ...(db.organisationen || []).map((o) => ({ art: "organisation", id: o.id, name: o.name })),
   398	    ...(db.nutzer || []).filter((u) => u.aktiv && u.rollen.includes("investor")).map((u) => ({ art: "konto", id: u.id, name: u.firma?.trim() ? `${u.firma} (${nameVon(db, u.id)})` : nameVon(db, u.id) })),
   399	  ].slice(0, 400);
   400	  return {
   401	    auskuenfte, offen, kandidaten,
   402	    einstellungen: { ...e, geltungWirksam: bonitaetGeltung(e, process.env), pilotKonten: (e.pilotKonten || []).map((id) => ({ id, name: nameVon(db, id) || id })) },
   403	    konfig: { anbieter: k.anbieter, label: k.label, aktiv: k.aktiv, demo: k.demo },
   404	    rechtsgrundlagen: BONITAET_RECHTSGRUNDLAGEN, anlaesse: BONITAET_ANLAESSE,
   405	    schreiben: stufe(n, "bonitaet") >= 2,
   406	  };
   407	}
