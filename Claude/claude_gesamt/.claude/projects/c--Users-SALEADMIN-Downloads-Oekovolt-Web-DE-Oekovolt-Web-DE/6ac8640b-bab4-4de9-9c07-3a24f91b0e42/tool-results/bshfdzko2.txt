def _js_round(x):
	import math

	return int(math.floor(x + 0.5))


def provision_satz_normalisieren(wert):
	"""Eingabe > 0,2 wird als Prozent interpretiert (2 → 0,02). Liefert Satz oder wirft ValidationError."""
	satz = v.zahl(wert, min=0, max=100)
	if satz is not None and satz > 0.2:
		satz = satz / 100
	v.pflicht(
		satz is not None and PROVISION_MIN - 1e-9 <= satz <= PROVISION_MAX + 1e-9,
		f"Provisionssatz muss zwischen {PROVISION_MIN * 100:g} und {PROVISION_MAX * 100:g} % liegen.".replace(".", ","),
	)
	return round(satz, 4)


def uebergang_anwenden(n, art, o, nach, als, extra=None, ip=""):
	"""Geprüften Übergang auf Dokument `o` anwenden und speichern (inkl. Nebenwirkungen)."""
	extra = extra if isinstance(extra, dict) else {}
	von = o.status
	kommentar = v.mehrzeilig(extra.get("grund") or extra.get("kommentar") or "", 1000)

	if extra.get("grund"):
		o.ablehn_grund = kommentar

	if art == "flaeche" and nach == "pachtangebot":
		pa = extra.get("pachtangebot") if isinstance(extra.get("pachtangebot"), dict) else {}
		euro = v.zahl(pa.get("euroProHaJahr"), min=50, max=50000)
		jahre = v.zahl(pa.get("laufzeitJahre"), min=1, max=60)
		v.pflicht(euro is not None and jahre is not None, "Pachtangebot: Pacht je ha und Laufzeit angeben.")
		o.pachtangebot = als_json(
			{
				"euroProHaJahr": euro,
				"laufzeitJahre": jahre,
				"verlaengerung": v.text(pa.get("verlaengerung"), 80),
				"text": v.mehrzeilig(pa.get("text"), 2000),
				"am": jetzt_iso(),
			}
		)

	if art == "projekt" and nach == "veroeffentlicht":
		o.provision_satz = provision_satz_normalisieren(extra.get("provisionSatz"))
		if not o.veroeffentlicht_am:
			o.veroeffentlicht_am = jetzt()

	if art == "interesse" and nach == "loi_abgegeben":
		loi = extra.get("loi") if isinstance(extra.get("loi"), dict) else {}
		t = v.mehrzeilig(loi.get("text"), 3000)
		v.pflicht(len(t) >= 10, "Bitte beschreiben Sie Ihre Absichtserklärung (mind. 10 Zeichen).")
		o.loi = als_json(
			{
				"betrag": v.zahl(loi.get("betrag"), min=0, max=1e10),
				"text": t,
				"bedingungen": v.mehrzeilig(loi.get("bedingungen"), 2000),
				"gueltigBis": v.text(loi.get("gueltigBis"), 10),
				"am": jetzt_iso(),
			}
		)

	if art == "projekt" and nach == "verkauft":
		kaeufer_id = v.kennung(extra.get("interesseId"))
		kaeufer = frappe.db.get_value("Plattform Interesse", kaeufer_id, ["name", "projekt_id", "status"], as_dict=True) if kaeufer_id else None
		v.pflicht(
			kaeufer and kaeufer.projekt_id == o.name and kaeufer.status in ("nda_freigegeben", "loi_abgegeben", "loi_angenommen"),
			"Käufer muss ein Interessent mit freigegebenem Datenraum sein.",
		)
		kaufpreis = v.zahl(extra.get("kaufpreis"), min=1, max=1e11)
		v.pflicht(kaufpreis, "Bitte Kaufpreis angeben.")
		o.kaeufer_interesse_id = kaeufer.name
		o.kaufpreis = kaufpreis

	o.status = nach
	verlauf(o, nach, n, als, kommentar)
	o.flags.plattform_uebergang = (von, nach)
	o.flags.plattform_nutzer = n
	o.flags.plattform_ip = ip
	o.save(ignore_permissions=True)

	if art == "projekt" and nach == "veroeffentlicht":
		investoren_benachrichtigen(o)

	if art == "interesse" and nach == "loi_angenommen":
		p = frappe.get_doc("Plattform Projekt", o.projekt_id)
		if p.status == "veroeffentlicht":
			system_wechsel("projekt", p, "reserviert", "LOI eines Investors angenommen.")

	if art != "provision":
		text = f"Status: „{status_label(art, von)}“ → „{status_label(art, nach)}“." + (f" {kommentar}" if kommentar else "")
		if als == "eigentuemer":
			nachricht_anlegen(art, o.name, von_id=n.get("id") if n else None, von_admin=False, system=True, text=text, gelesen_nutzer=True, gelesen_admin=False)
		else:
			nachricht_anlegen(art, o.name, von_id=n.get("id") if n else None, von_admin=(als == "admin"), system=True, text=text)

	protokoll(n, "status", art, o.name, f"{status_label(art, von)} → {status_label(art, nach)}", ip)
	return o


def system_wechsel(art, o, nach, kommentar):
	"""Automatischer Übergang (wer = system bzw. admin-Übergang ohne Pflichtangaben) mit Nachricht."""
	von = o.status
	erlaubt = any(u["nach"] == nach and von in u["von"] and u["wer"] in ("system", "admin") for u in PROZESSE[art]["uebergaenge"])
	if not erlaubt:
		return False
	o.status = nach
	verlauf(o, nach, None, "system", kommentar)
	if art == "interesse" and nach == "abgelehnt":
		o.ablehn_grund = kommentar
	o.flags.plattform_uebergang = (von, nach)
	o.save(ignore_permissions=True)
	nachricht_anlegen(art, o.name, von_admin=True, system=True, text=f"Status: „{status_label(art, von)}“ → „{status_label(art, nach)}“. {kommentar}")
	return True


def verkauf_abschliessen(projekt):
	"""Nebenwirkung Projekt → verkauft (aus PlattformProjekt.on_update): Provision + Interessen."""
	if frappe.db.exists("Plattform Provision", {"projekt_id": projekt.name}):
		return None
	kaeufer = frappe.get_doc("Plattform Interesse", projekt.kaeufer_interesse_id)
	satz = flt(projekt.provision_satz) or PROVISION_MIN
	kaufpreis = flt(projekt.kaufpreis)
	pv = frappe.get_doc(
		{
			"doctype": "Plattform Provision",
			"projekt_id": projekt.name,
			"anbieter_id": projekt.anbieter_id,
			"investor_id": kaeufer.investor_id,
			"interesse_id": kaeufer.name,
			"kaufpreis": kaufpreis,
			"satz": satz,
			"betrag": _js_round(kaufpreis * satz),
			"status": "offen",
		}
	)
	verlauf(pv, "offen", None, "system", "Automatisch bei Verkauf berechnet.")
	pv.insert(ignore_permissions=True)

	for name in frappe.get_all("Plattform Interesse", filters={"projekt_id": projekt.name}, pluck="name"):
		i = frappe.get_doc("Plattform Interesse", name)
		if i.name == kaeufer.name:
			system_wechsel("interesse", i, "abgeschlossen", "Kauf abgeschlossen.")
		elif i.status not in ("abgelehnt", "abgeschlossen"):
			system_wechsel("interesse", i, "abgelehnt", "Projekt anderweitig verkauft.")

	n = projekt.flags.get("plattform_nutzer")
	protokoll(
		n,
		"provision",
		"provision",
		pv.name,
		f"Provision {cint(pv.betrag)} € ({flt(satz) * 100:g} %) für {projekt.name}",
		projekt.flags.get("plattform_ip") or "",
	)
	return pv


def investoren_benachrichtigen(p):
	"""Neu veröffentlicht → Investoren mit passendem Suchprofil (Score ≥ 70) erhalten eine System-Nachricht."""
	from oekoinvest_app.plattform.ausgabe import projekt_json

	pj = projekt_json(p)
	konten = frappe.get_all(
		"Plattform Konto",
		filters={"aktiv": 1, "name": ("!=", p.anbieter_id)},
		fields=["name", "suchprofil"],
		limit_page_length=0,
	)
	investoren = set(frappe.get_all("Plattform Konto Rolle", filters={"rolle": "investor", "parenttype": "Plattform Konto"}, pluck="parent"))
	for k in konten:
		if k.name not in investoren:
			continue
		profil = aus_json(k.suchprofil) or {}
		if not profil.get("benachrichtigen"):
			continue
		m = matching(pj, profil)
		if m["score"] >= 70:
			nachricht_anlegen(
				"nutzer",
				k.name,
				von_admin=True,
				system=True,
				text=f"Neues Projekt passend zu Ihrem Suchprofil ({m['score']} % Passung): {p.titel}. Zum Exposé: /portal/boerse/{p.name}",
			)


# ================================================================ Konto-Verknüpfung


def verknuepfen(konto):
	"""Offene Anfragen mit gleicher (bestätigter) E-Mail dem Konto zuordnen, Rolle ergänzen, Flächen übernehmen."""
	if not cint(konto.email_bestaetigt):
		return 0
	anzahl = 0
	rollen = [r.rolle for r in (konto.get("rollen") or [])]
	neu = []
	for vg in frappe.get_all(
		"Plattform Vorgang",
		filters={"email": konto.email, "nutzer_id": ("is", "not set")},
		fields=["name", "typ", "flaeche_id"],
	):
		frappe.db.set_value("Plattform Vorgang", vg.name, "nutzer_id", konto.name, update_modified=False)
		anzahl += 1
		rolle = ROLLE_FUER_TYP.get(vg.typ)
		if rolle and vg.typ != "kontakt" and rolle not in rollen and rolle not in neu:
			neu.append(rolle)
		if vg.flaeche_id and not frappe.db.get_value("Plattform Flaeche", vg.flaeche_id, "nutzer_id"):
			frappe.db.set_value("Plattform Flaeche", vg.flaeche_id, "nutzer_id", konto.name, update_modified=False)
	if neu:
		for r in neu:
			konto.append("rollen", {"rolle": r})
		konto.save(ignore_permissions=True)
	return anzahl

def _admin_konten_namen():
	return set(frappe.get_all("Plattform Konto Rolle", filters={"rolle": "admin", "parenttype": "Plattform Konto"}, pluck="parent"))


@frappe.whitelist(methods=["POST"])
def zuweisen(art=None, id=None, zustaendigId=None, als_benutzer=None):  # noqa: A002
	n = nutzer(als_benutzer)
	admin_noetig(n)
	if art not in ZUWEISBAR:
		ungueltig("Unbekannte Objektart.")
	o = objekt_laden(art, id)
	if not o:
		nicht_gefunden()
	z = v.kennung(zustaendigId) if zustaendigId else None
	if zustaendigId:
		aktiv = frappe.db.get_value("Plattform Konto", z, "aktiv") if z else None
		v.pflicht(z and cint(aktiv) and z in _admin_konten_namen(), "Zuständig kann nur ein aktives Backoffice-Konto sein.")
	o.zustaendig_id = z or None
	o.save(ignore_permissions=True)
	protokoll(n, "zuweisen", art, o.name, f"Zuständig: {name_von(z) if z else '–'}")
	return {"id": o.name, "zustaendigId": o.zustaendig_id or None}


def _zaehler(doctype, feld):
	return {
		r[feld]: cint(r.anzahl)
		for r in frappe.get_all(doctype, fields=[feld, "count(name) as anzahl"], group_by=feld, limit_page_length=0)
		if r[feld]
	}


@frappe.whitelist(methods=["POST"])
def nutzer_liste(q=None, rolle=None, als_benutzer=None):
	n = nutzer(als_benutzer)
	admin_noetig(n)
	namen = frappe.get_all("Plattform Konto", pluck="name", order_by="creation desc", limit_page_length=0)
	r = v.auswahl(rolle, ALLE_ROLLEN_IDS)
	if r:
		mit_rolle = set(frappe.get_all("Plattform Konto Rolle", filters={"rolle": r, "parenttype": "Plattform Konto"}, pluck="parent"))
		namen = [x for x in namen if x in mit_rolle]
	vorgaenge = _zaehler("Plattform Vorgang", "nutzer_id")
	flaechen = _zaehler("Plattform Flaeche", "nutzer_id")
	projekte = _zaehler("Plattform Projekt", "anbieter_id")
	interessen = _zaehler("Plattform Interesse", "investor_id")
	suche = v.text(q, 120).lower()
	ergebnis = []
	for name in namen:
		k = frappe.get_doc("Plattform Konto", name)
		if suche and suche not in " ".join(x or "" for x in (k.email, k.vorname, k.nachname, k.firma)).lower():
			continue
		ergebnis.append(
			{
				**konto_json(k),
				"zaehler": {
					"vorgaenge": vorgaenge.get(name, 0),
					"flaechen": flaechen.get(name, 0),
					"projekte": projekte.get(name, 0),
					"interessen": interessen.get(name, 0),
				},
			}
		)
	return ergebnis


@frappe.whitelist(methods=["POST"])
def nutzer_aendern(id=None, rollen=None, aktiv=None, als_benutzer=None):  # noqa: A002
	n = nutzer(als_benutzer)
	admin_noetig(n)
	k = v.kennung(id)
	if not k or not frappe.db.exists("Plattform Konto", k):
		nicht_gefunden("Konto nicht gefunden.")
	konto = frappe.get_doc("Plattform Konto", k)
	if isinstance(rollen, list):
		r = v.mehrfach(rollen, ALLE_ROLLEN_IDS)
		v.pflicht(len(r) > 0, "Mindestens eine Rolle.")
		if konto.name == n.id and "admin" not in r:
			konflikt("Die eigene Admin-Rolle kann nicht entfernt werden.")
		konto.set("rollen", [{"rolle": x} for x in r])
	if isinstance(aktiv, bool):
		if konto.name == n.id and not aktiv:
			konflikt("Das eigene Konto kann nicht gesperrt werden.")
		konto.aktiv = 1 if aktiv else 0  # Sperren erhöht die Sitzungsversion (Controller)
	konto.save(ignore_permissions=True)
	protokoll(n, "nutzer", "nutzer", konto.name, f"Rollen: {', '.join(r.rolle for r in konto.rollen)}, aktiv: {'true' if cint(konto.aktiv) else 'false'}")
	return konto_json(konto)

def _status_zaehlen(doctype, feld="status"):
	return {
		r[feld]: cint(r.anzahl)
		for r in frappe.get_all(doctype, fields=[feld, "count(name) as anzahl"], group_by=feld, limit_page_length=0)
		if r[feld]
	}


def _trichter(art, doctype):
	"""Anzahl Objekte, die mindestens die jeweilige Pipeline-Stufe erreicht haben (laut Statusverlauf)."""
	pipe = PROZESSE[art]["pipeline"]
	hoechste = defaultdict(lambda: -1)
	for e in frappe.get_all(
		"Plattform Statusverlauf",
		filters={"parenttype": doctype, "parentfield": "status_verlauf"},
		fields=["parent", "status"],
		limit_page_length=0,
	):
		if e.status in pipe:
			hoechste[e.parent] = max(hoechste[e.parent], pipe.index(e.status))
	return [{"status": s, "anzahl": sum(1 for x in hoechste.values() if x >= i)} for i, s in enumerate(pipe)]

MAX_THREADS = 500


def eigene_bezuege(n):
	"""Kennungen aller Objekte, die dem Nutzer gehören – je Bezugsart."""
	return {
		"vorgang": frappe.get_all("Plattform Vorgang", filters={"nutzer_id": n.id}, pluck="name", limit_page_length=0),
		"flaeche": frappe.get_all("Plattform Flaeche", filters={"nutzer_id": n.id}, pluck="name", limit_page_length=0),
		"projekt": frappe.get_all("Plattform Projekt", filters={"anbieter_id": n.id}, pluck="name", limit_page_length=0),
		"interesse": frappe.get_all("Plattform Interesse", filters={"investor_id": n.id}, pluck="name", limit_page_length=0),
		"nutzer": [n.id],
	}


@frappe.whitelist(methods=["POST"])
def threads(nurUngelesen=None, als_benutzer=None):
	n = nutzer(als_benutzer)
	admin = ist_admin(n)
	werte = {}
	bedingung = ""
	if not admin:
		teile = []
		for i, (art, namen) in enumerate(eigene_bezuege(n).items()):
			if namen:
				teile.append(f"(bezug_art = %(art{i})s and bezug_id in %(ids{i})s)")
				werte[f"art{i}"] = art
				werte[f"ids{i}"] = tuple(namen)
		if not teile:
			return []
		bedingung = "where " + " or ".join(teile)
	gruppen = frappe.db.sql(
		f"""
		select bezug_art, bezug_id, count(*) as anzahl,
			sum(case when gelesen_admin = 0 then 1 else 0 end) as offen_admin,
			sum(case when gelesen_nutzer = 0 then 1 else 0 end) as offen_nutzer,
			max(am) as letzte_am
		from `tabPlattform Nachricht`
		{bedingung}
		group by bezug_art, bezug_id
		order by letzte_am desc
		limit {MAX_THREADS}
		""",
		werte,
		as_dict=True,
	)
	ergebnis = []
	for g in gruppen:
		o = objekt_laden(g.bezug_art, g.bezug_id)
		if not o:
			continue
		eigen = ist_eigen(g.bezug_art, o, n)
		if not admin and not eigen:
			continue
		ungelesen = cint(g.offen_admin) if (admin and not eigen) else cint(g.offen_nutzer)
		if v.wahr(nurUngelesen) and not ungelesen:
			continue
		letzte = frappe.get_all(
			"Plattform Nachricht",
			filters={"bezug_art": g.bezug_art, "bezug_id": g.bezug_id},
			fields=["text", "am", "von_admin", "system"],
			order_by="am desc, creation desc",
			limit_page_length=1,
		)[0]
		ergebnis.append(
			{
				"bezugArt": g.bezug_art,
				"bezugId": g.bezug_id,
				**(thread_info(g.bezug_art, g.bezug_id, n, o) or {}),
				"anzahl": cint(g.anzahl),
				"ungelesen": ungelesen,
				"letzte": {"text": (letzte.text or "")[:160], "am": iso(letzte.am), "vonAdmin": bool(cint(letzte.von_admin)), "system": bool(cint(letzte.system))},
			}
		)
	ergebnis.sort(key=lambda t: t["letzte"]["am"] or "", reverse=True)
	return ergebnis


@frappe.whitelist(methods=["POST"])
def liste(bezugArt=None, bezugId=None, markieren=None, als_benutzer=None):
	n = nutzer(als_benutzer)
	o = darf_bezug(n, bezugArt, bezugId)
	eigen = ist_eigen(bezugArt, o, n)
	als_admin = ist_admin(n) and not eigen
	nachrichten = frappe.get_all(
		"Plattform Nachricht",
		filters={"bezug_art": bezugArt, "bezug_id": o.name},
		fields=["name", "text", "am", "system", "von_admin", "von_id", "gelesen_nutzer", "gelesen_admin"],
		order_by="am asc, creation asc",
		limit_page_length=0,
	)
	feld = "gelesen_admin" if als_admin else "gelesen_nutzer"
	offen = [m.name for m in nachrichten if not cint(m[feld])]
	markieren = True if markieren is None else v.wahr(markieren)
	if markieren and offen:
		frappe.db.sql(f"update `tabPlattform Nachricht` set `{feld}` = 1 where name in %(namen)s", {"namen": tuple(offen)})
	namen_cache = {}

	def von_name(m):
		if cint(m.von_admin):
			return "ÖkoInvest Backoffice"
		if m.von_id not in namen_cache:
			namen_cache[m.von_id] = name_von(m.von_id)
		return namen_cache[m.von_id] or "Nutzer"

	return {
		**(thread_info(bezugArt, o.name, n, o) or {}),
		"bezugArt": bezugArt,
		"bezugId": o.name,
		"nachrichten": [
			{
				"id": m.name,
				"text": m.text,
				"am": iso(m.am),
				"system": bool(cint(m.system)),
				"vonAdmin": bool(cint(m.von_admin)),
				"eigene": m.von_id == n.id and not cint(m.system),
				"vonName": von_name(m),
				"neu": not cint(m[feld]),
			}
			for m in nachrichten
		],
	}


@frappe.whitelist(methods=["POST"])
def senden(bezugArt=None, bezugId=None, text=None, als_benutzer=None):
	n = nutzer(als_benutzer)
	t = v.mehrzeilig(text, 5000)
	v.pflicht(len(t) >= 2, "Bitte geben Sie eine Nachricht ein.")
	o = darf_bezug(n, bezugArt, bezugId)
	eigen = ist_eigen(bezugArt, o, n)
	als_admin = ist_admin(n) and not eigen
	m = nachricht_anlegen(
		bezugArt,

	)
	protokoll(n, "nachricht", bezugArt, o.name, "Nachricht gesendet")
	return {"id": m.name, "am": iso(m.am)}
