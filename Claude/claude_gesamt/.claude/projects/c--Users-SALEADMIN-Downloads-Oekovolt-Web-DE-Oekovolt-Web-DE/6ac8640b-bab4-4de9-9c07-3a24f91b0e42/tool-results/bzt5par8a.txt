    62	async function lesenBereit() {
    63	  const db = await lesen();
    64	  if (db.kyc?.v) return db;
    65	  await aendern(async (d) => {
    66	    if (d.kyc?.v) return;
    67	    d.kyc = leer();
    68	    try {
    69	      await kycSeed(d, { dateiSchreiben, abgleichen: (f) => abgleichAusfuehren(d, f, null, "Einreichung (Demo)") });
    70	    } catch (e) {
    71	      console.warn("[plattform-demo] KYC-Demodaten nicht ergänzt:", e?.message);
    72	    }
    73	  });
    74	  return lesen();
    75	}

    77	async function aendernBereit(fn) {
    78	  await lesenBereit();
    79	  return aendern(fn);

    97	let listeCache = null;
    98	export function sanktionsliste() {
    99	  try {
   100	    if (existsSync(LISTE_DATEI)) {
   101	      const roh = readFileSync(LISTE_DATEI, "utf8");
   102	      if (!listeCache || listeCache.roh !== roh.length) listeCache = { roh: roh.length, liste: JSON.parse(roh) };
   103	      return listeCache.liste;
   104	    }
   105	  } catch (e) {
   106	    console.warn("[plattform-kyc] Sanktionsliste nicht lesbar:", e?.message);
   107	  }
   108	  return { ...DEMO_SANKTIONSLISTE, demo: true };
   109	}

   111	function abgleichAusfuehren(db, f, n, anlass) {
   112	  const liste = sanktionsliste();
   113	  const e = kycEinstellungenDb(db);
   114	  const prueflinge = prueflingeAusFall(f);
   115	  const treffer = prueflinge.flatMap((p) => sanktionsAbgleich(p, liste.eintraege || [], { schwelle: e.trefferSchwelle }).map((t) => ({ ...t, pruefling: p.rolle, prueflingName: p.name })));
   116	  const alt = new Map((f.sanktionen?.treffer || []).map((t) => [`${t.eintragId}|${t.prueflingName}`, t]));
   117	  const neu = treffer.map((t) => {
   118	    const a = alt.get(`${t.eintragId}|${t.prueflingName}`);
   119	    return a?.bewertung ? { ...t, bewertung: a.bewertung, begruendung: a.begruendung, bewertetVon: a.bewertetVon, bewertetAm: a.bewertetAm } : { ...t, bewertung: null };
   120	  });
   121	  const offen = neu.some((t) => !t.bewertung);
   122	  const bestaetigt = neu.some((t) => t.bewertung === "bestaetigt");
   123	  f.sanktionen = { status: bestaetigt ? "bestaetigt" : offen ? "treffer_pruefen" : neu.length ? "ausgeschlossen" : "kein_treffer", treffer: neu, letzterAbgleich: jetzt(), listeStand: liste.stand || "", listeQuelle: liste.quelle || "" };
   124	  const eintrag = { id: neueId("KA"), fallId: f.id, am: jetzt(), anlass, prueflinge: prueflinge.length, treffer: neu.length, listeStand: liste.stand || "", vonId: n?.id || null };
   125	  db.kyc.abgleiche.unshift(eintrag);
   126	  if (db.kyc.abgleiche.length > 2000) db.kyc.abgleiche.length = 2000;
   127	  protokoll(db, n, "kyc_abgleich", "kyc", f.id, `Sanktionsabgleich (${anlass}): ${prueflinge.length} Namen, ${neu.length} Treffer`);
   128	  return f.sanktionen;
   129	}

   160	function statusSetzen(db, f, nach, n, wer, kommentar = "", opt = {}) {
   161	  if (!kycUebergangErlaubt(f.status, nach, wer)) throw konflikt(`Statuswechsel „${kycStatusLabel(f.status, { intern: true })}“ → „${kycStatusLabel(nach, { intern: true })}“ ist nicht möglich.`);
   162	  const von = f.status;
   163	  f.status = nach;
   164	  f.geaendertAm = jetzt();
   165	  verlauf(f, nach, n, wer, kommentar, opt);
   166	  protokoll(db, n, "kyc_status", "kyc", f.id, `${kycStatusLabel(von, { intern: true })} → ${kycStatusLabel(nach, { intern: true })}`);
   167	}

   196	function fallFuerNutzer(db, f) {
   197	  if (!f) return null;
   198	  const sichtStatus = f.status === "zweitpruefung" ? "in_pruefung" : f.status;
   199	  return {
   200	    id: f.id, kontotyp: f.kontotyp, status: sichtStatus, statusLabel: kycStatusLabel(sichtStatus), organisationId: f.organisationId || null,
   201	    investorTyp: f.investorTyp || "", investorVorgaben: kycInvestorVorgaben(f.investorTyp),
   202	    person: f.person || {}, firma: f.firma || {}, ubos: f.ubos || [], vertreter: f.vertreter || [], einreicherVertretungsberechtigt: f.einreicherVertretungsberechtigt ?? null,
   203	    pep: f.pep || {}, mittelherkunft: f.mittelherkunft || {}, steuer: f.steuer || {},
   204	    bank: f.bank ? { iban: f.bank.iban ? ibanFormatiert(f.bank.iban) : "", kontoinhaber: f.bank.kontoinhaber || "", bic: f.bank.bic || "", geprueft: f.status === "verifiziert" && !!f.bank.geprueft } : {},
   205	    nachforderung: f.status === "nachgefordert" ? f.nachforderung || "" : "",
   206	    eingereichtAm: f.eingereichtAm || null, verifiziertAm: f.verifiziertAm || null, naechstePruefung: f.naechstePruefung || null,
   207	    verlauf: (f.verlauf || []).filter((x) => x.oeffentlich !== false).map((x) => ({ status: x.status === "zweitpruefung" ? "in_pruefung" : x.status, am: x.am, kommentar: x.rolle === "nutzer" ? "" : x.kommentar })),
   208	    bearbeitbar: KYC_BEARBEITBAR.includes(f.status),
   209	    dokumente: dokumenteVon(db, f).map((d) => dokumentMeta(d)),
   210	  };
   211	}

   223	/** Eigener Fall mit Anforderungen, Fortschritt und Freigaben */
   224	export async function kycFall(n) {
   225	  angemeldet(n);
   226	  const db = await lesenBereit();
   227	  const f = eigenerFall(db, n);
   228	  const status = kycStatusDb(db, { kontoId: n.id, organisationId: n.organisation?.id || null });
   229	  const kontext = { rollen: n.rollen || [], flaechen: flaechenVon(db, n.id), einstellungen: kycEinstellungenDb(db) };
   230	  const anforderungen = kycAnforderungen(f ? { ...f, dokumente: dokumenteVon(db, f) } : null, kontext);
   231	  return {
   232	    fall: fallFuerNutzer(db, f),
   233	    anforderungen,
   234	    fortschritt: kycFortschritt(anforderungen),
   235	    status: { ...status, statusLabel: kycStatusLabel(status.status === "zweitpruefung" ? "in_pruefung" : status.status), status: status.status === "zweitpruefung" ? "in_pruefung" : status.status, risiko: undefined },
   236	    flaechen: kontext.flaechen,
   237	    freigaben: freigabenFuer(db, n.id, status),
   238	    investorVorgaben: kycInvestorVorgaben(f?.investorTyp || investorTypVon(db, n)), // Vorschlag Kontotyp/Unterlagen aus dem Suchprofil
   239	    ablage: kycSchluesselStatus(),
   240	  };
   241	}

   252	function listeBereinigen(liste, fn, max = 12) {
   253	  return (Array.isArray(liste) ? liste : []).slice(0, max).map((x) => ({ id: /^[a-z0-9-]{4,20}$/i.test(String(x?.id || "")) ? String(x.id) : neueId("WB").toLowerCase(), ...fn(x || {}) }));
   254	}

   256	const ABSCHNITTE = {
   257	  kontotyp: (d, f, n, db) => {
   258	    // ohne Angabe: Vorschlag aus dem Investorentyp des Suchprofils
   259	    const typ = v.auswahl(d.kontotyp, Object.keys(KYC_KONTOTYPEN), "") || kycInvestorVorgaben(f.investorTyp)?.kontotyp || "";
   260	    v.pflicht(typ, "Bitte wählen Sie den Kontotyp.");

   262	    if (typ === "unternehmen" && n.organisation?.id && ["inhaber", "admin"].includes(n.organisation.rolle) && !(db.kyc.faelle || []).some((x) => x.organisationId === n.organisation.id && x.id !== f.id)) {
   263	      f.organisationId = n.organisation.id;
   264	      f.firma = { ...(f.firma || {}), name: f.firma?.name || n.organisation.name };
   265	    }
   266	    if (typ === "privat") f.organisationId = null;
   267	  },
   268	  person: (d, f) => {
   269	    f.person = personDaten(d);
   270	  },
   271	  firma: (d, f) => {
   272	    f.firma = {
   273	      name: v.text(d.name, 160), rechtsform: v.text(d.rechtsform, 60), registernummer: v.text(d.registernummer, 60), registergericht: v.text(d.registergericht, 120),
   274	      land: land(d.land), sitz: v.text(d.sitz, 160), mehrstufig: v.wahr(d.mehrstufig), treuhand: v.wahr(d.treuhand),
   275	    };
   276	    if (d.einreicherVertretungsberechtigt !== undefined) f.einreicherVertretungsberechtigt = v.wahr(d.einreicherVertretungsberechtigt);
   277	  },
   278	  ubos: (d, f) => {
   279	    f.ubos = listeBereinigen(d.liste, (u) => ({
   280	      vorname: v.text(u.vorname, 80), nachname: v.text(u.nachname, 80), geburtsdatum: datum(u.geburtsdatum), staatsangehoerigkeit: land(u.staatsangehoerigkeit), wohnsitzland: land(u.wohnsitzland),
   281	      anteil: v.zahl(u.anteil, { min: 0, max: 100, stellen: 2 }), kontrolleArt: v.auswahl(u.kontrolleArt, Object.keys(KYC_KONTROLLE_ARTEN), "anteil"), pep: v.wahr(u.pep),
   282	    }), 10);
   283	  },
   284	  vertreter: (d, f) => {
   285	    f.vertreter = listeBereinigen(d.liste, (x) => ({ vorname: v.text(x.vorname, 80), nachname: v.text(x.nachname, 80), geburtsdatum: datum(x.geburtsdatum), funktion: v.text(x.funktion, 80), istEinreicher: v.wahr(x.istEinreicher) }), 8);
   286	  },
   287	  pep: (d, f) => {
   288	    f.pep = { istPep: d.istPep === undefined ? undefined : v.wahr(d.istPep), naheStehend: d.naheStehend === undefined ? undefined : v.wahr(d.naheStehend), funktion: v.text(d.funktion, 200), angaben: v.mehrzeilig(d.angaben, 1000) };
   289	  },
   290	  mittelherkunft: (d, f) => {
   291	    f.mittelherkunft = { ticket: v.auswahl(d.ticket, Object.keys(KYC_TICKETS), ""), quellen: v.mehrfach(d.quellen, Object.keys(KYC_MITTEL_QUELLEN)), text: v.mehrzeilig(d.text, 1500) };
   292	  },
   293	  steuer: (d, f) => {
   294	    f.steuer = { land: land(d.land), steuerId: v.text(d.steuerId, 40) };
   295	  },
   296	  bank: (d, f) => {
   297	    const ib = ibanPruefen(d.iban);
   298	    if (d.iban && !ib.ok) throw ungueltig(ib.fehler);
   299	    const inhaber = v.text(d.kontoinhaber, 160);
   300	    f.bank = { iban: ib.ok ? ib.iban : "", bic: v.text(d.bic, 11).toUpperCase(), kontoinhaber: inhaber, ibanOk: ib.ok, geprueft: false, geaendertAm: jetzt() };
   301	  },
   302	};

   304	/** Abschnitt speichern (nur solange bearbeitbar). Legt den Fall beim ersten Aufruf an. */
   305	export async function kycSpeichern(n, { abschnitt, daten = {} } = {}) {
   306	  angemeldet(n);
   307	  if (!ABSCHNITTE[abschnitt]) throw ungueltig("Unbekannter Abschnitt.");
   308	  return aendernBereit((db) => {
   309	    let f = eigenerFall(db, n);
   310	    if (!f) {
   311	      if (abschnitt !== "kontotyp") throw ungueltig("Bitte wählen Sie zuerst den Kontotyp.");
   312	      f = { id: neueId("KY"), kontoId: n.id, organisationId: null, kontotyp: "", investorTyp: investorTypVon(db, n), status: "offen", verlauf: [], erstelltAm: jetzt(), geaendertAm: jetzt() };
   313	      verlauf(f, "offen", n, "nutzer", f.investorTyp ? `Verifizierung begonnen (Investorentyp: ${kycInvestorVorgaben(f.investorTyp).label}).` : "Verifizierung begonnen.");
   314	      db.kyc.faelle.push(f);
   315	    }
   316	    if (!KYC_BEARBEITBAR.includes(f.status)) throw konflikt(f.status === "verifiziert" ? "Ihre Verifizierung ist abgeschlossen. Für Änderungen starten Sie bitte eine erneute Prüfung." : "Ihre Angaben liegen gerade zur Prüfung vor und können nicht geändert werden.");
   317	    f.investorTyp = investorTypVon(db, n) || f.investorTyp || ""; // Suchprofil kann sich geändert haben
   318	    ABSCHNITTE[abschnitt](daten || {}, f, n, db);
   319	    f.geaendertAm = jetzt();
   320	    protokoll(db, n, "kyc_angaben", "kyc", f.id, `Angaben gespeichert: ${abschnitt}`);
   321	    return kycFallSync(db, n, f);
   322	  });

   333	  angemeldet(n);

   496	export async function kycWarteschlange(n, { status = "offen", q = "" } = {}) {
   497	  angemeldet(n);
   498	  if (!complianceLesen(n)) throw verboten("Nur für die Compliance.");
   499	  const db = await lesenBereit();
   500	  const alle = (db.kyc.faelle || []).filter((f) => f.status !== "geloescht");
   501	  let liste = alle;
   502	  if (status === "offen") liste = alle.filter((f) => KYC_WARTESCHLANGE.includes(f.status));
   503	  else if (KYC_STATUS[status]) liste = alle.filter((f) => f.status === status);
   504	  else if (status === "treffer") liste = alle.filter((f) => ["treffer_pruefen", "bestaetigt"].includes(f.sanktionen?.status));
   505	  else if (status === "faellig") liste = alle.filter((f) => f.status === "verifiziert" && f.naechstePruefung && tageBis(f.naechstePruefung) <= 30);
   506	  const suche = v.text(q, 80).toLowerCase();
   507	  let zeilen = liste.map((f) => fallZeile(db, f));
   508	  if (suche) zeilen = zeilen.filter((z) => `${z.id} ${z.bezeichnung} ${z.konto?.email || ""} ${z.organisation?.name || ""}`.toLowerCase().includes(suche));
   509	  const rang = { zweitpruefung: 0, eingereicht: 1, in_pruefung: 2 };
   510	  zeilen.sort((a, b) => (rang[a.status] ?? 5) - (rang[b.status] ?? 5) || (b.risiko === "hoch") - (a.risiko === "hoch") || String(a.eingereichtAm || "").localeCompare(String(b.eingereichtAm || "")));
   511	  const liste2 = sanktionsliste();
   512	  return {
   513	    faelle: complianceLesen(n) && !istCompliance(n) ? zeilen.map((z) => ({ ...z, konto: z.konto ? { name: z.konto.name, rollen: z.konto.rollen } : null })) : zeilen,
   514	    kennzahlen: {
   515	      warteschlange: alle.filter((f) => KYC_WARTESCHLANGE.includes(f.status)).length,
   516	      zweitpruefung: alle.filter((f) => f.status === "zweitpruefung").length,
   517	      treffer: alle.filter((f) => f.sanktionen?.status === "treffer_pruefen").length,
   518	      hoch: alle.filter((f) => f.risiko?.stufe === "hoch" && f.status !== "abgelehnt").length,
   519	      verifiziert: alle.filter((f) => f.status === "verifiziert").length,
   520	      faellig: alle.filter((f) => f.status === "verifiziert" && f.naechstePruefung && tageBis(f.naechstePruefung) <= 30).length,
   521	      nachgefordert: alle.filter((f) => f.status === "nachgefordert").length,
   522	    },
   523	    liste: { quelle: liste2.quelle, stand: liste2.stand, anzahl: (liste2.eintraege || []).length, demo: !!liste2.demo },
   524	    schreiben: istCompliance(n),
   525	  };
   526	}

   578	      if (f.status === "eingereicht") statusSetzen(db, f, "in_pruefung", n, "compliance", "Prüfung begonnen.");
   579	      else protokoll(db, n, "kyc_bearbeiter", "kyc", f.id, "Fall übernommen");
   580	    } else if (aktion === "pruefliste") {
   581	      if (!KYC_WARTESCHLANGE.includes(f.status)) throw konflikt("Die Prüfliste kann nur während der Prüfung bearbeitet werden.");
   582	      const d = dokumenteVon(db, f).find((x) => x.id === v.kennung(daten.dokumentId));
   583	      if (!d) throw nichtGefunden("Unterlage nicht gefunden.");
   584	      const punkte = {};
   585	      for (const p of KYC_DOKUMENTE[d.art]?.pruefpunkte || []) punkte[p] = v.wahr(daten.punkte?.[p]);
   586	      const ergebnis = v.auswahl(daten.ergebnis, ["ok", "abgelehnt", "neu"], "neu");
   587	      const notiz = v.mehrzeilig(daten.notiz, 600);
   588	      if (ergebnis === "ok") v.pflicht(Object.values(punkte).every(Boolean), `Für „geprüft“ müssen alle Prüfpunkte bestätigt sein: ${(KYC_DOKUMENTE[d.art]?.pruefpunkte || []).map((p) => KYC_PRUEFPUNKTE[p]).join(" · ")}`);
   589	      if (ergebnis === "abgelehnt") v.pflicht(notiz.length >= 5, "Bitte begründen Sie die Ablehnung der Unterlage (wird dem Nutzer angezeigt).");
   590	      d.pruefung = { punkte, notiz, vonId: n.id, am: jetzt() };
   591	      d.status = ergebnis;
   592	      if (f.status === "eingereicht") statusSetzen(db, f, "in_pruefung", n, "compliance", "Prüfung begonnen.");
   593	      protokoll(db, n, "kyc_pruefliste", "kyc", f.id, `Unterlage ${d.id}: ${ergebnis}`);
   594	    } else if (aktion === "risiko") {
   595	      const st = v.auswahl(daten.stufe, ["niedrig", "mittel", "hoch"], "");
   596	      const begr = v.mehrzeilig(daten.begruendung, 600);
   597	      v.pflicht(st, "Bitte Risikostufe wählen.");
   598	      v.pflicht(begr.length >= 5, "Bitte begründen Sie die manuelle Einstufung.");
   599	      const regel = kycRisiko(f, e);
   600	      f.risiko = { ...regel, stufe: st, manuell: true, begruendung: begr, regelStufe: regel.stufe, vonId: n.id, am: jetzt() };
   601	      protokoll(db, n, "kyc_risiko", "kyc", f.id, `Risiko manuell: ${st} (Regel: ${regel.stufe})`);
   602	    } else if (aktion === "abgleich") {
   603	      const s = abgleichAusfuehren(db, f, n, "manuell");
   604	      if (!f.risiko?.manuell) f.risiko = { ...kycRisiko(f, e), manuell: false };
   605	      if (f.status === "verifiziert" && s.status === "treffer_pruefen") {
   606	        f.status = "in_pruefung";
   607	        verlauf(f, "in_pruefung", n, "compliance", "Erneute Prüfung nach Sanktionsabgleich.", { oeffentlich: false });
   608	      }
   609	      hinweis = s.treffer.length ? `${s.treffer.length} möglicher Treffer – manuelle Prüfung erforderlich.` : "Kein Treffer.";
   610	    } else if (aktion === "treffer") {
   611	      const t = (f.sanktionen?.treffer || []).find((x) => x.eintragId === String(daten.eintragId || "") && (!daten.prueflingName || x.prueflingName === daten.prueflingName));
   612	      if (!t) throw nichtGefunden("Treffer nicht gefunden.");
   613	      const bew = v.auswahl(daten.bewertung, ["ausgeschlossen", "bestaetigt"], "");
   614	      const begr = v.mehrzeilig(daten.begruendung, 1000);
   615	      v.pflicht(bew, "Bitte bewerten Sie den Treffer.");
   616	      v.pflicht(begr.length >= 10, "Bitte dokumentieren Sie die Begründung (mind. 10 Zeichen).");
   617	      Object.assign(t, { bewertung: bew, begruendung: begr, bewertetVon: n.id, bewertetAm: jetzt() });
   618	      const liste = f.sanktionen.treffer;
   619	      f.sanktionen.status = liste.some((x) => x.bewertung === "bestaetigt") ? "bestaetigt" : liste.some((x) => !x.bewertung) ? "treffer_pruefen" : "ausgeschlossen";
   620	      if (!f.risiko?.manuell) f.risiko = { ...kycRisiko(f, e), manuell: false };
   621	      protokoll(db, n, "kyc_treffer", "kyc", f.id, `Sanktionstreffer ${t.eintragId} (${t.prueflingName}): ${bew}`);
   622	      if (bew === "bestaetigt") hinweis = "Treffer bestätigt: Geldwäschebeauftragten informieren, keine Mittel bereitstellen, Meldepflicht prüfen (§ 43 GwG). Den Kunden nicht informieren (§ 47 GwG).";
   623	    } else if (aktion === "entscheiden" || aktion === "zweitfreigabe") {
   624	      const ergebnis = v.auswahl(daten.ergebnis, ERGEBNISSE, "");
   625	      const grund = v.mehrzeilig(daten.grund, 1500);
   626	      v.pflicht(ergebnis, "Bitte wählen Sie ein Ergebnis.");
   627	      if (aktion === "zweitfreigabe") {
   628	        if (f.status !== "zweitpruefung") throw konflikt("Der Fall wartet nicht auf eine Zweitfreigabe.");
   629	        if (f.pruefung.erstPrueferId === n.id) throw konflikt("Vier-Augen-Prinzip: Die Zweitfreigabe muss eine andere Person erteilen.");
   630	      } else if (!["eingereicht", "in_pruefung"].includes(f.status)) throw konflikt("Der Fall ist nicht in Prüfung.");
   631	      if (ergebnis === "nachgefordert") v.pflicht(grund.length >= 5, "Bitte nennen Sie, was nachzureichen ist (wird dem Nutzer angezeigt).");
   632	      if (ergebnis === "abgelehnt") v.pflicht(grund.length >= 5, "Bitte begründen Sie die Ablehnung (intern).");
   633	      if (ergebnis === "verifiziert") {
   634	        if (["treffer_pruefen", "bestaetigt"].includes(f.sanktionen?.status)) throw konflikt("Offener oder bestätigter Sanktionstreffer – eine Verifizierung ist nicht möglich.");
   635	        const ungeprueft = dokumenteVon(db, f).filter((d) => d.status === "neu");
   636	        if (ungeprueft.length) throw konflikt(`Bitte zuerst alle Unterlagen prüfen (Prüfliste): ${ungeprueft.map((d) => KYC_DOKUMENTE[d.art]?.label || d.art).join(", ")}`);
   637	        const fp = kycFortschritt(anforderungenVon(db, f, { nurGeprueft: true }));
   638	        if (fp.fehlend.length) throw konflikt(`Pflichtangaben/-unterlagen fehlen oder sind nicht geprüft: ${fp.fehlend.map((x) => x.label).join(" · ")}`);
   639	      }
   640	      const vierAugen = vierAugenNoetig(f, e);
   641	      if (ergebnis === "verifiziert" && aktion === "entscheiden" && vierAugen) {
   642	        f.pruefung.erstPrueferId = n.id;
   643	        f.pruefung.erstAm = jetzt();
   644	        f.pruefung.erstGrund = grund;
   645	        statusSetzen(db, f, "zweitpruefung", n, "compliance", "Erstprüfung positiv – Zweitfreigabe (Vier-Augen) erforderlich.", { oeffentlich: false });
   646	        complianceMelden(db, f, `Zweitfreigabe nötig: ${f.id}`, "Hohes Risiko – bitte als zweite Person prüfen.", { ohneId: n.id });
   647	        hinweis = "Zweitfreigabe angefordert (Vier-Augen-Prinzip).";
   648	      } else {
   649	        if (aktion === "zweitfreigabe") {
   650	          f.pruefung.zweitPrueferId = n.id;
   651	          f.pruefung.zweitAm = jetzt();
   652	        } else {
   653	          f.pruefung.erstPrueferId = n.id;
   654	          f.pruefung.erstAm = jetzt();
   655	        }
   656	        f.entscheidung = { ergebnis, grund, vonId: n.id, am: jetzt() };
   657	        if (ergebnis === "verifiziert") {
   658	          statusSetzen(db, f, "verifiziert", n, "compliance", "Verifizierung abgeschlossen.");
   659	          f.verifiziertAm = jetzt();
   660	          f.naechstePruefung = monateSpaeter(heuteIso(), e.wiederholungMonate?.[f.risiko?.stufe || "mittel"] || 24);
   661	          if (f.bank?.iban) {
   662	            f.bank.ibanOk = ibanPruefen(f.bank.iban).ok;
   663	            f.bank.inhaberPasst = kontoinhaberPasst(f.bank.kontoinhaber, f.kontotyp === "unternehmen" ? { firma: f.firma?.name } : { vorname: f.person?.vorname, nachname: f.person?.nachname });
   664	            f.bank.geprueft = f.bank.ibanOk && f.bank.inhaberPasst;
   665	          }
   666	          f.erinnerungen = {};
   667	          nutzerMelden(db, f, "Verifizierung abgeschlossen", "Ihre Identität ist bestätigt. Alle Funktionen stehen Ihnen offen.");
   668	        } else if (ergebnis === "nachgefordert") {
   669	          f.nachforderung = grund;
   670	          statusSetzen(db, f, "nachgefordert", n, "compliance", grund);
   671	          nutzerMelden(db, f, "Bitte Unterlagen ergänzen", "Für Ihre Verifizierung benötigen wir noch Angaben oder Unterlagen.");
   672	        } else {
   673	          statusSetzen(db, f, "abgelehnt", n, "compliance", "Die Verifizierung konnte nicht abgeschlossen werden.");
   674	          f.pruefung.ablehnGrund = grund;
   675	          f.loeschenAb = loeschenAb(heuteIso(), e.aufbewahrungJahre || 5);
   676	          nutzerMelden(db, f, "Verifizierung nicht abgeschlossen", "Ihre Verifizierung konnte nicht abgeschlossen werden. Bei Fragen schreiben Sie uns über „Nachrichten“.");
   677	        }
   678	      }
   679	    } else if (aktion === "beziehungEnde") {
   680	      const d = datum(daten.datum);
   681	      v.pflicht(d, "Bitte geben Sie das Ende der Geschäftsbeziehung an (JJJJ-MM-TT).");
   682	      f.beziehungEnde = d;
   683	      f.loeschenAb = loeschenAb(d, e.aufbewahrungJahre || 5);
   684	      protokoll(db, n, "kyc_frist", "kyc", f.id, `Geschäftsbeziehung beendet am ${d}, Löschung ab ${f.loeschenAb}`);
   685	    } else if (aktion === "notiz") {
   686	      const t = v.mehrzeilig(daten.text, 1000);
   687	      v.pflicht(t.length >= 2, "Bitte Notiz eingeben.");
   688	      verlauf(f, f.status, n, "compliance", t, { oeffentlich: false });
   689	      protokoll(db, n, "kyc_notiz", "kyc", f.id, "Interne Notiz");
   690	    }
   691	    f.geaendertAm = jetzt();
   692	    return { ok: true, status: f.status, hinweis };
   693	  });
   694	}

   713	    if (Array.isArray(daten.hochrisikoLaender)) e.hochrisikoLaender = [...new Set(daten.hochrisikoLaender.map(land).filter(Boolean))].slice(0, 80);

   746	/**
   747	 * Fristen-Lauf (täglich; Demo: manuell im Backoffice bzw. Test): Ausweis-Ablauf erinnern, Wiederholungsprüfung,
   748	 * Sanktions-Neuabgleich bei neuem Listenstand, Löschung nach Aufbewahrungsfrist. `stichtag` nur für Tests/Vorschau.
   749	 */
   750	export async function kycFristen(n, { stichtag = null, fallId = null } = {}) {
   751	  complianceNoetig(n);
   752	  const zuLoeschen = [];
   753	  const r = await aendernBereit((db) => {
   754	    const e = kycEinstellungenDb(db);
   755	    const heute = datum(stichtag) || heuteIso();
   756	    const tage = (d) => (d ? Math.floor((new Date(`${d}T00:00:00Z`) - new Date(`${heute}T00:00:00Z`)) / 86400000) : null);
   757	    const liste = sanktionsliste();
   758	    const erg = { ausweisErinnert: 0, wiederholungErinnert: 0, abgelaufen: 0, neuAbgeglichen: 0, neueTreffer: 0, geloescht: 0 };
   759	    for (const f of db.kyc.faelle) {
   760	      if (f.status === "geloescht" || (fallId && f.id !== fallId)) continue;
   761	      f.erinnerungen = f.erinnerungen || {};
   762	      const gb = f.person?.ausweis?.gueltigBis;
   763	      const t = tage(gb);
   764	      if (f.status === "verifiziert" && t !== null && t <= (e.erinnerungAusweisTage || 30) && f.erinnerungen.ausweis !== gb) {
   765	        f.erinnerungen.ausweis = gb;
   766	        erg.ausweisErinnert++;
   767	        nutzerMelden(db, f, t < 0 ? "Ausweis abgelaufen" : "Ausweis läuft bald ab", "Bitte aktualisieren Sie Ihre Ausweisdaten unter „Verifizierung“ (erneute Prüfung).");
   768	        protokoll(db, n, "kyc_frist", "kyc", f.id, `Erinnerung Ausweis-Ablauf (${gb})`);
   769	      }
   770	      const w = tage(f.naechstePruefung);
   771	      if (f.status === "verifiziert" && w !== null) {
   772	        if (w <= 0) {
   773	          f.status = "abgelaufen";
   774	          verlauf(f, "abgelaufen", null, "system", "Wiederkehrende Überprüfung fällig.");
   775	          erg.abgelaufen++;
   776	          nutzerMelden(db, f, "Verifizierung erneuern", "Ihre regelmäßige Überprüfung ist fällig. Bitte bestätigen bzw. aktualisieren Sie Ihre Angaben.");
   777	          protokoll(db, n, "kyc_frist", "kyc", f.id, "Wiederholungsprüfung fällig → abgelaufen");
   778	        } else if (w <= 30 && f.erinnerungen.wiederholung !== f.naechstePruefung) {
   779	          f.erinnerungen.wiederholung = f.naechstePruefung;
   780	          erg.wiederholungErinnert++;
   781	          nutzerMelden(db, f, "Überprüfung Ihrer Angaben steht an", "In den nächsten Wochen ist die regelmäßige Überprüfung fällig.");
   782	        }
   783	      }
   784	      if (["verifiziert", "in_pruefung", "eingereicht", "zweitpruefung"].includes(f.status) && liste.stand && f.sanktionen?.listeStand !== liste.stand) {
   785	        const vorher = (f.sanktionen?.treffer || []).length;
   786	        const s = abgleichAusfuehren(db, f, null, "Neuer Listenstand");
   787	        erg.neuAbgeglichen++;
   788	        if (s.status === "treffer_pruefen" && s.treffer.length > vorher) {
   789	          erg.neueTreffer++;
   790	          if (f.status === "verifiziert") {
   791	            f.status = "in_pruefung";
   792	            verlauf(f, "in_pruefung", null, "system", "Möglicher Treffer bei Neuabgleich.", { oeffentlich: false });
   793	          }
   794	          complianceMelden(db, f, `Sanktionsabgleich: möglicher Treffer ${f.id}`, "Manuelle Prüfung erforderlich.");
   795	        }
   796	      }
   797	      if (f.loeschenAb && f.loeschenAb <= heute) {
   798	        for (const d of db.kyc.dokumente.filter((x) => x.fallId === f.id && !x.geloescht)) {
   799	          d.geloescht = true;
   800	          d.geloeschtAm = jetzt();
   801	          d.name = "(gelöscht)";
   802	          zuLoeschen.push(d.datei);
   803	        }
   804	        const rest = { id: f.id, kontoId: null, organisationId: null, kontotyp: f.kontotyp, status: "geloescht", verlauf: [], geloeschtAm: jetzt(), loeschenAb: f.loeschenAb, erstelltAm: f.erstelltAm };
   805	        for (const k of Object.keys(f)) delete f[k];
   806	        Object.assign(f, rest);
   807	        erg.geloescht++;
   808	        protokoll(db, n, "kyc_loeschung", "kyc", f.id, "Aufbewahrungsfrist abgelaufen – Unterlagen und personenbezogene Angaben gelöscht");
   809	      }
   810	    }
   811	    return { ok: true, stichtag: heute, ...erg };
   812	  });
   813	  for (const d of zuLoeschen) await dateiLoeschen(d);
   814	  return r;
   815	}
