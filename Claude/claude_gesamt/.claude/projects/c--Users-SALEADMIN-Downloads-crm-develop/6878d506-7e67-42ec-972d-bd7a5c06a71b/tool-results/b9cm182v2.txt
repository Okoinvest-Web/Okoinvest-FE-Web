"use client";

import Link from "next/link";
import { useOptimistic, useState, useTransition } from "react";
import { setOpportunityStageAction } from "@/lib/actions";
import { money, date, STAGE_LABEL } from "@/lib/format";
import type { Opportunity } from "@/lib/types";

// Chancen per Drag & Drop zwischen Vertriebsphasen verschieben.
export function Kanban({ stages, items }: { stages: readonly string[]; items: Opportunity[] }) {
  const [, start] = useTransition();
  const [over, setOver] = useState<string | null>(null);
  const [opps, move] = useOptimistic(items, (state, { name, stage }: { name: string; stage: string }) =>
    state.map((o) => (o.name === name ? { ...o, sales_stage: stage } : o)),
  );

  function drop(stage: string, name: string) {
    setOver(null);
    const current = opps.find((o) => o.name === name);
    if (!current || current.sales_stage === stage) return;
    start(async () => {
      move({ name, stage });
      await setOpportunityStageAction(name, stage);
    });
  }

  return (
    <div className="flex min-h-0 flex-1 gap-3 overflow-x-auto p-5">
      {stages.map((stage) => {
        const col = opps.filter((o) => o.sales_stage === stage);
        const sum = col.reduce((s, o) => s + o.opportunity_amount, 0);
        return (
          <section
            key={stage}
            onDragOver={(e) => {
              e.preventDefault();
              setOver(stage);
            }}
            onDragLeave={() => setOver((s) => (s === stage ? null : s))}
            onDrop={(e) => drop(stage, e.dataTransfer.getData("text/plain"))}
            className={`flex w-72 shrink-0 flex-col rounded-lg p-2 transition-colors ${over === stage ? "bg-gray-200" : "bg-gray-50"}`}
          >
            <div className="flex items-baseline justify-between px-1.5 pb-2">
              <h2 className="font-medium text-gray-900">
                {STAGE_LABEL[stage] ?? stage} <span className="ml-1 text-gray-400">{col.length}</span>
              </h2>
              <span className="text-xs text-gray-500">{money(sum)}</span>
            </div>
            <div className="flex flex-col gap-2">
              {col.map((o) => (
                <Link
                  key={o.name}
                  href={`/opportunities/${encodeURIComponent(o.name)}`}
                  draggable
                  onDragStart={(e) => e.dataTransfer.setData("text/plain", o.name)}
                  className="cursor-grab rounded-md border border-gray-200 bg-white p-3 shadow-sm hover:border-gray-300 active:cursor-grabbing"
                >
                  <div className="font-medium text-gray-900">{o.title ?? o.name}</div>
                  <div className="mt-0.5 text-xs text-gray-500">{o.customer_name ?? o.party_name}</div>
                  <div className="mt-3 flex items-center justify-between text-xs">
                    <span className="font-medium text-gray-900">{money(o.opportunity_amount, o.currency)}</span>
                    <span className="text-gray-500">
                      {o.probability} % · {date(o.expected_closing)}
                    </span>
                  </div>
                </Link>
              ))}
            </div>
          </section>
        );
      })}
    </div>
  );
}

"use client";

import { usePathname, useRouter, useSearchParams } from "next/navigation";
import { Search } from "lucide-react";
import { useEffect, useState, useTransition } from "react";

// Suche und Statusfilter landen in der URL (?q=…&status=…), damit Listen teilbar sind.
export function ListToolbar({ statuses, labels, count }: { statuses?: readonly string[]; labels?: Record<string, string>; count: number }) {
  const router = useRouter();
  const pathname = usePathname();
  const params = useSearchParams();
  const [q, setQ] = useState(params.get("q") ?? "");
  const [pending, start] = useTransition();

  function push(next: Record<string, string>) {
    const sp = new URLSearchParams(params.toString());
    for (const [k, v] of Object.entries(next)) {
      if (v) sp.set(k, v);
      else sp.delete(k);
    }
    start(() => router.replace(`${pathname}?${sp}`));
  }

  useEffect(() => {
    if (q === (params.get("q") ?? "")) return;
    const t = setTimeout(() => push({ q }), 250);
    return () => clearTimeout(t);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [q]);

  return (
    <div className="flex items-center gap-2 border-b border-gray-200 px-5 py-2.5">
      <label className="relative">
        <Search className="pointer-events-none absolute left-2.5 top-1/2 size-3.5 -translate-y-1/2 text-gray-400" />
        <input
          value={q}
          onChange={(e) => setQ(e.target.value)}
          placeholder="Suchen…"
          className="h-8 w-64 rounded-md bg-gray-100 pl-8 pr-3 outline-none placeholder:text-gray-400 focus:bg-white focus:ring-1 focus:ring-gray-300"
        />
      </label>
      {statuses && (
        <select
          value={params.get("status") ?? ""}
          onChange={(e) => push({ status: e.target.value })}
          className="h-8 rounded-md bg-gray-100 px-2 outline-none focus:ring-1 focus:ring-gray-300"
        >
          <option value="">Alle Status</option>
          {statuses.map((s) => (
            <option key={s} value={s}>
              {labels?.[s] ?? s}
            </option>
          ))}
        </select>
      )}
      <span className={`ml-auto text-xs text-gray-500 ${pending ? "opacity-50" : ""}`}>{count} Einträge</span>
    </div>
  );
}

import { useRouter } from "next/navigation";
import { useEffect, useRef, useState } from "react";
import { Building2, Contact, Handshake, Search, Users } from "lucide-react";
import type { SearchHit } from "@/lib/types";

const ICON = { Lead: Users, Chance: Handshake, Kunde: Building2, Kontakt: Contact };

// Globale Suche: Strg+K / ⌘K öffnet, Pfeiltasten wählen, Enter öffnet.
export function CommandPalette() {
  const router = useRouter();
  const [open, setOpen] = useState(false);
  const [q, setQ] = useState("");
  const [hits, setHits] = useState<SearchHit[]>([]);
  const [active, setActive] = useState(0);
  const [loading, setLoading] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") {
        e.preventDefault();
        setOpen((o) => !o);
      }
      if (e.key === "Escape") setOpen(false);
    }
    function onOpen() {
      setOpen(true);
    }
    window.addEventListener("keydown", onKey);
    window.addEventListener("open-search", onOpen);
    return () => {
      window.removeEventListener("keydown", onKey);
      window.removeEventListener("open-search", onOpen);
    };
  }, []);

  useEffect(() => {
    if (open) setTimeout(() => inputRef.current?.focus(), 0);
    else {
      setQ("");
      setHits([]);
    }
  }, [open]);

  useEffect(() => {
    if (q.trim().length < 2) {
      setHits([]);
      return;
    }
    const ctrl = new AbortController();
    const t = setTimeout(async () => {
      setLoading(true);
      try {
        const res = await fetch(`/api/search?q=${encodeURIComponent(q)}`, { signal: ctrl.signal });
        if (res.ok) {
          setHits(((await res.json()) as { hits: SearchHit[] }).hits);
          setActive(0);
        }
      } catch {
        /* abgebrochen */
      } finally {
        setLoading(false);
      }
    }, 150);
    return () => {
      clearTimeout(t);
      ctrl.abort();
    };
  }, [q]);

  function go(hit: SearchHit | undefined) {
    if (!hit) return;
    setOpen(false);
    router.push(hit.href);
  }

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-start justify-center bg-black/20 p-4 pt-[12vh]" onMouseDown={() => setOpen(false)}>
      <div className="w-full max-w-lg overflow-hidden rounded-xl border border-gray-200 bg-white shadow-2xl" onMouseDown={(e) => e.stopPropagation()}>
        <div className="flex items-center gap-2 border-b border-gray-200 px-4">
          <Search className="size-4 text-gray-400" />
          <input
            ref={inputRef}
            value={q}
            onChange={(e) => setQ(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === "ArrowDown") {
                e.preventDefault();
                setActive((a) => Math.min(a + 1, hits.length - 1));
              } else if (e.key === "ArrowUp") {
                e.preventDefault();
                setActive((a) => Math.max(a - 1, 0));
              } else if (e.key === "Enter") {
                go(hits[active]);
              }
            }}
            placeholder="Leads, Chancen, Kunden, Kontakte suchen…"
            className="h-12 flex-1 outline-none placeholder:text-gray-400"
          />
          <kbd className="rounded border border-gray-200 px-1.5 text-[10px] text-gray-500">Esc</kbd>
        </div>
        <ul className="max-h-80 overflow-y-auto py-1">
          {hits.map((h, i) => {
            const Icon = ICON[h.type];
            return (
              <li key={h.href + i}>
                <button
                  onMouseEnter={() => setActive(i)}
                  onClick={() => go(h)}
                  className={`flex w-full items-center gap-3 px-4 py-2 text-left ${i === active ? "bg-gray-100" : ""}`}
                >
                  <Icon className="size-4 shrink-0 text-gray-400" strokeWidth={1.75} />
                  <span className="truncate font-medium text-gray-900">{h.title}</span>
                  {h.subtitle && <span className="truncate text-gray-500">{h.subtitle}</span>}
                  <span className="ml-auto shrink-0 text-xs text-gray-400">{h.type}</span>
                </button>
              </li>
            );
          })}
          {q.trim().length >= 2 && !loading && hits.length === 0 && <li className="px-4 py-6 text-center text-gray-500">Keine Treffer</li>}
          {q.trim().length < 2 && <li className="px-4 py-6 text-center text-gray-400">Mindestens 2 Zeichen eingeben</li>}
        </ul>
      </div>
    </div>
  );
}

export function money(amount: number, currency = "EUR") {
  return new Intl.NumberFormat("de-DE", { style: "currency", currency, maximumFractionDigits: 0 }).format(amount);
}

export function date(iso?: string) {
  if (!iso) return "–";
  return new Intl.DateTimeFormat("de-DE", { dateStyle: "medium" }).format(new Date(iso));
}

export function relative(iso: string) {
  const diff = (new Date(iso).getTime() - Date.now()) / 1000;
  const rtf = new Intl.RelativeTimeFormat("de-DE", { numeric: "auto" });
  const abs = Math.abs(diff);
  if (abs < 3600) return rtf.format(Math.round(diff / 60), "minute");
  if (abs < 86_400) return rtf.format(Math.round(diff / 3600), "hour");
  if (abs < 86_400 * 30) return rtf.format(Math.round(diff / 86_400), "day");
  return rtf.format(Math.round(diff / (86_400 * 30)), "month");
}

export function initials(name: string) {
  return name
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((p) => p[0]?.toUpperCase())
    .join("");
}

export function isWithinDays(iso: string, days: number) {
  return Date.now() - new Date(iso).getTime() < days * 86_400_000;
}
